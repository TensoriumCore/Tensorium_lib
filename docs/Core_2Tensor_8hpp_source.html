154 size_t acc = 0;
155 size_t i = 0;
156
157 for (; i + W - 1 < Rank; i += W) {
158 reg idx = Simd::load(&indices[i]);
159 reg str = Simd::load(&strides[i]);
160 reg prod = Simd::mul(idx, str);

220 indices[d] = tmp % result.dimensions[d_idx];
221 tmp /= result.dimensions[d_idx++];

223 }

226 size_t k = 0;

231 for (size_t w = 0; w < W; ++w) {

233 indices[i] = k_vec[w];

238 for (size_t w = 0; w < W; ++w)
239 vals[w] = t(indices);

242 acc = SimdValue::add(acc, vec);
243 }

258 }
