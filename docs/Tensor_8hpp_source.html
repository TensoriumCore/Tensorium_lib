147 using reg = typename Simd::reg;
148 constexpr size_t W = Simd::width / sizeof(size_t);
149
150 size_t acc = 0;
151 size_t i = 0;
152
153 for (; i + W - 1 < Rank; i += W) {
154 reg idx = Simd::load(&indices[i]);
155 reg str = Simd::load(&strides[i]);
156 reg prod = Simd::mul(idx, str);
157 acc += detail::reduce_sum(prod);
158 }
159 for (; i < Rank; ++i)
160 acc += indices[i] * strides[i];

207 std::array<size_t, Rank - 2> reduced_idx{};

212 for (size_t d = 0; d < Rank; ++d) {
213 if (d == i || d == j) {
214 indices[d] = 0;
215 } else {
216 indices[d] = tmp % result.dimensions[d_idx];
217 tmp /= result.dimensions[d_idx++];
218 }
219 }
220
221 const size_t dim = t.dimensions[i];
222 size_t k = 0;
223 reg acc = SimdValue::zero();
224
225 for (; k + W - 1 < dim; k += W) {
226 alignas(64) size_t k_vec[W];
227 for (size_t w = 0; w < W; ++w) {
228 k_vec[w] = k + w;
229 indices[i] = k_vec[w];
230 indices[j] = k_vec[w];

234 for (size_t w = 0; w < W; ++w)
235 vals[w] = t(indices);
236
237 reg vec = SimdValue::load(vals);
238 acc = SimdValue::add(acc, vec);
239 }

242 for (; k < dim; ++k) {
243 indices[i] = k;
244 indices[j] = k;
245 sum += t(indices);
246 }
247
248 d_idx = 0;
249 for (size_t d = 0; d < Rank; ++d) {
250 if (d != i && d != j)
251 reduced_idx[d_idx++] = indices[d];
252 }
253 result(reduced_idx) = sum;
254 }
