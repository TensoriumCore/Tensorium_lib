62template <typename T>
63inline void compute_rhs_alpha(const BSSNGridSoA<T> &G, Field3D<T> &rhs_alpha, size_t padding = 4,
64 const GaugeParameters<T> &params = {}) {
65 BSSN_PROFILE_KERNEL(Alpha);
66 using namespace tensorium_RG::fd;
67 const T ko_sigma = scaled_ko_sigma(params.ko_sigma); // Adaptive KO6 strength.
68 const bool use_theta = params.use_theta_in_lapse;
69
70 size_t I0, I1, J0, J1, K0, K1;
71 G.domain_bounds(I0, I1, J0, J1, K0, K1);
72
73 const size_t i0 = clamped_lower(I0, padding, I1);
74 const size_t j0 = clamped_lower(J0, padding, J1);
75 const size_t k0 = clamped_lower(K0, padding, K1);
76 const size_t i1 = clamped_upper(I1, padding, I0);
77 const size_t j1 = clamped_upper(J1, padding, J0);
78 const size_t k1 = clamped_upper(K1, padding, K0);
79
80 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
81 return;
82
83 // Constants
84 const double inv_2dx = 1.0 / (2.0 * G.dx);
85 const double inv_2dy = 1.0 / (2.0 * G.dy);
86 const double inv_2dz = 1.0 / (2.0 * G.dz);
87 const ptrdiff_t sx = G.alpha.st.sx;
88 const ptrdiff_t sy = G.alpha.st.sy;
89
90#pragma omp parallel for collapse(2)
91 for (size_t i = i0; i < i1; ++i) {
92 for (size_t j = j0; j < j1; ++j) {
93
94 size_t idx_start = G.alpha.idx(i, j, k0);
95
96 const T *p_alpha = G.alpha.ptr() + idx_start;
97 const T *p_K = G.K.ptr() + idx_start;
98 const T *p_theta = use_theta ? (G.Theta.ptr() + idx_start) : nullptr;
99 const T *p_beta[3] = {G.beta[0].ptr() + idx_start, G.beta[1].ptr() + idx_start,
100 G.beta[2].ptr() + idx_start};
101 T *p_rhs = rhs_alpha.ptr() + idx_start;
102
103 for (size_t k = k0; k < k1; ++k) {
104 const T alpha = *p_alpha;
105 const T K = *p_K;
106 const T theta = use_theta ? *p_theta : T(0);
107 const T bx = *p_beta[0];
108 const T by = *p_beta[1];
109 const T bz = *p_beta[2];
110
111 const T advection = bx * Dx_upwind_ptr(p_alpha, sx, inv_2dx, bx) +
112 by * Dy_upwind_ptr(p_alpha, sy, inv_2dy, by) +
113 bz * Dz_upwind_ptr(p_alpha, inv_2dz, bz);
114
115 const T diss = KO6_axis_ptr(p_alpha, sx) + KO6_axis_ptr(p_alpha, sy) +
116 KO6_axis_ptr(p_alpha, 1);
117 const T diss_scaled = (ko_sigma / G.dx) * diss;
118
119 const T khat_val = use_theta ? Khat(K, theta) : K;
120 *p_rhs = advection - T(2) * alpha * khat_val + diss_scaled;
121
122 // Increments
123 ++p_alpha;
124 ++p_K;
125 ++p_rhs;
126 if (use_theta)
127 ++p_theta;
128 ++p_beta[0];
129 ++p_beta[1];
130 ++p_beta[2];
131 }
132 }
133 }
134}

139template <typename T>
140inline void compute_rhs_beta(const BSSNGridSoA<T> &G, Field3D<T> rhs[3],
141 const GaugeParameters<T> &params = {}, size_t padding = 4) {
142 BSSN_PROFILE_KERNEL(Beta);
143 using namespace tensorium_RG::fd;
144 const T ko_sigma = scaled_ko_sigma(params.ko_sigma); // Shared adaptive filter strength.
145
146 size_t I0, I1, J0, J1, K0, K1;
147 G.domain_bounds(I0, I1, J0, J1, K0, K1);
148
149 const size_t i0 = clamped_lower(I0, padding, I1);
150 const size_t j0 = clamped_lower(J0, padding, J1);
151 const size_t k0 = clamped_lower(K0, padding, K1);
152 const size_t i1 = clamped_upper(I1, padding, I0);
153 const size_t j1 = clamped_upper(J1, padding, J0);
154 const size_t k1 = clamped_upper(K1, padding, K0);
155
156 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
157 return;
158
159 // Constants
160 const double inv_2dx = 1.0 / (2.0 * G.dx);
161 const double inv_2dy = 1.0 / (2.0 * G.dy);
162 const double inv_2dz = 1.0 / (2.0 * G.dz);
163 const ptrdiff_t sx = G.beta[0].st.sx;
164 const ptrdiff_t sy = G.beta[0].st.sy;
165
166#pragma omp parallel for collapse(2)
167 for (size_t i = i0; i < i1; ++i) {
168 for (size_t j = j0; j < j1; ++j) {
169
170 size_t idx_start = G.beta[0].idx(i, j, k0);
171
172 const T *p_beta[3] = {G.beta[0].ptr() + idx_start, G.beta[1].ptr() + idx_start,
173 G.beta[2].ptr() + idx_start};
174 const T *p_B[3] = {G.B[0].ptr() + idx_start, G.B[1].ptr() + idx_start,
175 G.B[2].ptr() + idx_start};
176 T *p_rhs[3] = {rhs[0].ptr() + idx_start, rhs[1].ptr() + idx_start,
177 rhs[2].ptr() + idx_start};
178
179 for (size_t k = k0; k < k1; ++k) {
180 const T bx = *p_beta[0];
181 const T by = *p_beta[1];
182 const T bz = *p_beta[2];
183
184 for (int comp = 0; comp < 3; ++comp) {
185 const T *p_f = p_beta[comp];
186 const T adv = bx * Dx_upwind_ptr(p_f, sx, inv_2dx, bx) +
187 by * Dy_upwind_ptr(p_f, sy, inv_2dy, by) +
188 bz * Dz_upwind_ptr(p_f, inv_2dz, bz);
189
190 const T diss =
191 KO6_axis_ptr(p_f, sx) + KO6_axis_ptr(p_f, sy) + KO6_axis_ptr(p_f, 1);
192 const T diss_scaled = (ko_sigma / G.dx) * diss;
193
194 *p_rhs[comp] = adv + params.beta_B_coeff * (*p_B[comp]) + diss_scaled;
195 }
196
197 // Increments
198 for (int c = 0; c < 3; ++c) {
199 ++p_beta[c];
200 ++p_B[c];
201 ++p_rhs[c];
202 }
203 }
204 }
205 }
206}

218template <typename T>
219inline void compute_rhs_B(const BSSNGridSoA<T> &G, const Field3D<T> rhs_Gamma[3],
220 Field3D<T> rhs_B[3], const GaugeParameters<T> &params = {},
221 size_t padding = 4) {
222 BSSN_PROFILE_KERNEL(B);
223 using namespace tensorium_RG::fd;
224 const T ko_sigma = scaled_ko_sigma(params.ko_sigma); // Adaptive KO6 dial for B^i.
225 size_t I0, I1, J0, J1, K0, K1;
226 G.domain_bounds(I0, I1, J0, J1, K0, K1);
227
228 const size_t i0 = clamped_lower(I0, padding, I1);
229 const size_t j0 = clamped_lower(J0, padding, J1);
230 const size_t k0 = clamped_lower(K0, padding, K1);
231 const size_t i1 = clamped_upper(I1, padding, I0);
232 const size_t j1 = clamped_upper(J1, padding, J0);
233 const size_t k1 = clamped_upper(K1, padding, K0);
234
235 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
236 return;
237
238 const double inv_2dx = 1.0 / (2.0 * G.dx);
239 const double inv_2dy = 1.0 / (2.0 * G.dy);
240 const double inv_2dz = 1.0 / (2.0 * G.dz);
241 const ptrdiff_t sx = G.B[0].st.sx;
242 const ptrdiff_t sy = G.B[0].st.sy;
243 const T eta_coeff = params.effective_eta();
244 const bool use_shift_advection = params.use_shift_advection;
245
246#pragma omp parallel for collapse(2)
247 for (size_t i = i0; i < i1; ++i) {
248 for (size_t j = j0; j < j1; ++j) {
249
250 size_t idx_start = G.B[0].idx(i, j, k0);
251
252 const T *p_beta[3] = {G.beta[0].ptr() + idx_start, G.beta[1].ptr() + idx_start,
253 G.beta[2].ptr() + idx_start};
254 const T *p_B[3] = {G.B[0].ptr() + idx_start, G.B[1].ptr() + idx_start,
255 G.B[2].ptr() + idx_start};
256 const T *p_tildeGamma[3] = {G.tildeGamma[0].ptr() + idx_start,
257 G.tildeGamma[1].ptr() + idx_start,
258 G.tildeGamma[2].ptr() + idx_start};
259 const T *p_Z[3] = {G.Z[0].ptr() + idx_start, G.Z[1].ptr() + idx_start,
260 G.Z[2].ptr() + idx_start};
261 const T *p_rhs_G[3] = {rhs_Gamma[0].ptr() + idx_start, rhs_Gamma[1].ptr() + idx_start,
262 rhs_Gamma[2].ptr() + idx_start};
263 T *p_rhs_B[3] = {rhs_B[0].ptr() + idx_start, rhs_B[1].ptr() + idx_start,
264 rhs_B[2].ptr() + idx_start};
265
266 for (size_t k = k0; k < k1; ++k) {
267 const T bx = *p_beta[0];
268 const T by = *p_beta[1];
269 const T bz = *p_beta[2];
270
271 for (int comp = 0; comp < 3; ++comp) {
272 const T adv_B = use_shift_advection
273 ? (bx * Dx_upwind_ptr(p_B[comp], sx, inv_2dx, bx) +
274 by * Dy_upwind_ptr(p_B[comp], sy, inv_2dy, by) +
275 bz * Dz_upwind_ptr(p_B[comp], inv_2dz, bz))
276 : T(0);
277 const T adv_Gamma = use_shift_advection
278 ? (bx * Dx_upwind_ptr(p_tildeGamma[comp], sx, inv_2dx, bx) +
279 by * Dy_upwind_ptr(p_tildeGamma[comp], sy, inv_2dy, by) +
280 bz * Dz_upwind_ptr(p_tildeGamma[comp], inv_2dz, bz))
281 : T(0);
282
283 const T diss = KO6_axis_ptr(p_B[comp], sx) + KO6_axis_ptr(p_B[comp], sy) +
284 KO6_axis_ptr(p_B[comp], 1);
285 const T diss_scaled = (ko_sigma / G.dx) * diss;
286 const T rhs_gamma_eff = *p_rhs_G[comp] - adv_Gamma;
287 const T z_feedback = -params.kappa_z * (*p_Z[comp]);
288
289 *p_rhs_B[comp] = rhs_gamma_eff + adv_B - eta_coeff * (*p_B[comp]) +
290 z_feedback + diss_scaled;
291 }
292
293 // Increments
294 for (int c = 0; c < 3; ++c) {
295 ++p_beta[c];
296 ++p_B[c];
297 ++p_tildeGamma[c];
298 ++p_Z[c];
299 ++p_rhs_G[c];
300 ++p_rhs_B[c];
301 }
302 }
303 }
304 }
