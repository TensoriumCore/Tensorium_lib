60 const size_t i0 = I0 + 2, i1 = I1 - 2;
61 const size_t j0 = J0 + 2, j1 = J1 - 2;
62 const size_t k0 = K0 + 2, k1 = K1 - 2;

64 const double dx = G.dx;
65 const double dy = G.dy;
66 const double dz = G.dz;

68#pragma omp parallel for collapse(2)
69 for (size_t i = i0; i < i1; ++i) {
70 for (size_t j = j0; j < j1; ++j) {
71 for (size_t k = k0; k < k1; ++k) {
72
73 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
74
75 double g[3][3], gI[3][3];
76 for (int a = 0; a < 3; ++a)
77 for (int b = 0; b < 3; ++b) {
78 g[a][b] = (double)tensorium_RG::sym6_get(G.gamma_tilde, id, a, b);
79 gI[a][b] = (double)tensorium_RG::sym6_get(G.gamma_tilde_inv, id, a, b);
80 }
81
82 double dg[3][3][3];
83 for (int a = 0; a < 3; ++a) {
84 for (int b = a; b < 3; ++b) {
85 const int s = tensorium_RG::sym6_index(a, b);
86 const Field3D<T> &F = G.gamma_tilde[s];
87
88 const double gx = Dx(F, i, j, k, dx);
89 const double gy = Dy(F, i, j, k, dy);
90 const double gz = Dz(F, i, j, k, dz);
91
92 dg[0][a][b] = gx;
93 dg[0][b][a] = gx;
94 dg[1][a][b] = gy;
95 dg[1][b][a] = gy;
96 dg[2][a][b] = gz;
97 dg[2][b][a] = gz;
98 }
99 }
100
101 double Gijk[3][3][3];
102 for (int k1 = 0; k1 < 3; ++k1)
103 for (int i1 = 0; i1 < 3; ++i1)
104 for (int j1 = 0; j1 < 3; ++j1)
105 Gijk[k1][i1][j1] = 0.0;
106
107 for (int kk = 0; kk < 3; ++kk) {
108 for (int ii = 0; ii < 3; ++ii) {
109 for (int jj = ii; jj < 3; ++jj) {
110 double sum = 0.0;
111 for (int ll = 0; ll < 3; ++ll) {
112 const double term =
113 dg[ii][jj][ll] + dg[jj][ii][ll] - dg[ll][ii][jj];
114 sum += gI[kk][ll] * term;
115 }
116 const double val = 0.5 * sum;
117 Gijk[kk][ii][jj] = val;
118 if (ii != jj)
119 Gijk[kk][jj][ii] = val;
120 }
121 }
122 }
123
124 const double chi = (double)G.chi.ptr()[id];
125 const double inv_chi = 1.0 / chi;
126 const double inv_chi2 = inv_chi * inv_chi;
127
128 const double dchi_x = Dx(G.chi, i, j, k, dx);
129 const double dchi_y = Dy(G.chi, i, j, k, dy);
130 const double dchi_z = Dz(G.chi, i, j, k, dz);
131
132 const double d2chi_xx = Dxx(G.chi, i, j, k, dx);
133 const double d2chi_yy = Dyy(G.chi, i, j, k, dy);
134 const double d2chi_zz = Dzz(G.chi, i, j, k, dz);
135 const double d2chi_xy = Dxy4(G.chi, i, j, k, dx, dy);
136 const double d2chi_xz = Dxz4(G.chi, i, j, k, dx, dz);
137 const double d2chi_yz = Dyz4(G.chi, i, j, k, dy, dz);
138
139 double cov_dchi[3][3];
140
141 cov_dchi[0][0] = d2chi_xx - (Gijk[0][0][0] * dchi_x + Gijk[1][0][0] * dchi_y +
142 Gijk[2][0][0] * dchi_z);
143 cov_dchi[0][1] = d2chi_xy - (Gijk[0][0][1] * dchi_x + Gijk[1][0][1] * dchi_y +
144 Gijk[2][0][1] * dchi_z);
145 cov_dchi[0][2] = d2chi_xz - (Gijk[0][0][2] * dchi_x + Gijk[1][0][2] * dchi_y +
146 Gijk[2][0][2] * dchi_z);
147
148 cov_dchi[1][0] = cov_dchi[0][1];
149 cov_dchi[1][1] = d2chi_yy - (Gijk[0][1][1] * dchi_x + Gijk[1][1][1] * dchi_y +
150 Gijk[2][1][1] * dchi_z);
151 cov_dchi[1][2] = d2chi_yz - (Gijk[0][1][2] * dchi_x + Gijk[1][1][2] * dchi_y +
152 Gijk[2][1][2] * dchi_z);
153
154 cov_dchi[2][0] = cov_dchi[0][2];
155 cov_dchi[2][1] = cov_dchi[1][2];
156 cov_dchi[2][2] = d2chi_zz - (Gijk[0][2][2] * dchi_x + Gijk[1][2][2] * dchi_y +
157 Gijk[2][2][2] * dchi_z);
158
159 const double grad2 = gI[0][0] * dchi_x * dchi_x + gI[1][1] * dchi_y * dchi_y +
160 gI[2][2] * dchi_z * dchi_z + 2.0 * gI[0][1] * dchi_x * dchi_y +
161 2.0 * gI[0][2] * dchi_x * dchi_z +
162 2.0 * gI[1][2] * dchi_y * dchi_z;
163
164 const double lap = gI[0][0] * cov_dchi[0][0] + gI[1][1] * cov_dchi[1][1] +
165 gI[2][2] * cov_dchi[2][2] + 2.0 * gI[0][1] * cov_dchi[0][1] +
166 2.0 * gI[0][2] * cov_dchi[0][2] +
167 2.0 * gI[1][2] * cov_dchi[1][2];
168
169 double R_chi[3][3];
170 const double dchi[3] = {dchi_x, dchi_y, dchi_z};
171
172 for (int a = 0; a < 3; ++a)
173 for (int b = a; b < 3; ++b) {
174 const double term1 = 0.5 * inv_chi * (cov_dchi[a][b] + g[a][b] * lap);
175 const double term2 = -0.25 * inv_chi2 * (dchi[a] * dchi[b]);
176 const double term3 = -0.75 * inv_chi2 * g[a][b] * grad2;
177 const double val = term1 + term2 + term3;
178 R_chi[a][b] = val;
179 if (a != b)
180 R_chi[b][a] = val;
181 }
182
183 const double Gt[3] = {(double)G.tildeGamma[0].ptr()[id],
184 (double)G.tildeGamma[1].ptr()[id],
185 (double)G.tildeGamma[2].ptr()[id]};
186
187 double dGt[3][3];

189 dGt[0][1] = Dy(G.tildeGamma[0], i, j, k, dy);
190 dGt[0][2] = Dz(G.tildeGamma[0], i, j, k, dz);
191 dGt[1][0] = Dx(G.tildeGamma[1], i, j, k, dx);
192 dGt[1][1] = Dy(G.tildeGamma[1], i, j, k, dy);
193 dGt[1][2] = Dz(G.tildeGamma[1], i, j, k, dz);
194 dGt[2][0] = Dx(G.tildeGamma[2], i, j, k, dx);
195 dGt[2][1] = Dy(G.tildeGamma[2], i, j, k, dy);
196 dGt[2][2] = Dz(G.tildeGamma[2], i, j, k, dz);
197
198 double R_tilde[3][3];
199
200 for (int a = 0; a < 3; ++a)
201 for (int b = a; b < 3; ++b) {
202
203 const int s_ab = tensorium_RG::sym6_index(a, b);
204 const Field3D<T> &Gab = G.gamma_tilde[s_ab];
205
206 const double d2xx = Dxx(Gab, i, j, k, dx);
207 const double d2yy = Dyy(Gab, i, j, k, dy);
208 const double d2zz = Dzz(Gab, i, j, k, dz);
209 const double d2xy = Dxy4(Gab, i, j, k, dx, dy);
210 const double d2xz = Dxz4(Gab, i, j, k, dx, dz);
211 const double d2yz = Dyz4(Gab, i, j, k, dy, dz);
212
213 const double lap_gab = gI[0][0] * d2xx + gI[1][1] * d2yy + gI[2][2] * d2zz +
214 2.0 * gI[0][1] * d2xy + 2.0 * gI[0][2] * d2xz +
215 2.0 * gI[1][2] * d2yz;
216
217 const double term1 = -0.5 * lap_gab;
218 double term2 = 0.0;
219 for (int kk = 0; kk < 3; ++kk)
220 term2 += 0.5 * (g[a][kk] * dGt[kk][b] + g[b][kk] * dGt[kk][a]);
221
222 const double term3 = 0.5 * (Gt[0] * Gijk[0][a][b] + Gt[1] * Gijk[1][a][b] +
223 Gt[2] * Gijk[2][a][b]);
224
225 double term4 = 0.0;
226 for (int m = 0; m < 3; ++m)
227 for (int n = 0; n < 3; ++n) {
228 double sum = 0.0;
229 for (int kk = 0; kk < 3; ++kk) {
230 sum += 2.0 * Gijk[kk][m][a] * Gijk[kk][n][b];
231 sum -= Gijk[kk][a][b] * Gijk[kk][m][n];
232 }
233 term4 += gI[m][n] * sum;
234 }
235
236 const double val = term1 + term2 + term3 + term4;
237 R_tilde[a][b] = val;
238 if (a != b)
239 R_tilde[b][a] = val;
240 }
241
242 Ricci6[XX].ptr()[id] = (T)(R_tilde[0][0] + R_chi[0][0]);
243 Ricci6[XY].ptr()[id] = (T)(R_tilde[0][1] + R_chi[0][1]);
244 Ricci6[XZ].ptr()[id] = (T)(R_tilde[0][2] + R_chi[0][2]);
245 Ricci6[YY].ptr()[id] = (T)(R_tilde[1][1] + R_chi[1][1]);
246 Ricci6[YZ].ptr()[id] = (T)(R_tilde[1][2] + R_chi[1][2]);
247 Ricci6[ZZ].ptr()[id] = (T)(R_tilde[2][2] + R_chi[2][2]);
248 }
249 }
250 }
