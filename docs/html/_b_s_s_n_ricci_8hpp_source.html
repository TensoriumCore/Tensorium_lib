63 const size_t i0 = I0 + 3, i1 = I1 - 3;
64 const size_t j0 = J0 + 3, j1 = J1 - 3;
65 const size_t k0 = K0 + 3, k1 = K1 - 3;
66
67 const double inv_12dx = 1.0 / (60.0 * G.dx);
68 const double inv_12dy = 1.0 / (60.0 * G.dy);
69 const double inv_12dz = 1.0 / (60.0 * G.dz);
70 const double inv_12dx2 = 1.0 / (180.0 * G.dx * G.dx);
71 const double inv_12dy2 = 1.0 / (180.0 * G.dy * G.dy);
72 const double inv_12dz2 = 1.0 / (180.0 * G.dz * G.dz);
73 const double inv_144dxdy = 1.0 / (144.0 * G.dx * G.dy);
74 const double inv_144dxdz = 1.0 / (144.0 * G.dx * G.dz);
75 const double inv_144dydz = 1.0 / (144.0 * G.dy * G.dz);
76
77 const ptrdiff_t sx = G.gamma_tilde[0].st.sx;
78 const ptrdiff_t sy = G.gamma_tilde[0].st.sy;

80#pragma omp parallel for collapse(2)
81 for (size_t i = i0; i < i1; ++i) {
82 for (size_t j = j0; j < j1; ++j) {
83
84 size_t idx_start = G.gamma_tilde[XX].idx(i, j, k0);
85
86 const T *p_gam[6];
87 const T *p_gam_inv[6];
88 const T *p_Gt[3];
89 T *p_Ricci[6];
90
91 for (int s = 0; s < 6; ++s) {
92 p_gam[s] = G.gamma_tilde[s].ptr() + idx_start;
93 p_gam_inv[s] = G.gamma_tilde_inv[s].ptr() + idx_start;
94 p_Ricci[s] = Ricci6[s].ptr() + idx_start;
95 }
96 for (int c = 0; c < 3; ++c)
97 p_Gt[c] = G.tildeGamma[c].ptr() + idx_start;
98
99 const T *p_chi = G.chi.ptr() + idx_start;
100
101 for (size_t k = k0; k < k1; ++k) {
102
103 double g[3][3], gI[3][3];
104 g[0][0] = *p_gam[0];
105 g[0][1] = *p_gam[1];
106 g[0][2] = *p_gam[2];
107 g[1][1] = *p_gam[3];
108 g[1][2] = *p_gam[4];
109 g[2][2] = *p_gam[5];
110 g[1][0] = g[0][1];
111 g[2][0] = g[0][2];
112 g[2][1] = g[1][2];
113
114 gI[0][0] = *p_gam_inv[0];
115 gI[0][1] = *p_gam_inv[1];
116 gI[0][2] = *p_gam_inv[2];
117 gI[1][1] = *p_gam_inv[3];
118 gI[1][2] = *p_gam_inv[4];
119 gI[2][2] = *p_gam_inv[5];
120 gI[1][0] = gI[0][1];
121 gI[2][0] = gI[0][2];
122 gI[2][1] = gI[1][2];
123
124 double dg[3][3][3];
125 const int map_s[3][3] = {{0, 1, 2}, {1, 3, 4}, {2, 4, 5}};
126
127 for (int a = 0; a < 3; ++a) {
128 for (int b = a; b < 3; ++b) {
129 const T *p = p_gam[map_s[a][b]];
130 const double dx_g = Dx_ptr(p, sx, inv_12dx);
131 const double dy_g = Dy_ptr(p, sy, inv_12dy);
132 const double dz_g = Dz_ptr(p, inv_12dz);
133 dg[0][a][b] = dx_g;
134 dg[0][b][a] = dx_g;
135 dg[1][a][b] = dy_g;
136 dg[1][b][a] = dy_g;
137 dg[2][a][b] = dz_g;
138 dg[2][b][a] = dz_g;
139 }
140 }
141
142 double G_up[3][3][3];
143 for (int kk = 0; kk < 3; ++kk) {
144 for (int ii = 0; ii < 3; ++ii) {
145 for (int jj = ii; jj < 3; ++jj) {
146 double sum = 0.0;
147 for (int ll = 0; ll < 3; ++ll) {
148 sum +=
149 gI[kk][ll] * (dg[ii][jj][ll] + dg[jj][ii][ll] - dg[ll][ii][jj]);
150 }
151 double val = 0.5 * sum;
152 G_up[kk][ii][jj] = val;
153 if (ii != jj)
154 G_up[kk][jj][ii] = val;
155 }
156 }
157 }
158
159 double R_chi[3][3];
160 {
161 const double chi = *p_chi;
162 const double inv_chi = 1.0 / chi;
163 const double inv_chi2 = inv_chi * inv_chi;
164
165 const double dchi[3] = {Dx_ptr(p_chi, sx, inv_12dx),
166 Dy_ptr(p_chi, sy, inv_12dy), Dz_ptr(p_chi, inv_12dz)};

168 double hess_chi[3][3];
169 double lap_chi = 0.0;
170 double grad_chi_sq = 0.0;
171
172 const double d2_xx = Dxx_ptr(p_chi, sx, inv_12dx2);
173 const double d2_yy = Dyy_ptr(p_chi, sy, inv_12dy2);
174 const double d2_zz = Dzz_ptr(p_chi, inv_12dz2);
175 const double d2_xy = Dxy4_ptr(p_chi, sx, sy, inv_144dxdy);
176 const double d2_xz = Dxz4_ptr(p_chi, sx, inv_144dxdz);
177 const double d2_yz = Dyz4_ptr(p_chi, sy, inv_144dydz);
178 const double d2chi_raw[3][3] = {
179 {d2_xx, d2_xy, d2_xz}, {d2_xy, d2_yy, d2_yz}, {d2_xz, d2_yz, d2_zz}};
180
181 for (int a = 0; a < 3; ++a) {
182 for (int b = a; b < 3; ++b) {
183 double conn = 0.0;
184 for (int m = 0; m < 3; ++m)
185 conn += G_up[m][a][b] * dchi[m];
186
187 double h = d2chi_raw[a][b] - conn;
188 hess_chi[a][b] = h;
189 if (a != b)
190 hess_chi[b][a] = h;
191 }
192 }
193
194 for (int a = 0; a < 3; ++a) {
195 for (int b = 0; b < 3; ++b) {
196 lap_chi += gI[a][b] * hess_chi[a][b];

198 }
199 }
200
201 for (int a = 0; a < 3; ++a) {
202 for (int b = a; b < 3; ++b) {
203 double val = 0.5 * inv_chi * (hess_chi[a][b] + g[a][b] * lap_chi) -
204 0.25 * inv_chi2 * dchi[a] * dchi[b] -
205 0.75 * inv_chi2 * g[a][b] * grad_chi_sq;
206 R_chi[a][b] = val;
207 if (a != b)
208 R_chi[b][a] = val;
209 }
210 }
211 }

213 double R_tilde[3][3];

216 for (int c = 0; c < 3; ++c) {
217 const T *p = p_Gt[c];

222 const double Gt_vec[3] = {*p_Gt[0], *p_Gt[1], *p_Gt[2]};
223
224 double G_low[3][3][3];
225 for (int k = 0; k < 3; ++k)
226 for (int i = 0; i < 3; ++i)
227 for (int j = i; j < 3; ++j) {
228 double s = 0.0;
229 for (int m = 0; m < 3; ++m)

231 G_low[k][i][j] = s;
232 if (i != j)
233 G_low[k][j][i] = s;
234 }
235
236 for (int a = 0; a < 3; ++a) {
237 for (int b = a; b < 3; ++b) {
238 const T *p_gab = p_gam[map_s[a][b]];
239 double lap_g = gI[0][0] * Dxx_ptr(p_gab, sx, inv_12dx2) +
240 gI[1][1] * Dyy_ptr(p_gab, sy, inv_12dy2) +
241 gI[2][2] * Dzz_ptr(p_gab, inv_12dz2) +
242 2.0 * (gI[0][1] * Dxy4_ptr(p_gab, sx, sy, inv_144dxdy) +
243 gI[0][2] * Dxz4_ptr(p_gab, sx, inv_144dxdz) +
244 gI[1][2] * Dyz4_ptr(p_gab, sy, inv_144dydz));
245
246 double t2 = 0.0;
247 for (int k = 0; k < 3; ++k)
248 t2 += 0.5 * (g[a][k] * dGt[k][b] + g[b][k] * dGt[k][a]);
249
250 double t3 = 0.0;
251 for (int k = 0; k < 3; ++k)
252 t3 += 0.5 * Gt_vec[k] * G_low[k][a][b];
253
254 double t4 = 0.0;
255 for (int l = 0; l < 3; ++l) {
256 for (int m = 0; m < 3; ++m) {
257 double g_inv_lm = gI[l][m];
258 double inner = 0.0;
259 for (int k = 0; k < 3; ++k) {
260 inner += 2.0 * G_up[k][l][a] * G_low[k][b][m] -
261 G_up[k][a][b] * G_low[k][l][m];
262 }
263 t4 += g_inv_lm * inner;
264 }
265 }
266
267 double val = -0.5 * lap_g + t2 + t3 + t4;
268 R_tilde[a][b] = val;
269 if (a != b)
270 R_tilde[b][a] = val;
271 }
272 }
273 }
274
275 for (int s = 0; s < 6; ++s) {
276 int a = (s == 0 || s == 1 || s == 2) ? 0 : ((s == 3 || s == 4) ? 1 : 2);
277 int b = (s == 0) ? 0 : ((s == 1 || s == 3) ? 1 : 2);
278 *p_Ricci[s] = (T)(R_tilde[a][b] + R_chi[a][b]);
279 }
280
281 for (int s = 0; s < 6; ++s) {
282 ++p_gam[s];
283 ++p_gam_inv[s];
284 ++p_Ricci[s];
285 }
286 for (int c = 0; c < 3; ++c)
287 ++p_Gt[c];
288 ++p_chi;
289 }
290 }
291 }
