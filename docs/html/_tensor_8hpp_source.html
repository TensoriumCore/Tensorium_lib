163 using Simd = simd::SimdTraits<size_t, DefaultISA>;
164 using reg = typename Simd::reg;
165 constexpr size_t W = Simd::width / sizeof(size_t);
166
167 size_t acc = 0;
168 size_t i = 0;
169
170 for (; i + W - 1 < Rank; i += W) {
171 reg idx = Simd::load(&indices[i]);
172 reg str = Simd::load(&strides[i]);
173 reg prod = Simd::mul(idx, str);
174 acc += detail::reduce_sum(prod);
175 }
176 for (; i < Rank; ++i)
177 acc += indices[i] * strides[i];
178

207 contract_simd(const Tensor<K, Rank> &t, size_t i, size_t j) const {
208 static_assert(Rank >= 2, "Cannot contract tensor of rank < 2");
209 assert(i < Rank && j < Rank && i != j);
210 assert(t.dimensions[i] == t.dimensions[j]);
211
212 using SimdIndex = simd::SimdTraits<size_t, DefaultISA>;
213 using SimdValue = simd::SimdTraits<K, DefaultISA>;
214 using reg = typename SimdValue::reg;
215 constexpr size_t W = SimdValue::width;
216
217 std::array<size_t, Rank - 2> new_dims;
218 size_t d_idx = 0;
219 for (size_t d = 0; d < Rank; ++d) {
220 if (d != i && d != j)
221 new_dims[d_idx++] = t.dimensions[d];
222 }
223 Tensor<K, Rank - 2> result(new_dims);
224
225 std::array<size_t, Rank> indices{};
226 std::array<size_t, Rank - 2> reduced_idx{};
227
228 for (size_t flat = 0; flat < result.data.size(); ++flat) {
229 size_t tmp = flat;
230 d_idx = 0;
231 for (size_t d = 0; d < Rank; ++d) {
232 if (d == i || d == j) {
233 indices[d] = 0;
234 } else {
235 indices[d] = tmp % result.dimensions[d_idx];
236 tmp /= result.dimensions[d_idx++];
237 }
238 }
239
240 const size_t dim = t.dimensions[i];
241 size_t k = 0;
242 reg acc = SimdValue::zero();
243
244 for (; k + W - 1 < dim; k += W) {
245 alignas(64) size_t k_vec[W];
246 for (size_t w = 0; w < W; ++w) {
247 k_vec[w] = k + w;
248 indices[i] = k_vec[w];
249 indices[j] = k_vec[w];
250 }
251
252 alignas(64) K vals[W];
253 for (size_t w = 0; w < W; ++w)
254 vals[w] = t(indices);
255
256 reg vec = SimdValue::load(vals);
257 acc = SimdValue::add(acc, vec);
258 }
259
260 K sum = detail::reduce_sum(acc);
261 for (; k < dim; ++k) {
262 indices[i] = k;
263 indices[j] = k;
264 sum += t(indices);
265 }
266
267 d_idx = 0;
268 for (size_t d = 0; d < Rank; ++d) {
269 if (d != i && d != j)
270 reduced_idx[d_idx++] = indices[d];
271 }
272 result(reduced_idx) = sum;
273 }
274
275 return result;
