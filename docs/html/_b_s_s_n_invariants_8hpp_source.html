84inline double det3(const double m[3][3]) {
85 return m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[1][2]) -
86 m[0][1] * (m[0][1] * m[2][2] - m[0][2] * m[1][2]) +
87 m[0][2] * (m[0][1] * m[1][2] - m[0][2] * m[1][1]);
88}

104template <typename T>
105inline InvariantStats compute_invariant_stats(const BSSNGridSoA<T> &G, size_t padding = 4,
106 double r_min = 0.0,
107 double r_max = std::numeric_limits<double>::max(),
108 double xc = 0.0, double yc = 0.0, double zc = 0.0) {
109 InvariantStats stats;
110
111 size_t I0, I1, J0, J1, K0, K1;
112 G.domain_bounds(I0, I1, J0, J1, K0, K1);
113
114 // Samples are gathered strictly inside the physical domain. We skip
115 // "padding" cells from the computational boundary to (1) avoid touching
116 // halo points (domain_bounds already drops ng cells) and (2) guarantee
117 // that the 4th-order stencils (which require ±2 points) operate on valid
118 // data. The default padding value of 4 matches the standard halo width.
119 const size_t guard = std::max<size_t>(padding, size_t(2));
120 const size_t i0 = std::min(I0 + guard, I1);
121 const size_t j0 = std::min(J0 + guard, J1);
122 const size_t k0 = std::min(K0 + guard, K1);
123 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
124 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
125 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
126
127 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
128 return stats;
129
130 double det_sum = 0.0, det_sq = 0.0;
131 double tr_sum = 0.0, tr_sq = 0.0;
132 double metric_sum = 0.0, metric_sq = 0.0;
133 double gamma_sum = 0.0, gamma_sq = 0.0;
134
135 for (size_t i = i0; i < i1; ++i)
136 for (size_t j = j0; j < j1; ++j)
137 for (size_t k = k0; k < k1; ++k) {
138 double x, y, z;
139 G.coords(i, j, k, x, y, z);
140 const double X = x - xc;
141 const double Y = y - yc;
142 const double Z = z - zc;
143 const double r = std::sqrt(X * X + Y * Y + Z * Z);
144 if (r <= r_min || r >= r_max)
145 continue;
146
147 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
148
149 double g[3][3];
150 double ginv[3][3];
151 double A[3][3];
152 for (int a = 0; a < 3; ++a)
153 for (int b = a; b < 3; ++b) {
154 const double gval = tensorium_RG::sym6_get(G.gamma_tilde, id, a, b);
155 g[a][b] = gval;
156 g[b][a] = gval;
157 const double gIval = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, a, b);
158 ginv[a][b] = gIval;
159 ginv[b][a] = gIval;
160 const double Aval = tensorium_RG::sym6_get(G.A_tilde, id, a, b);
161 A[a][b] = Aval;
162 A[b][a] = Aval;
163 }
164
165 const double det_dev = std::abs(detail::det3(g) - 1.0);
166 if (det_dev > stats.max_det_deviation) {
167 stats.max_det_deviation = det_dev;
168 stats.det_location = SampleLocation{i, j, k, x, y, z};
169 }
170 det_sum += det_dev;
171 det_sq += det_dev * det_dev;
172
173 double traceA = 0.0;
174 for (int a = 0; a < 3; ++a)
175 for (int b = 0; b < 3; ++b)
176 traceA += ginv[a][b] * A[a][b];
177 const double abs_trace = std::abs(traceA);
178 if (abs_trace > stats.max_trace_A) {
179 stats.max_trace_A = abs_trace;
180 stats.trace_location = SampleLocation{i, j, k, x, y, z};
181 }
182 tr_sum += abs_trace;
183 tr_sq += abs_trace * abs_trace;
184
185 double metric_err = 0.0;
186 for (int a = 0; a < 3; ++a)
187 for (int b = 0; b < 3; ++b) {
188 double sum = 0.0;
189 for (int c = 0; c < 3; ++c)
190 sum += g[a][c] * ginv[c][b];
191 const double delta = (a == b) ? 1.0 : 0.0;
192 metric_err = std::max(metric_err, std::abs(sum - delta));
193 }
194 if (metric_err > stats.max_metric_identity) {
195 stats.max_metric_identity = metric_err;
196 stats.metric_location = SampleLocation{i, j, k, x, y, z};
197 }
198 metric_sum += metric_err;
199 metric_sq += metric_err * metric_err;
200
201 T div_gamma[3];
202 metric_inverse_divergence(G, i, j, k, div_gamma);
203 for (int a = 0; a < 3; ++a) {
204 const double gamma_constraint =
205 (double)G.tildeGamma[a].ptr()[id] + (double)div_gamma[a];
206 const double abs_gamma = std::abs(gamma_constraint);
207 if (abs_gamma > stats.max_gamma_constraint) {
208 stats.max_gamma_constraint = abs_gamma;
209 stats.gamma_location = SampleLocation{i, j, k, x, y, z};
210 }
211 gamma_sum += abs_gamma;
212 gamma_sq += abs_gamma * abs_gamma;
213 }
214
215 const double chi = (double)G.chi.ptr()[id];
216 if (chi < stats.min_chi) {
217 stats.min_chi = chi;
218 stats.chi_location = SampleLocation{i, j, k, x, y, z};
219 }
220
221 ++stats.samples;
222 }
223
224 if (stats.samples > 0) {
225 const double inv_n = 1.0 / (double)stats.samples;
226 stats.mean_det_deviation = det_sum * inv_n;
227 stats.mean_trace_A = tr_sum * inv_n;
228 stats.mean_metric_identity = metric_sum * inv_n;
229 stats.mean_gamma_constraint = gamma_sum * inv_n;
230
231 stats.l2_det_deviation = std::sqrt(det_sq * inv_n);
232 stats.l2_trace_A = std::sqrt(tr_sq * inv_n);
233 stats.l2_metric_identity = std::sqrt(metric_sq * inv_n);
234 stats.l2_gamma_constraint = std::sqrt(gamma_sq * inv_n);
235 }
236
237 return stats;
238}

248template <typename T>
249inline InvariantTolerances compute_invariant_tolerances(const BSSNGridSoA<T> &G,
250 double scheme_order = 4.0,
251 double safety_factor = 4.0) {
252 const double hx = (double)G.dx;
253 const double hy = (double)G.dy;
254 const double hz = (double)G.dz;
255 const double h = std::max({hx, hy, hz});
256 const double Lx = G.dims.nx * hx;
257 const double Ly = G.dims.ny * hy;
258 const double Lz = G.dims.nz * hz;
259 const double L = std::max({Lx, Ly, Lz, h});
260 const double ratio = (L > 0.0) ? h / L : 1.0;
261 const double scheme_error = std::pow(ratio, scheme_order);
262 const double machine = std::sqrt(std::numeric_limits<double>::epsilon());
263
264 auto make_tol = [&](double weight) {
265 const double scaled = safety_factor * scheme_error * weight;
266 return std::max(machine * weight, scaled);
267 };
268
269 InvariantTolerances tol;
270 tol.det_tol = make_tol(1.0);
271 tol.trace_tol = make_tol(1.0);
272 tol.metric_tol = make_tol(1.5);
273 tol.gamma_tol = make_tol(2.0);
274 tol.chi_tol = machine;
275 const double chi_override = chi_tolerance_override();
276 if (!std::isnan(chi_override))
277 tol.chi_tol = chi_override;
278 return tol;
279}

288template <typename T>
289inline void assert_invariants(const BSSNGridSoA<T> &G, const char *stage, size_t padding,
290 const InvariantTolerances &tol, bool throw_on_violation = true) {
291 const auto stats = compute_invariant_stats(G, padding);
292 if (stats.samples == 0)
293 return;
294
295 auto describe = [](const SampleLocation &p) {
296 std::ostringstream oss;
297 oss << "(i=" << p.i << ", j=" << p.j << ", k=" << p.k << ")"
298 << " @ (" << p.x << ", " << p.y << ", " << p.z << ')';
299 return oss.str();
300 };
301
302 auto handle_violation = [&](const std::string &quantity, double value, double bound, double l2,
303 double mean, const SampleLocation &loc) {
304 std::ostringstream oss;
305 oss << "[BSSN::" << (stage ? stage : "unknown") << "] " << quantity
306 << " violation: max=" << value << " (tol=" << bound << ")"
307 << " L2=" << l2 << " mean=" << mean << " at " << describe(loc);
308 if (throw_on_violation)
309 throw std::runtime_error(oss.str());
310 std::cerr << "[WARN] " << oss.str() << '\n';
311 };
312
313 if (stats.max_det_deviation > tol.det_tol)
314 handle_violation("det(gamma_tilde)", stats.max_det_deviation, tol.det_tol,
315 stats.l2_det_deviation, stats.mean_det_deviation, stats.det_location);
316 if (stats.max_trace_A > tol.trace_tol)
317 handle_violation("tr(A_tilde)", stats.max_trace_A, tol.trace_tol, stats.l2_trace_A,
318 stats.mean_trace_A, stats.trace_location);
319 if (stats.max_metric_identity > tol.metric_tol)
320 handle_violation("gamma * gamma^{-1}", stats.max_metric_identity, tol.metric_tol,
321 stats.l2_metric_identity, stats.mean_metric_identity, stats.metric_location);
322 if (stats.max_gamma_constraint > tol.gamma_tol)
323 handle_violation("Gamma coherence (tildeGamma + ∂_j g^{ij})", stats.max_gamma_constraint,
324 tol.gamma_tol, stats.l2_gamma_constraint, stats.mean_gamma_constraint,
325 stats.gamma_location);
326 if (!(stats.min_chi > tol.chi_tol))
327 handle_violation("chi", stats.min_chi, tol.chi_tol, 0.0, stats.min_chi, stats.chi_location);
328}
