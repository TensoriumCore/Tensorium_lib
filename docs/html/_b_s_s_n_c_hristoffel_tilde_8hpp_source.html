51template <typename T>
52inline void compute_tildeGamma_symbols(const BSSNGridSoA<T> &G, size_t i, size_t j, size_t k,
53 T (&Gamma)[3][3][3]) {
54 using tensorium_RG::fd::Dx;
55 using tensorium_RG::fd::Dy;
56 using tensorium_RG::fd::Dz;
57
58 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
59
60 auto d_g = [&](int dir, int a, int b) -> T {
61 if (a > b)
62 std::swap(a, b);
63 const Field3D<T> &F = (a == 0 && b == 0) ? G.gamma_tilde[XX]
64 : (a == 0 && b == 1) ? G.gamma_tilde[XY]
65 : (a == 0 && b == 2) ? G.gamma_tilde[XZ]
66 : (a == 1 && b == 1) ? G.gamma_tilde[YY]
67 : (a == 1 && b == 2) ? G.gamma_tilde[YZ]
68 : G.gamma_tilde[ZZ];
69 return (dir == 0) ? Dx(F, i, j, k, G.dx)
70 : (dir == 1) ? Dy(F, i, j, k, G.dy)
71 : Dz(F, i, j, k, G.dz);
72 };
73
74 for (int up = 0; up < 3; ++up)
75 for (int lo1 = 0; lo1 < 3; ++lo1)
76 for (int lo2 = 0; lo2 < 3; ++lo2) {
77 T sum = T(0);
78 for (int ell = 0; ell < 3; ++ell) {
79 const T ginv = sym6_get(G.gamma_tilde_inv, id, up, ell);
80 const T term = d_g(lo1, lo2, ell) + d_g(lo2, lo1, ell) - d_g(ell, lo1, lo2);
81 sum += ginv * term;
82 }
83 Gamma[up][lo1][lo2] = T(0.5) * sum;
84 }
85}

184template <typename T>
185inline void compute_tildeGamma_symbols_ref(const BSSNGridSoA<T> &G, size_t i, size_t j, size_t k,
186 T (&Gamma)[3][3][3]) {
187 T grad_metric[3][6];
188 for (int dir = 0; dir < 3; ++dir)
189 for (int comp = 0; comp < 6; ++comp)
190 grad_metric[dir][comp] = detail::diff_gamma_component(G, i, j, k, dir, comp);
191
192 auto grad = [&](int dir, int a, int b) -> T {
193 const int idx = tensorium_RG::sym6_index(a, b);
194 return grad_metric[dir][idx];
195 };
196
197 T ginv[3][3];
198 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
199 for (int a = 0; a < 3; ++a)
200 for (int b = 0; b < 3; ++b)
201 ginv[a][b] = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, a, b);
202
203 for (int up = 0; up < 3; ++up)
204 for (int lo1 = 0; lo1 < 3; ++lo1)
205 for (int lo2 = 0; lo2 < 3; ++lo2) {
206 const T q0 = grad(lo1, 0, lo2) + grad(lo2, 0, lo1) - grad(0, lo1, lo2);
207 const T q1 = grad(lo1, 1, lo2) + grad(lo2, 1, lo1) - grad(1, lo1, lo2);
208 const T q2 = grad(lo1, 2, lo2) + grad(lo2, 2, lo1) - grad(2, lo1, lo2);
209 Gamma[up][lo1][lo2] = T(0.5) * (ginv[up][0] * q0 + ginv[up][1] * q1 + ginv[up][2] * q2);
210 }
211}

213template <typename T>
214inline void validate_tildeGamma_symbols(const BSSNGridSoA<T> &G, size_t i, size_t j, size_t k,
215 const T (&Gamma)[3][3][3]) {
216 const double spacing = std::max(detail::validation_spacing(G), 1e-12);
217 const double symmetry_tol = 1e-11;
218 const double contracted_abs_c0 = 5e-4;
219 const double contracted_rel_c1 = 5e-3;
220 const double ref_tol = 2e-7 / spacing;
221
222 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
223
224 T Gamma_ref[3][3][3];
225 compute_tildeGamma_symbols_ref(G, i, j, k, Gamma_ref);
226
227 double max_diff = 0.0;
228 int max_idx[3] = {0, 0, 0};
229
230 for (int up = 0; up < 3; ++up)
231 for (int lo1 = 0; lo1 < 3; ++lo1)
232 for (int lo2 = 0; lo2 < 3; ++lo2) {
233 const double sym_err = std::abs(double(Gamma[up][lo1][lo2] - Gamma[up][lo2][lo1]));
234 if (lo1 <= lo2 && sym_err > symmetry_tol)
235 detail::report_validation_failure("symmetry", i, j, k, up, lo1, lo2,
236 Gamma[up][lo1][lo2], Gamma[up][lo2][lo1],
237 symmetry_tol);
238
239 const double diff =
240 std::abs(double(Gamma[up][lo1][lo2] - Gamma_ref[up][lo1][lo2]));
241 if (diff > max_diff) {
242 max_diff = diff;
243 max_idx[0] = up;
244 max_idx[1] = lo1;
245 max_idx[2] = lo2;
246 }
247 }
248
249 if (max_diff > ref_tol)
250 detail::report_validation_failure("ref_symbols", i, j, k, max_idx[0], max_idx[1], max_idx[2],
251 Gamma[max_idx[0]][max_idx[1]][max_idx[2]],
252 Gamma_ref[max_idx[0]][max_idx[1]][max_idx[2]], ref_tol);
253
254 T contracted[3] = {T(0), T(0), T(0)};
255 for (int up = 0; up < 3; ++up) {
256 T sum = T(0);
257 for (int lo1 = 0; lo1 < 3; ++lo1)
258 for (int lo2 = 0; lo2 < 3; ++lo2) {
259 const T ginv = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, lo1, lo2);
260 sum += ginv * Gamma[up][lo1][lo2];
261 }
262 contracted[up] = sum;
263 }
264
265 T div[3];
266 metric_inverse_divergence(G, i, j, k, div);
267 T contracted_ref[3] = {-div[0], -div[1], -div[2]};
268
269 for (int comp = 0; comp < 3; ++comp) {
270 const double lhs = double(contracted[comp]);
271 const double rhs = double(contracted_ref[comp]);
272 const double diff = std::abs(lhs - rhs);
273 const double scale = std::max({1.0, std::abs(lhs), std::abs(rhs)});
274 const double tol_abs = contracted_abs_c0 * spacing * spacing;
275 const double tol_rel = contracted_rel_c1 * scale;
276 const double rel = diff / scale;
277 if (diff > tol_abs + tol_rel)
278 detail::update_contracted_warning_state(i, j, k, comp, diff, rel, lhs, rhs);
279 }
280}

321inline void validate_gamma_source_term(const BSSNGridSoA<T> &G, size_t i, size_t j, size_t k,

323 const T (&A_up_ref)[3][3]) {
