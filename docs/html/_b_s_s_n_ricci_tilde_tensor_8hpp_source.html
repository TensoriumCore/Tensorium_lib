55 static tensorium::Tensor<T, 2>
56 compute_laplacian_term(const tensorium::Vector<T> &X, T dx, T dy, T dz, const Metric<T> &metric,
57 const tensorium::Tensor<T, 2> &gamma_tilde_inv) {
58 using namespace tensorium;
59
60 Tensor<T, 2> R1({3, 3});
61
62 const int off = (X.size() == 4) ? 1 : 0;
63
64 for (int i = 0; i < 3; ++i) {
65 for (int j = 0; j < 3; ++j) {
66
67 auto gij = [&](const Vector<T> &Xs) -> T {
68 T alpha;
69 Vector<T> beta(3);
70 Tensor<T, 2> g_phys({3, 3});
71 metric.BSSN(Xs, alpha, beta, g_phys);
72
73 T chi = compute_conformal_factor(metric, g_phys);
74 Tensor<T, 2> gtilde = compute_conformal_metric(metric, g_phys, chi);
75 return gtilde(i, j);
76 };
77
78 Tensor<T, 2> hess({3, 3});
79
80 if (off == 0) {
81 compute_second_derivatives_scalar(X, dx, dy, dz, gij, hess);
82 } else {
83 Vector<T> X3(3);
84 X3(0) = X(1);
85 X3(1) = X(2);
86 X3(2) = X(3);
87
88 auto gij_from_X3 = [&](const Vector<T> &X3s) -> T {
89 Vector<T> X4 = X;
90 X4(1) = X3s(0);
91 X4(2) = X3s(1);
92 X4(3) = X3s(2);
93 return gij(X4);
94 };
95
96 compute_second_derivatives_scalar(X3, dx, dy, dz, gij_from_X3, hess);
97 }
98
99 T lap = T(0);
100 for (int k = 0; k < 3; ++k)
101 for (int l = 0; l < 3; ++l)
102 lap += gamma_tilde_inv(k, l) * hess(k, l);
103
104 R1(i, j) = T(-0.5) * lap;
105 }
106 }
107
108 for (int i = 0; i < 3; ++i)
109 for (int j = i + 1; j < 3; ++j) {
110 T avg = T(0.5) * (R1(i, j) + R1(j, i));
111 R1(i, j) = R1(j, i) = avg;
112 }
113
114 return R1;
115 }

127 static tensorium::Tensor<T, 2>
128 compute_dGamma_term(const tensorium::Vector<T> &X, T dx, T dy, T dz,
129 const tensorium::Vector<T> &tilde_Gamma,
130 const tensorium::Tensor<T, 2> &tilde_gamma_contract) {
131 using namespace tensorium;
132
133 Tensor<T, 2> dGamma({3, 3});
134 for (int k = 0; k < 3; ++k) {
135 auto scalar_func = [&](const Vector<T> &Xs) -> T { return tilde_Gamma(k); };
136 Vector<T> grad_k = partial_scalar(X, dx, dy, dz, scalar_func);
137 for (int j = 0; j < 3; ++j) {
138 dGamma(j, k) = grad_k(j);
139 }
140 }
141
142 Tensor<T, 2> R2({3, 3});
143 for (int i = 0; i < 3; ++i) {
144 for (int j = 0; j < 3; ++j) {
145 T sum = 0;
146 for (int k = 0; k < 3; ++k) {
147 sum += tilde_gamma_contract(k, i) * dGamma(j, k) +
148 tilde_gamma_contract(k, j) * dGamma(i, k);
149 }
150 R2(i, j) = static_cast<T>(0.5) * sum;
151 }
152 }
153
154 return R2;
155 }

169 static tensorium::Tensor<T, 2>
170 compute_GammaGamma_term(const tensorium::Vector<T> &tilde_Gamma,
171 const tensorium::Tensor<T, 3> &christoffel_tilde,
172 const tensorium::Tensor<T, 2> &gamma_tilde) {
173 using namespace tensorium;
174 Tensor<T, 2> R3({3, 3});
175
176 for (size_t i = 0; i < 3; ++i) {
177 for (size_t j = 0; j < 3; ++j) {
178 T sum_ij = 0;
179 for (size_t k = 0; k < 3; ++k) {
180 T G_ijk = 0, G_jik = 0;
181 for (size_t m = 0; m < 3; ++m) {
182 G_ijk += gamma_tilde(i, m) * christoffel_tilde(m, j, k);
183 G_jik += gamma_tilde(j, m) * christoffel_tilde(m, i, k);
184 }
185 sum_ij += tilde_Gamma(k) * (G_ijk + G_jik);
186 }
187 R3(i, j) = T(0.5) * sum_ij;
188 }
189 }
190
191 return R3;
192 }

205 static tensorium::Tensor<T, 2>
206 compute_GammaProduct_term(const tensorium::Tensor<T, 2> &gamma_tilde_inv,
207 const tensorium::Tensor<T, 3> &christoffel_tilde,
208 const tensorium::Tensor<T, 2> &gamma_tilde) {
209 using namespace tensorium;
210 Tensor<T, 2> R4({3, 3});
211
212 for (size_t i = 0; i < 3; ++i) {
213 for (size_t j = 0; j < 3; ++j) {
214 T sum = 0;
215 for (size_t ell = 0; ell < 3; ++ell) {
216 for (size_t m = 0; m < 3; ++m) {
217 T g_inv = gamma_tilde_inv(ell, m);
218 T term_k_sum = 0;
219
220 for (size_t k = 0; k < 3; ++k) {
221 T Gamma_jkm = 0;
222 T Gamma_ikm = 0;
223 for (size_t p = 0; p < 3; ++p) {
224 Gamma_jkm += gamma_tilde(j, p) * christoffel_tilde(p, k, m);
225 Gamma_ikm += gamma_tilde(i, p) * christoffel_tilde(p, k, m);
226 }
227
228 T Gamma_klj = 0;
229 for (size_t p = 0; p < 3; ++p) {
230 Gamma_klj += gamma_tilde(k, p) * christoffel_tilde(p, ell, j);
231 }
232
233 T t1 = christoffel_tilde(k, ell, i) * Gamma_jkm;
234 T t2 = christoffel_tilde(k, ell, j) * Gamma_ikm;
235
236 T t3 = christoffel_tilde(k, i, m) * Gamma_klj;
237
238 term_k_sum += (t1 + t2 + t3);
239 }
240 sum += g_inv * term_k_sum;
241 }
242 }
243 R4(i, j) = sum;
244 }
245 }
246 return R4;
247 }

260 static tensorium::Tensor<T, 2> compute_Ricci_Tilde_tensor(
261 const ChiContext<T> &chi_context, const tensorium::Tensor<T, 2> &gamma_tilde_inv,
262 const tensorium::Vector<T> &tilde_Gamma, const tensorium::Tensor<T, 3> &christoffel_tilde,
263 const tensorium::Tensor<T, 2> &gamma_tilde) {
264 using namespace tensorium;
265 const auto &X = chi_context.X;
266 const T dx = chi_context.dx;
267 const T dy = chi_context.dy;
268 const T dz = chi_context.dz;
269 const auto &metric = chi_context.metric;
270 auto R1 = compute_laplacian_term(X, dx, dy, dz, metric, gamma_tilde_inv);
271 auto R2 = compute_dGamma_term(X, dx, dy, dz, tilde_Gamma, gamma_tilde);
272 auto R3 = compute_GammaGamma_term(tilde_Gamma, christoffel_tilde, gamma_tilde);
273 auto R4 = compute_GammaProduct_term(gamma_tilde_inv, christoffel_tilde, gamma_tilde);
274 tensorium::Tensor<T, 2> Ricci({3, 3});
275 for (size_t i = 0; i < 3; ++i) {
276 for (size_t j = 0; j < 3; ++j) {
277 Ricci(i, j) = R1(i, j) + R2(i, j) + R3(i, j) + R4(i, j);
278 }
279 }
280
281 return Ricci;
282 }
