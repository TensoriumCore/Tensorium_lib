29inline double det3(double gxx, double gxy, double gxz, double gyy, double gyz, double gzz) {
30 return gxx * (gyy * gzz - gyz * gyz) - gxy * (gxy * gzz - gxz * gyz) +

39template <typename T>
40inline ConstraintMonitorStats compute_constraint_monitor(const BSSNGridSoA<T> &G,
41 const Field3D<T> &H, size_t padding = 4) {
42 ConstraintMonitorStats stats;
43
44 size_t I0, I1, J0, J1, K0, K1;
45 G.domain_bounds(I0, I1, J0, J1, K0, K1);
46
47 const size_t guard = std::max<size_t>(padding, size_t(2));
48 const size_t i0 = std::min(I0 + guard, I1);
49 const size_t j0 = std::min(J0 + guard, J1);
50 const size_t k0 = std::min(K0 + guard, K1);
51 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
52 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
53 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
54
55 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
56 return stats;
57
58 double accum_H = 0.0;
59
60 for (size_t i = i0; i < i1; ++i)
61 for (size_t j = j0; j < j1; ++j)
62 for (size_t k = k0; k < k1; ++k) {
63 const size_t id = H.idx(i, j, k);
64 const double h = static_cast<double>(H.ptr()[id]);
65 stats.max_H = std::max(stats.max_H, std::abs(h));
66 accum_H += h * h;
67 stats.samples += 1;
68
69 const double gxx = tensorium_RG::sym6_get(G.gamma_tilde, id, 0, 0);
70 const double gxy = tensorium_RG::sym6_get(G.gamma_tilde, id, 0, 1);
71 const double gxz = tensorium_RG::sym6_get(G.gamma_tilde, id, 0, 2);
72 const double gyy = tensorium_RG::sym6_get(G.gamma_tilde, id, 1, 1);
73 const double gyz = tensorium_RG::sym6_get(G.gamma_tilde, id, 1, 2);
74 const double gzz = tensorium_RG::sym6_get(G.gamma_tilde, id, 2, 2);
75 const double det_val = constraint_monitor_detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
76 stats.max_det_drift = std::max(stats.max_det_drift, std::abs(det_val - 1.0));
77
78 const double Axx = tensorium_RG::sym6_get(G.A_tilde, id, 0, 0);
79 const double Axy = tensorium_RG::sym6_get(G.A_tilde, id, 0, 1);
80 const double Axz = tensorium_RG::sym6_get(G.A_tilde, id, 0, 2);
81 const double Ayy = tensorium_RG::sym6_get(G.A_tilde, id, 1, 1);
82 const double Ayz = tensorium_RG::sym6_get(G.A_tilde, id, 1, 2);
83 const double Azz = tensorium_RG::sym6_get(G.A_tilde, id, 2, 2);
84
85 const double gixx = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 0, 0);
86 const double gixy = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 0, 1);
87 const double gixz = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 0, 2);
88 const double giyy = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 1, 1);
89 const double giyz = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 1, 2);
90 const double gizz = tensorium_RG::sym6_get(G.gamma_tilde_inv, id, 2, 2);
91 const double trace = gixx * Axx + giyy * Ayy + gizz * Azz +
92 2.0 * (gixy * Axy + gixz * Axz + giyz * Ayz);
93 stats.max_trace_A = std::max(stats.max_trace_A, std::abs(trace));
94 }
95
96 if (stats.samples > 0)
97 stats.l2_H = std::sqrt(accum_H / static_cast<double>(stats.samples));
98 else
99 stats.l2_H = 0.0;
100
101 return stats;
102}

104template <typename T>
105inline void populate_constraint_norms(const BSSNGridSoA<T> &G, const Field3D<T> M[3],
106 ConstraintMonitorStats &stats, size_t padding = 4) {
107 size_t I0, I1, J0, J1, K0, K1;
108 G.domain_bounds(I0, I1, J0, J1, K0, K1);
109
110 const size_t guard = std::max<size_t>(padding, size_t(2));
111 const size_t i0 = std::min(I0 + guard, I1);
112 const size_t j0 = std::min(J0 + guard, J1);
113 const size_t k0 = std::min(K0 + guard, K1);
114 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
115 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
116 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
117
118 if (i0 >= i1 || j0 >= j1 || k0 >= k1) {
119 stats.l2_theta = 0.0;
120 stats.l2_Z = 0.0;
121 stats.l2_M = 0.0;
122 return;
123 }
124
125 double accum_theta = 0.0;
126 double accum_Z = 0.0;
127 double accum_M = 0.0;
128 size_t count = 0;
129
130 for (size_t i = i0; i < i1; ++i)
131 for (size_t j = j0; j < j1; ++j)
132 for (size_t k = k0; k < k1; ++k) {
133 const size_t idx = G.alpha.idx(i, j, k);
134 const double theta = static_cast<double>(G.Theta.ptr()[idx]);
135 accum_theta += theta * theta;
136
137 const double zx = static_cast<double>(G.Z[0].ptr()[idx]);
138 const double zy = static_cast<double>(G.Z[1].ptr()[idx]);
139 const double zz = static_cast<double>(G.Z[2].ptr()[idx]);
140 accum_Z += zx * zx + zy * zy + zz * zz;
141
142 const size_t midx = M[0].idx(i, j, k);
143 const double mx = static_cast<double>(M[0].ptr()[midx]);
144 const double my = static_cast<double>(M[1].ptr()[midx]);
145 const double mz = static_cast<double>(M[2].ptr()[midx]);
146 accum_M += mx * mx + my * my + mz * mz;
147 ++count;
148 }
149
150 if (count > 0) {
151 const double inv = 1.0 / static_cast<double>(count);
152 stats.l2_theta = std::sqrt(accum_theta * inv);
153 stats.l2_Z = std::sqrt(accum_Z * inv);
154 stats.l2_M = std::sqrt(accum_M * inv);
155 } else {
156 stats.l2_theta = stats.l2_Z = stats.l2_M = 0.0;
157 }
158}
