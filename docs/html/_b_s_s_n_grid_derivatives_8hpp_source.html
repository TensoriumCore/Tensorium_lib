45template <typename T> inline double Dx_ptr(const T *p, ptrdiff_t sx, double inv_60dx) {
46 return (p[3 * sx] - 9.0 * p[2 * sx] + 45.0 * p[sx] - 45.0 * p[-sx] + 9.0 * p[-2 * sx] -
47 p[-3 * sx]) *
48 inv_60dx;
49}

51template <typename T> inline double Dy_ptr(const T *p, ptrdiff_t sy, double inv_60dy) {
52 return (p[3 * sy] - 9.0 * p[2 * sy] + 45.0 * p[sy] - 45.0 * p[-sy] + 9.0 * p[-2 * sy] -
53 p[-3 * sy]) *
54 inv_60dy;
55}
56
57template <typename T> inline double Dz_ptr(const T *p, double inv_60dz) {
58 // Stride Z is always 1 in this layout
59 return (p[3] - 9.0 * p[2] + 45.0 * p[1] - 45.0 * p[-1] + 9.0 * p[-2] - p[-3]) * inv_60dz;
60}

95// Mixed Derivatives (Order 4)
96template <typename T>
97inline double Dxy4_ptr(const T *p, ptrdiff_t sx, ptrdiff_t sy, double inv_144dxdy) {
98 double sum = 0.0;
99 const ptrdiff_t idx_x[4] = {-2 * sx, -sx, sx, 2 * sx};
100 const ptrdiff_t idx_y[4] = {-2 * sy, -sy, sy, 2 * sy};
101 const double w[4] = {-1.0, 8.0, -8.0, 1.0};
102
103 for (int a = 0; a < 4; ++a) {
104 for (int b = 0; b < 4; ++b) {
105 sum += w[a] * w[b] * p[idx_x[a] + idx_y[b]];
106 }
107 }
108 return sum * inv_144dxdy;
109}
110
111template <typename T> inline double Dxz4_ptr(const T *p, ptrdiff_t sx, double inv_144dxdz) {
112 double sum = 0.0;
113 const ptrdiff_t idx_x[4] = {-2 * sx, -sx, sx, 2 * sx};
114 const ptrdiff_t idx_z[4] = {-2, -1, 1, 2};
115 const double w[4] = {-1.0, 8.0, -8.0, 1.0};
116
117 for (int a = 0; a < 4; ++a) {
118 for (int b = 0; b < 4; ++b) {
119 sum += w[a] * w[b] * p[idx_x[a] + idx_z[b]];
120 }
121 }
122 return sum * inv_144dxdz;
123}
124
125template <typename T> inline double Dyz4_ptr(const T *p, ptrdiff_t sy, double inv_144dydz) {
126 double sum = 0.0;
127 const ptrdiff_t idx_y[4] = {-2 * sy, -sy, sy, 2 * sy};
128 const ptrdiff_t idx_z[4] = {-2, -1, 1, 2};
129 const double w[4] = {-1.0, 8.0, -8.0, 1.0};
130
131 for (int a = 0; a < 4; ++a) {
132 for (int b = 0; b < 4; ++b) {
133 sum += w[a] * w[b] * p[idx_y[a] + idx_z[b]];
134 }
135 }
136 return sum * inv_144dydz;
137}

305template <typename Grid, typename Field3>
306inline void covariant_D_vec(const Grid &G, const Field3 Vfield[3], size_t i, size_t j, size_t k,
307 double DV[3][3], Variance in_var, Variance out_var) {
308 const size_t id = G.alpha.idx(i, j, k);
309
310 // Optimization: Pre-calculate inverses
311 const double inv_12dx = 1.0 / (60.0 * G.dx);
312 const double inv_12dy = 1.0 / (60.0 * G.dy);
313 const double inv_12dz = 1.0 / (60.0 * G.dz);
314 const ptrdiff_t sx = G.alpha.st.sx;
315 const ptrdiff_t sy = G.alpha.st.sy;
316
317 const double chi = G.chi.ptr()[id];
318 const double inv_chi = 1.0 / chi;
319
320 double tgamma[3][3], tginv[3][3];
321 for (int a = 0; a < 3; ++a)
322 for (int b = 0; b < 3; ++b) {
323 tgamma[a][b] = G.gamma_tilde[sym6(a, b)].ptr()[id];
324 tginv[a][b] = G.gamma_tilde_inv[sym6(a, b)].ptr()[id];
325 }
326
327 const double *p_chi = G.chi.ptr() + id;
328 const double dchi[3] = {Dx_ptr(p_chi, sx, inv_12dx), Dy_ptr(p_chi, sy, inv_12dy),
329 Dz_ptr(p_chi, inv_12dz)};
330
331 double V_up[3], V_dn[3];
332 if (in_var == Variance::Up) {
333 V_up[0] = Vfield[0].ptr()[id];
334 V_up[1] = Vfield[1].ptr()[id];
335 V_up[2] = Vfield[2].ptr()[id];
336 for (int a = 0; a < 3; ++a)
337 V_dn[a] = inv_chi *
338 (tgamma[a][0] * V_up[0] + tgamma[a][1] * V_up[1] + tgamma[a][2] * V_up[2]);
339 } else {
340 V_dn[0] = Vfield[0].ptr()[id];
341 V_dn[1] = Vfield[1].ptr()[id];
342 V_dn[2] = Vfield[2].ptr()[id];
343 for (int a = 0; a < 3; ++a)
344 V_up[a] = chi * (tginv[a][0] * V_dn[0] + tginv[a][1] * V_dn[1] + tginv[a][2] * V_dn[2]);
345 }
346
347 double dV_in[3][3];
348 for (int m = 0; m < 3; ++m)
349 for (int a = 0; a < 3; ++a) {
350 const auto *p_v = Vfield[a].ptr() + id;
351 const double dv = (m == 0) ? Dx_ptr(p_v, sx, inv_12dx)
352 : (m == 1) ? Dy_ptr(p_v, sy, inv_12dy)
353 : Dz_ptr(p_v, inv_12dz);
354 dV_in[m][a] = dv;
355 }
356
357 double Gamma_tilde[3][3][3];
358 tensorium_RG::bssn::compute_tildeGamma_symbols(G, i, j, k, Gamma_tilde);
359
360 double Gamma_phys[3][3][3];
361 for (int up = 0; up < 3; ++up)
362 for (int lo1 = 0; lo1 < 3; ++lo1)
363 for (int lo2 = 0; lo2 < 3; ++lo2) {
364
365 const double Gt = Gamma_tilde[up][lo1][lo2];
366 const double term1 = (up == lo1) ? dchi[lo2] : 0.0;
367 const double term2 = (up == lo2) ? dchi[lo1] : 0.0;
368
369 double contracted = 0.0;
370 for (int ell = 0; ell < 3; ++ell)
371 contracted += tginv[up][ell] * dchi[ell];
372
373 const double term3 = tgamma[lo1][lo2] * contracted;
374 const double C = -(0.5 / chi) * (term1 + term2 - term3);
375
376 Gamma_phys[up][lo1][lo2] = Gt + C;
377 }
378
379 for (int m = 0; m < 3; ++m)
380 for (int a = 0; a < 3; ++a)
381 DV[m][a] = 0.0;
382
383 if (in_var == Variance::Up && out_var == Variance::Up) {
384 for (int m = 0; m < 3; ++m)
385 for (int jidx = 0; jidx < 3; ++jidx) {
386 double val = dV_in[m][jidx];
387 for (int kidx = 0; kidx < 3; ++kidx)
388 val += Gamma_phys[jidx][m][kidx] * V_up[kidx];
389 DV[m][jidx] = val;
390 }
391 return;
392 }
393
394 if (in_var == Variance::Down && out_var == Variance::Down) {
395 for (int m = 0; m < 3; ++m)
396 for (int jidx = 0; jidx < 3; ++jidx) {
397 double val = dV_in[m][jidx];
398 for (int kidx = 0; kidx < 3; ++kidx)
399 val -= Gamma_phys[kidx][m][jidx] * V_dn[kidx];
400 DV[m][jidx] = val;
401 }
402 return;
403 }
404
405 if (in_var == Variance::Up && out_var == Variance::Down) {
406 const double inv_chi2 = inv_chi * inv_chi;
407
408 for (int m = 0; m < 3; ++m)
409 for (int jidx = 0; jidx < 3; ++jidx) {
410 double partial = 0.0;
411 for (int kidx = 0; kidx < 3; ++kidx) {
412 // Inline dgamma_phys logic using pointers
413 const auto *p_gam = G.gamma_tilde[sym6(jidx, kidx)].ptr() + id;
414 const double gt = *p_gam;
415 const double dgt = (m == 0) ? Dx_ptr(p_gam, sx, inv_12dx)
416 : (m == 1) ? Dy_ptr(p_gam, sy, inv_12dy)
417 : Dz_ptr(p_gam, inv_12dz);
418
419 const double dgamma_val = inv_chi * dgt - inv_chi2 * dchi[m] * gt;
420
421 partial += dgamma_val * V_up[kidx];
422 partial += (inv_chi * tgamma[jidx][kidx]) * dV_in[m][kidx];
423 }
424
425 double corr = 0.0;
426 for (int kidx = 0; kidx < 3; ++kidx)
427 corr += Gamma_phys[kidx][m][jidx] * V_dn[kidx];
428
429 DV[m][jidx] = partial - corr;
430 }
431 return;
432 }
433
434 if (in_var == Variance::Down && out_var == Variance::Up) {
435 double D_dn[3];
436 for (int m = 0; m < 3; ++m) {
437 for (int jidx = 0; jidx < 3; ++jidx) {
438 double val = dV_in[m][jidx];
439 for (int kidx = 0; kidx < 3; ++kidx)
440 val -= Gamma_phys[kidx][m][jidx] * V_dn[kidx];
441 D_dn[jidx] = val;
442 }
443 for (int jidx = 0; jidx < 3; ++jidx) {
444 DV[m][jidx] = chi * (tginv[jidx][0] * D_dn[0] + tginv[jidx][1] * D_dn[1] +
445 tginv[jidx][2] * D_dn[2]);
446 }
447 }
448 return;
449 }
