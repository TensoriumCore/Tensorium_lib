55template <typename T> inline void project_bssn_state(BSSNGridSoA<T> &G, const ProjectionConfig &cfg = {}) {
56 size_t I0, I1, J0, J1, K0, K1;
57 G.domain_bounds(I0, I1, J0, J1, K0, K1);
58
59 const size_t guard = std::max<size_t>(cfg.padding, size_t(2));
60 const size_t i0 = std::min(I0 + guard, I1);
61 const size_t j0 = std::min(J0 + guard, J1);
62 const size_t k0 = std::min(K0 + guard, K1);
63 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
64 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
65 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
66
67 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
68 return;
69
70#pragma omp parallel for collapse(2)
71 for (size_t i = i0; i < i1; ++i) {
72 for (size_t j = j0; j < j1; ++j) {
73 for (size_t k = k0; k < k1; ++k) {
74 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
75
76 double gxx, gxy, gxz, gyy, gyz, gzz;
77 tensorium_RG::load_sym6(G.gamma_tilde, id, gxx, gxy, gxz, gyy, gyz, gzz);
78
79 if (cfg.renormalize_metric) {
80 const double det = detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
81 if (det > 0.0) {
82 const double scale = std::pow(det, -1.0 / 3.0);
83 gxx *= scale;
84 gxy *= scale;
85 gxz *= scale;
86 gyy *= scale;
87 gyz *= scale;
88 gzz *= scale;
89 tensorium_RG::store_sym6(G.gamma_tilde, id, (T)gxx, (T)gxy, (T)gxz, (T)gyy,
90 (T)gyz, (T)gzz);
91 }
92 }
93
94 double gixx, gixy, gixz, giyy, giyz, gizz;
95 if (cfg.recompute_inverse || cfg.project_A_tilde) {
96 const double det_new = detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
97 const double inv_det = (det_new != 0.0) ? (1.0 / det_new) : 0.0;
98 gixx = (gyy * gzz - gyz * gyz) * inv_det;
99 gixy = (gxz * gyz - gxy * gzz) * inv_det;
100 gixz = (gxy * gyz - gxz * gyy) * inv_det;
101 giyy = (gxx * gzz - gxz * gxz) * inv_det;
102 giyz = (gxy * gxz - gxx * gyz) * inv_det;
103 gizz = (gxx * gyy - gxy * gxy) * inv_det;
104
105 if (cfg.recompute_inverse) {
106 tensorium_RG::store_sym6(G.gamma_tilde_inv, id, (T)gixx, (T)gixy, (T)gixz,
107 (T)giyy, (T)giyz, (T)gizz);
108 }
109 }
110
111 if (cfg.project_A_tilde) {
112 double Axx, Axy, Axz, Ayy, Ayz, Azz;
113 tensorium_RG::load_sym6(G.A_tilde, id, Axx, Axy, Axz, Ayy, Ayz, Azz);
114 const double trace = gixx * Axx + giyy * Ayy + gizz * Azz +
115 2.0 * (gixy * Axy + gixz * Axz + giyz * Ayz);
116 const double one_third_trace = (1.0 / 3.0) * trace;
117 Axx -= gxx * one_third_trace;
118 Axy -= gxy * one_third_trace;
119 Axz -= gxz * one_third_trace;
120 Ayy -= gyy * one_third_trace;
121 Ayz -= gyz * one_third_trace;
122 Azz -= gzz * one_third_trace;
123 tensorium_RG::store_sym6(G.A_tilde, id, (T)Axx, (T)Axy, (T)Axz, (T)Ayy,
124 (T)Ayz, (T)Azz);
125 }
126 }
127 }
128 }
129
130 if (cfg.resync_contracted_gamma) {
131#pragma omp parallel for collapse(2)
132 for (size_t i = i0; i < i1; ++i) {
133 for (size_t j = j0; j < j1; ++j) {
134 for (size_t k = k0; k < k1; ++k) {
135 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
136 T div[3];
137 metric_inverse_divergence(G, i, j, k, div);
138 G.tildeGamma[0].ptr()[id] = -div[0];
139 G.tildeGamma[1].ptr()[id] = -div[1];
140 G.tildeGamma[2].ptr()[id] = -div[2];
141 }
142 }
143 }
144 }
145}
