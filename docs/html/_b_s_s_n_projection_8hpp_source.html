35inline double det3(double gxx, double gxy, double gxz, double gyy, double gyz, double gzz) {
36 return gxx * (gyy * gzz - gyz * gyz) - gxy * (gxy * gzz - gxz * gyz) +

42template <typename T>
43inline void project_conformal_metric_and_A(BSSNGridSoA<T> &G, size_t padding = 4,
44 bool renormalize_metric = true,
45 bool project_A_tilde = true) {
46 if (!renormalize_metric && !project_A_tilde)
47 return;
48
49 size_t I0, I1, J0, J1, K0, K1;
50 G.domain_bounds(I0, I1, J0, J1, K0, K1);
51 const size_t guard = std::max<size_t>(padding, size_t(2));
52 const size_t i0 = std::min(I0 + guard, I1);
53 const size_t j0 = std::min(J0 + guard, J1);
54 const size_t k0 = std::min(K0 + guard, K1);
55 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
56 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
57 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
58
59 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
60 return;
61
62#pragma omp parallel for collapse(2)
63 for (size_t i = i0; i < i1; ++i) {
64 for (size_t j = j0; j < j1; ++j) {
65 for (size_t k = k0; k < k1; ++k) {
66 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
67
68 double gxx, gxy, gxz, gyy, gyz, gzz;
69 tensorium_RG::load_sym6(G.gamma_tilde, id, gxx, gxy, gxz, gyy, gyz, gzz);
70
71 const double det = detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
72 double scale = 1.0;
73 if (renormalize_metric && det > 0.0)
74 scale = 1.0 / std::cbrt(det);
75
76 if (renormalize_metric && std::isfinite(scale)) {
77 gxx *= scale;
78 gxy *= scale;
79 gxz *= scale;
80 gyy *= scale;
81 gyz *= scale;
82 gzz *= scale;
83 tensorium_RG::store_sym6(G.gamma_tilde, id, (T)gxx, (T)gxy, (T)gxz, (T)gyy,
84 (T)gyz, (T)gzz);
85 } else {
86 scale = 1.0;
87 }
88
89 tensorium_RG::load_sym6(G.gamma_tilde, id, gxx, gxy, gxz, gyy, gyz, gzz);
90
91 double inv_det = detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
92 inv_det = (inv_det != 0.0) ? (1.0 / inv_det) : 0.0;
93 const double gixx = (gyy * gzz - gyz * gyz) * inv_det;
94 const double gixy = (gxz * gyz - gxy * gzz) * inv_det;
95 const double gixz = (gxy * gyz - gxz * gyy) * inv_det;
96 const double giyy = (gxx * gzz - gxz * gxz) * inv_det;
97 const double giyz = (gxy * gxz - gxx * gyz) * inv_det;
98 const double gizz = (gxx * gyy - gxy * gxy) * inv_det;
99
100 double Axx, Axy, Axz, Ayy, Ayz, Azz;
101 tensorium_RG::load_sym6(G.A_tilde, id, Axx, Axy, Axz, Ayy, Ayz, Azz);
102
103 if (renormalize_metric && std::isfinite(scale)) {
104 Axx *= scale;
105 Axy *= scale;
106 Axz *= scale;
107 Ayy *= scale;
108 Ayz *= scale;
109 Azz *= scale;
110 }
111
112 if (!project_A_tilde) {
113 tensorium_RG::store_sym6(G.A_tilde, id, (T)Axx, (T)Axy, (T)Axz, (T)Ayy,
114 (T)Ayz, (T)Azz);
115 continue;
116 }
117
118 const double trace = gixx * Axx + giyy * Ayy + gizz * Azz +
119 2.0 * (gixy * Axy + gixz * Axz + giyz * Ayz);
120 const double one_third_trace = trace / 3.0;
121 Axx -= gxx * one_third_trace;
122 Axy -= gxy * one_third_trace;
123 Axz -= gxz * one_third_trace;
124 Ayy -= gyy * one_third_trace;
125 Ayz -= gyz * one_third_trace;
126 Azz -= gzz * one_third_trace;
127
128 tensorium_RG::store_sym6(G.A_tilde, id, (T)Axx, (T)Axy, (T)Axz, (T)Ayy, (T)Ayz,
129 (T)Azz);
130 }
131 }
132 }

150template <typename T>
151inline void project_bssn_state(BSSNGridSoA<T> &G, const ProjectionConfig &cfg = {}) {
152 size_t I0, I1, J0, J1, K0, K1;
153 G.domain_bounds(I0, I1, J0, J1, K0, K1);
154
155 const size_t guard = std::max<size_t>(cfg.padding, size_t(2));
156 const size_t i0 = std::min(I0 + guard, I1);
157 const size_t j0 = std::min(J0 + guard, J1);
158 const size_t k0 = std::min(K0 + guard, K1);
159 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
160 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
161 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
162
163 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
164 return;
165
166 if (cfg.renormalize_metric || cfg.project_A_tilde)
167 project_conformal_metric_and_A(G, cfg.padding, cfg.renormalize_metric, cfg.project_A_tilde);
168
169 if (cfg.recompute_inverse) {
170#pragma omp parallel for collapse(2)
171 for (size_t i = i0; i < i1; ++i) {
172 for (size_t j = j0; j < j1; ++j) {
173 for (size_t k = k0; k < k1; ++k) {
174 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
175 double gxx, gxy, gxz, gyy, gyz, gzz;
176 tensorium_RG::load_sym6(G.gamma_tilde, id, gxx, gxy, gxz, gyy, gyz, gzz);
177 const double det = detail::det3(gxx, gxy, gxz, gyy, gyz, gzz);
178 const double inv_det = (det != 0.0) ? (1.0 / det) : 0.0;
179 const double gixx = (gyy * gzz - gyz * gyz) * inv_det;
180 const double gixy = (gxz * gyz - gxy * gzz) * inv_det;
181 const double gixz = (gxy * gyz - gxz * gyy) * inv_det;
182 const double giyy = (gxx * gzz - gxz * gxz) * inv_det;
183 const double giyz = (gxy * gxz - gxx * gyz) * inv_det;
184 const double gizz = (gxx * gyy - gxy * gxy) * inv_det;
185 tensorium_RG::store_sym6(G.gamma_tilde_inv, id, (T)gixx, (T)gixy, (T)gixz,
186 (T)giyy, (T)giyz, (T)gizz);
187 }
188 }
189 }
190 }
191
192 if (cfg.resync_contracted_gamma) {
193#pragma omp parallel for collapse(2)
194 for (size_t i = i0; i < i1; ++i) {
195 for (size_t j = j0; j < j1; ++j) {
196 for (size_t k = k0; k < k1; ++k) {
197 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
198 T div[3];
199 metric_inverse_divergence(G, i, j, k, div);
200 G.tildeGamma[0].ptr()[id] = -div[0];
201 G.tildeGamma[1].ptr()[id] = -div[1];
202 G.tildeGamma[2].ptr()[id] = -div[2];
203 }
204 }
205 }
206 }
207}
