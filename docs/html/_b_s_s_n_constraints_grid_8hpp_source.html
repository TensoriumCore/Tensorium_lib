37inline void compute_momentum_constraint_Mi(const BSSNGridSoA<T> &G, Field3D<T> Mi[3],
38 size_t padding = 4) {
39 using namespace tensorium_RG::fd;
40
41 const size_t total = G.chi.st.nx_tot * G.chi.st.ny_tot * G.chi.st.nz_tot;
42 for (int comp = 0; comp < 3; ++comp)
43 std::fill_n(Mi[comp].ptr(), total, T(0));
44
45 size_t I0, I1, J0, J1, K0, K1;
46 G.domain_bounds(I0, I1, J0, J1, K0, K1);
47
48 const size_t guard = std::max<size_t>(padding, size_t(2));
49 const size_t i0 = std::min(I0 + guard, I1);
50 const size_t j0 = std::min(J0 + guard, J1);
51 const size_t k0 = std::min(K0 + guard, K1);
52 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
53 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
54 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
55
56 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
57 return;
58
59 auto deriv = [&](const Field3D<T> &F, size_t ii, size_t jj, size_t kk, int dir) -> double {
60 return (dir == 0) ? Dx(F, ii, jj, kk, G.dx)
61 : (dir == 1) ? Dy(F, ii, jj, kk, G.dy)
62 : Dz(F, ii, jj, kk, G.dz);
63 };
64
65#pragma omp parallel for collapse(2)
66 for (size_t i = i0; i < i1; ++i) {
67 for (size_t j = j0; j < j1; ++j) {
68 for (size_t k = k0; k < k1; ++k) {
69 const size_t id = G.chi.idx(i, j, k);
70
71 double g_inv[3][3];
72 for (int a = 0; a < 3; ++a)
73 for (int b = 0; b < 3; ++b)
74 g_inv[a][b] = (double)tensorium_RG::sym6_get(G.gamma_tilde_inv, id, a, b);
75
76 double A_lo[3][3];
77 for (int a = 0; a < 3; ++a)
78 for (int b = 0; b < 3; ++b)
79 A_lo[a][b] = (double)tensorium_RG::sym6_get(G.A_tilde, id, a, b);
80
81 double d_g_inv[3][3][3] = {};
82 double dA_lo[3][3][3] = {};
83 for (int dir = 0; dir < 3; ++dir) {
84 for (int s = 0; s < 6; ++s) {
85 const int row = detail::sym_row_from_index[s];
86 const int col = detail::sym_col_from_index[s];
87 const Field3D<T> &F_inv = G.gamma_tilde_inv[s];
88 const Field3D<T> &F_A = G.A_tilde[s];
89 const double dginv = deriv(F_inv, i, j, k, dir);
90 const double dA = deriv(F_A, i, j, k, dir);
91 d_g_inv[dir][row][col] = dginv;
92 d_g_inv[dir][col][row] = dginv;
93 dA_lo[dir][row][col] = dA;
94 dA_lo[dir][col][row] = dA;
95 }
96 }
97
98 const double chi = (double)G.chi.ptr()[id];
99 const double inv_chi = (chi != 0.0) ? (1.0 / chi) : 0.0;
100 const double d_chi[3] = {deriv(G.chi, i, j, k, 0), deriv(G.chi, i, j, k, 1),
101 deriv(G.chi, i, j, k, 2)};
102 const double dK[3] = {deriv(G.K, i, j, k, 0), deriv(G.K, i, j, k, 1),
103 deriv(G.K, i, j, k, 2)};
104
105 double A_mix[3][3];
106 for (int up = 0; up < 3; ++up)
107 for (int lo = 0; lo < 3; ++lo) {
108 double sum = 0.0;
109 for (int m = 0; m < 3; ++m)
110 sum += g_inv[up][m] * A_lo[m][lo];
111 A_mix[up][lo] = sum;
112 }
113
114 double dA_mix[3][3][3];
115 for (int dir = 0; dir < 3; ++dir)
116 for (int up = 0; up < 3; ++up)
117 for (int lo = 0; lo < 3; ++lo) {
118 double sum = 0.0;
119 for (int m = 0; m < 3; ++m)
120 sum += d_g_inv[dir][up][m] * A_lo[m][lo] +
121 g_inv[up][m] * dA_lo[dir][m][lo];
122 dA_mix[dir][up][lo] = sum;
123 }
124
125 T Gamma_vals_T[3][3][3];
126 tensorium_RG::bssn::compute_tildeGamma_symbols(G, i, j, k, Gamma_vals_T);
127 double Gamma_vals[3][3][3];
128 for (int up = 0; up < 3; ++up)
129 for (int lo1 = 0; lo1 < 3; ++lo1)
130 for (int lo2 = 0; lo2 < 3; ++lo2)
131 Gamma_vals[up][lo1][lo2] = (double)Gamma_vals_T[up][lo1][lo2];
132
133 double traceGamma[3] = {0.0, 0.0, 0.0};
134 for (int kidx = 0; kidx < 3; ++kidx)
135 for (int jidx = 0; jidx < 3; ++jidx)
136 traceGamma[kidx] += Gamma_vals[jidx][jidx][kidx];
137
138 const double grad_phi_factor[3] = {-(1.5) * d_chi[0] * inv_chi,
139 -(1.5) * d_chi[1] * inv_chi,
140 -(1.5) * d_chi[2] * inv_chi};
141
142 for (int lo = 0; lo < 3; ++lo) {
143 double divA = dA_mix[0][0][lo] + dA_mix[1][1][lo] + dA_mix[2][2][lo];
144 for (int kidx = 0; kidx < 3; ++kidx)
145 divA += traceGamma[kidx] * A_mix[kidx][lo];
146
147 double conn_term = 0.0;
148 for (int jidx = 0; jidx < 3; ++jidx)
149 for (int kidx = 0; kidx < 3; ++kidx)
150 conn_term += Gamma_vals[kidx][jidx][lo] * A_mix[jidx][kidx];
151 divA -= conn_term;
152
153 double phi_term = 0.0;
154 for (int jidx = 0; jidx < 3; ++jidx)
155 phi_term += A_mix[jidx][lo] * grad_phi_factor[jidx];
156
157 const double Mi_val = divA + phi_term - (2.0 / 3.0) * dK[lo];
158 Mi[lo].ptr()[id] = (T)Mi_val;
159 }
160 }
161 }
162 }

177template <typename T>
178static inline void compute_bssn_constraints(BSSNGridSoA<T> &G, const Field3D<T> *Ricci6,
179 Field3D<T> &H_out, Field3D<T> M_out[3],
180 Field3D<T> C_out[3], double r_min, double r_max,
181 double xc, double yc, double zc,
182 bool throw_on_violation = true) {
183 using namespace tensorium_RG::fd;
184
185 size_t I0, I1, J0, J1, K0, K1;
186 G.domain_bounds(I0, I1, J0, J1, K0, K1);
187
188 const size_t total = G.chi.st.nx_tot * G.chi.st.ny_tot * G.chi.st.nz_tot;
189 std::fill_n(H_out.ptr(), total, T(0));
190 for (int a = 0; a < 3; ++a)
191 std::fill_n(C_out[a].ptr(), total, T(0));
192
193 compute_momentum_constraint_Mi(G, M_out, 4);
194
195 const size_t i0 = I0 + 4, i1 = I1 - 4;
196 const size_t j0 = J0 + 4, j1 = J1 - 4;
197 const size_t k0 = K0 + 4, k1 = K1 - 4;
198
199#pragma omp parallel for collapse(2)
200 for (size_t i = i0; i < i1; ++i) {
201 for (size_t j = j0; j < j1; ++j) {
202 for (size_t k = k0; k < k1; ++k) {
203
204 double x, y, z;
205 G.coords(i, j, k, x, y, z);
206 x -= xc;
207 y -= yc;
208 z -= zc;
209 const size_t id = G.chi.idx(i, j, k);
210 const double r = std::sqrt(x * x + y * y + z * z);
211 if (r <= r_min || r >= r_max) {
212 for (int comp = 0; comp < 3; ++comp)
213 M_out[comp].ptr()[id] = T(0);
214 continue;
215 }
216
217 const double chi = (double)G.chi.ptr()[id];
218
219 const double Ktr = (double)G.K.ptr()[id];
220
221 double gtI[3][3];
222 for (int a = 0; a < 3; ++a)
223 for (int b = 0; b < 3; ++b)
224 gtI[a][b] = (double)tensorium_RG::sym6_get(G.gamma_tilde_inv, id, a, b);
225
226 double Atd[3][3];
227 for (int a = 0; a < 3; ++a)
228 for (int b = 0; b < 3; ++b)
229 Atd[a][b] = (double)tensorium_RG::sym6_get(G.A_tilde, id, a, b);
230
231 double Atu[3][3];
232 for (int a = 0; a < 3; ++a)
233 for (int b = 0; b < 3; ++b) {
234 double s = 0.0;
235 for (int k1 = 0; k1 < 3; ++k1)
236 for (int k2 = 0; k2 < 3; ++k2)
237 s += gtI[a][k1] * gtI[b][k2] * Atd[k1][k2];
238 Atu[a][b] = s;
239 }
240
241 double A2 = 0.0;
242 for (int a = 0; a < 3; ++a)
243 for (int b = 0; b < 3; ++b)
244 A2 += Atd[a][b] * Atu[a][b];
245
246 const double Rxx = (double)Ricci6[XX].ptr()[id];
247 const double Rxy = (double)Ricci6[XY].ptr()[id];
248 const double Rxz = (double)Ricci6[XZ].ptr()[id];
249 const double Ryy = (double)Ricci6[YY].ptr()[id];
250 const double Ryz = (double)Ricci6[YZ].ptr()[id];
251 const double Rzz = (double)Ricci6[ZZ].ptr()[id];
252
253 const double gI_xx = chi * gtI[0][0];
254 const double gI_xy = chi * gtI[0][1];
255 const double gI_xz = chi * gtI[0][2];
256 const double gI_yy = chi * gtI[1][1];
257 const double gI_yz = chi * gtI[1][2];
258 const double gI_zz = chi * gtI[2][2];
259
260 const double Rsc = gI_xx * Rxx + 2.0 * gI_xy * Rxy + 2.0 * gI_xz * Rxz +
261 gI_yy * Ryy + 2.0 * gI_yz * Ryz + gI_zz * Rzz;
262
263 const double H = Rsc + (2.0 / 3.0) * Ktr * Ktr - A2;
264 H_out.ptr()[id] = (T)H;
265
266 T div_gamma[3];
267 tensorium_RG::bssn::metric_inverse_divergence(G, i, j, k, div_gamma);
268 for (int iC = 0; iC < 3; ++iC) {
269 const double Ci = (double)G.tildeGamma[iC].ptr()[id] + (double)div_gamma[iC];
270 C_out[iC].ptr()[id] = (T)Ci;
271 }
272 }
273 }
274 }
275
276 tensorium_RG::bssn::assert_invariants(G, "constraints", 4,
277 tensorium_RG::bssn::compute_invariant_tolerances(G),
278 throw_on_violation);
279}

284static inline void print_constraint_norms(BSSNGridSoA<double> &G, const Field3D<double> &H,
285 const Field3D<double> M[3], const Field3D<double> C[3],
286 double r_min, double r_max, double xc, double yc,
287 double zc) {
288 size_t I0, I1, J0, J1, K0, K1;
289 G.domain_bounds(I0, I1, J0, J1, K0, K1);
290
291 const size_t i0 = I0 + 4, i1 = I1 - 4;
292 const size_t j0 = J0 + 4, j1 = J1 - 4;
293 const size_t k0 = K0 + 4, k1 = K1 - 4;
294
295 double maxH = 0, maxM = 0, maxC = 0;
296 double l2H = 0, l2M = 0, l2C = 0;
297 size_t n = 0;
298
299 for (size_t i = i0; i < i1; ++i)
300 for (size_t j = j0; j < j1; ++j)
301 for (size_t k = k0; k < k1; ++k) {
302 double x, y, z;
303 G.coords(i, j, k, x, y, z);
304 x -= xc;
305 y -= yc;
306 z -= zc;
307 const double r = std::sqrt(x * x + y * y + z * z);
308 if (r <= r_min || r >= r_max)
309 continue;
310
311 const size_t id = G.chi.idx(i, j, k);
312
313 const double h = std::abs(H.ptr()[id]);
314 const double m =
315 std::sqrt(M[0].ptr()[id] * M[0].ptr()[id] + M[1].ptr()[id] * M[1].ptr()[id] +
316 M[2].ptr()[id] * M[2].ptr()[id]);
317 const double c =
318 std::sqrt(C[0].ptr()[id] * C[0].ptr()[id] + C[1].ptr()[id] * C[1].ptr()[id] +
319 C[2].ptr()[id] * C[2].ptr()[id]);
320
321 maxH = std::max(maxH, h);
322 maxM = std::max(maxM, m);
323 maxC = std::max(maxC, c);
324
325 l2H += h * h;
326 l2M += m * m;
327 l2C += c * c;
328 n++;
329 }
330
331 l2H = std::sqrt(l2H / std::max<size_t>(1, n));
332 l2M = std::sqrt(l2M / std::max<size_t>(1, n));
333 l2C = std::sqrt(l2C / std::max<size_t>(1, n));
334
335 std::printf("[CONS] r in (%.3f, %.3f) samples=%zu\n", r_min, r_max, n);
336 std::printf("[CONS] H: Linf=%.3e L2=%.3e\n", maxH, l2H);
337 std::printf("[CONS] M: Linf=%.3e L2=%.3e\n", maxM, l2M);
338 std::printf("[CONS] C: Linf=%.3e L2=%.3e\n", maxC, l2C);
339}
