32template <typename T>
33inline void compute_rhs_Theta(const BSSNGridSoA<T> &G, Field3D<T> &rhs_theta,
34 const GaugeParameters<T> &params = {}, size_t padding = 4) {
35 BSSN_PROFILE_KERNEL(Theta);
36 using namespace tensorium_RG::fd;
37 const T ko_sigma = scaled_ko_sigma(params.ko_sigma); // Adaptive KO6 strength.
38
39 size_t I0, I1, J0, J1, K0, K1;
40 G.domain_bounds(I0, I1, J0, J1, K0, K1);
41
42 const size_t i0 = clamped_lower(I0, padding, I1);
43 const size_t j0 = clamped_lower(J0, padding, J1);
44 const size_t k0 = clamped_lower(K0, padding, K1);
45 const size_t i1 = clamped_upper(I1, padding, I0);
46 const size_t j1 = clamped_upper(J1, padding, J0);
47 const size_t k1 = clamped_upper(K1, padding, K0);
48
49 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
50 return;
51
52 const double inv_12dx = 1.0 / (60.0 * G.dx);
53 const double inv_12dy = 1.0 / (60.0 * G.dy);
54 const double inv_12dz = 1.0 / (60.0 * G.dz);
55 const double inv_2dx = 1.0 / (2.0 * G.dx);
56 const double inv_2dy = 1.0 / (2.0 * G.dy);
57 const double inv_2dz = 1.0 / (2.0 * G.dz);
58
59 const ptrdiff_t sx = G.Theta.st.sx;
60 const ptrdiff_t sy = G.Theta.st.sy;
61
62 const T two_plus_kappa2 = T(2) + params.kappa2;
63 const T two_thirds = T(2) / T(3);
64
65#pragma omp parallel for collapse(2)
66 for (size_t i = i0; i < i1; ++i) {
67 for (size_t j = j0; j < j1; ++j) {
68 size_t idx_start = G.Theta.idx(i, j, k0);
69
70 const T *p_theta = G.Theta.ptr() + idx_start;
71 const T *p_alpha = G.alpha.ptr() + idx_start;
72 const T *p_K = G.K.ptr() + idx_start;
73 const T *p_beta[3] = {G.beta[0].ptr() + idx_start, G.beta[1].ptr() + idx_start,
74 G.beta[2].ptr() + idx_start};
75 const T *p_Z[3] = {G.Z[0].ptr() + idx_start, G.Z[1].ptr() + idx_start,
76 G.Z[2].ptr() + idx_start};
77 const T *p_chi = G.chi.ptr() + idx_start;
78 const T *p_ginv[6];
79 const T *p_A[6];
80 const T *p_R[6];
81 for (int s = 0; s < 6; ++s) {
82 p_ginv[s] = G.gamma_tilde_inv[s].ptr() + idx_start;
83 p_A[s] = G.A_tilde[s].ptr() + idx_start;
84 p_R[s] = G.Ricci[s].ptr() + idx_start;
85 }
86
87 T *p_rhs = rhs_theta.ptr() + idx_start;
88
89 for (size_t k = k0; k < k1; ++k) {
90 const T theta = *p_theta;
91 const T alpha = *p_alpha;
92 const T K_val = *p_K;
93 const T khat = Khat(K_val, theta);
94 T RicciZ4[6];
95 compute_RicciZ4(G, i, j, k, RicciZ4);
96 const T bx = *p_beta[0];
97 const T by = *p_beta[1];
98 const T bz = *p_beta[2];
99 const T chi = *p_chi;
100 const T z_x = *p_Z[0];
101 const T z_y = *p_Z[1];
102 const T z_z = *p_Z[2];
103
104 const T d_alpha[3] = {Dx_ptr(p_alpha, sx, inv_12dx), Dy_ptr(p_alpha, sy, inv_12dy),
105 Dz_ptr(p_alpha, inv_12dz)};
106 const T d_chi[3] = {Dx_ptr(p_chi, sx, inv_12dx), Dy_ptr(p_chi, sy, inv_12dy),
107 Dz_ptr(p_chi, inv_12dz)};
108
109 const T adv = bx * Dx_upwind_ptr(p_theta, sx, inv_2dx, bx) +
110 by * Dy_upwind_ptr(p_theta, sy, inv_2dy, by) +
111 bz * Dz_upwind_ptr(p_theta, inv_2dz, bz);
112
113 const T inv_chi = T(1) / chi;
114 const T grad_ln_sqrt_gamma[3] = {-T(1.5) * d_chi[0] * inv_chi,
115 -T(1.5) * d_chi[1] * inv_chi,
116 -T(1.5) * d_chi[2] * inv_chi};
117 const T div_Z_partial = Dx_ptr(p_Z[0], sx, inv_12dx) +
118 Dy_ptr(p_Z[1], sy, inv_12dy) + Dz_ptr(p_Z[2], inv_12dz);
119 const T div_Z_conn = z_x * grad_ln_sqrt_gamma[0] + z_y * grad_ln_sqrt_gamma[1] +
120 z_z * grad_ln_sqrt_gamma[2];
121 const T div_Z = div_Z_partial + div_Z_conn;
122
123 const T g_xx = *p_ginv[0];
124 const T g_xy = *p_ginv[1];
125 const T g_xz = *p_ginv[2];
126 const T g_yy = *p_ginv[3];
127 const T g_yz = *p_ginv[4];
128 const T g_zz = *p_ginv[5];
129
130 const T R_xx = *p_R[0] + RicciZ4[0];
131 const T R_xy = *p_R[1] + RicciZ4[1];
132 const T R_xz = *p_R[2] + RicciZ4[2];
133 const T R_yy = *p_R[3] + RicciZ4[3];
134 const T R_yz = *p_R[4] + RicciZ4[4];
135 const T R_zz = *p_R[5] + RicciZ4[5];
136
137 const T R_conformal = g_xx * R_xx + g_yy * R_yy + g_zz * R_zz +
138 T(2) * (g_xy * R_xy + g_xz * R_xz + g_yz * R_yz);
139 const T R_scalar = chi * R_conformal;
140
141 const T row0_x = g_xx;
142 const T row0_y = g_xy;
143 const T row0_z = g_xz;
144 const T row1_x = g_xy;
145 const T row1_y = g_yy;
146 const T row1_z = g_yz;
147 const T row2_x = g_xz;
148 const T row2_y = g_yz;
149 const T row2_z = g_zz;
150
151 const T A_xx = *p_A[0];
152 const T A_xy = *p_A[1];
153 const T A_xz = *p_A[2];
154 const T A_yy = *p_A[3];
155 const T A_yz = *p_A[4];
156 const T A_zz = *p_A[5];
157
158 const T tmp0_x = A_xx * row0_x + A_xy * row0_y + A_xz * row0_z;
159 const T tmp0_y = A_xy * row0_x + A_yy * row0_y + A_yz * row0_z;
160 const T tmp0_z = A_xz * row0_x + A_yz * row0_y + A_zz * row0_z;
161
162 const T tmp1_x = A_xx * row1_x + A_xy * row1_y + A_xz * row1_z;
163 const T tmp1_y = A_xy * row1_x + A_yy * row1_y + A_yz * row1_z;
164 const T tmp1_z = A_xz * row1_x + A_yz * row1_y + A_zz * row1_z;
165
166 const T tmp2_x = A_xx * row2_x + A_xy * row2_y + A_xz * row2_z;
167 const T tmp2_y = A_xy * row2_x + A_yy * row2_y + A_yz * row2_z;
168 const T tmp2_z = A_xz * row2_x + A_yz * row2_y + A_zz * row2_z;
169
170 const T A_up_xx = row0_x * tmp0_x + row0_y * tmp0_y + row0_z * tmp0_z;
171 const T A_up_xy = row0_x * tmp1_x + row0_y * tmp1_y + row0_z * tmp1_z;
172 const T A_up_xz = row0_x * tmp2_x + row0_y * tmp2_y + row0_z * tmp2_z;
173 const T A_up_yy = row1_x * tmp1_x + row1_y * tmp1_y + row1_z * tmp1_z;
174 const T A_up_yz = row1_x * tmp2_x + row1_y * tmp2_y + row1_z * tmp2_z;
175 const T A_up_zz = row2_x * tmp2_x + row2_y * tmp2_y + row2_z * tmp2_z;
176
177 const T A_contract = A_xx * A_up_xx + A_yy * A_up_yy + A_zz * A_up_zz +
178 T(2) * (A_xy * A_up_xy + A_xz * A_up_xz + A_yz * A_up_yz);
179
180 const T geom_source =
181 T(0.5) * (R_scalar + T(2) * div_Z - A_contract + (T(2) / T(3)) * K_val * K_val);
182 const T geom = alpha * std::clamp(geom_source, T(-1e3), T(1e3));
183 const T damping = -alpha * params.kappa1 * two_plus_kappa2 * theta;
184 const T diss = KO6_axis_ptr(p_theta, sx) + KO6_axis_ptr(p_theta, sy) +
185 KO6_axis_ptr(p_theta, 1);
186
187 *p_rhs = geom + damping + adv + (ko_sigma / G.dx) * diss;
188
189 ++p_theta;
190 ++p_alpha;
191 ++p_K;
192 ++p_rhs;
193 ++p_chi;
194 for (int c = 0; c < 3; ++c) {
195 ++p_beta[c];
196 ++p_Z[c];
197 }
198 for (int s = 0; s < 6; ++s) {
199 ++p_ginv[s];
200 ++p_A[s];
201 ++p_R[s];
202 }
203 }
204 }
205 }

211template <typename T>
212inline void compute_rhs_Z(const BSSNGridSoA<T> &G, Field3D<T> rhs_Z[3],
213 const GaugeParameters<T> &params = {}, size_t padding = 4) {
214 BSSN_PROFILE_KERNEL(Z);
215 using namespace tensorium_RG::fd;
216 const T ko_sigma = scaled_ko_sigma(params.ko_sigma); // Adaptive KO6 for Z^i.
217
218 size_t I0, I1, J0, J1, K0, K1;
219 G.domain_bounds(I0, I1, J0, J1, K0, K1);
220
221 const size_t i0 = clamped_lower(I0, padding, I1);
222 const size_t j0 = clamped_lower(J0, padding, J1);
223 const size_t k0 = clamped_lower(K0, padding, K1);
224 const size_t i1 = clamped_upper(I1, padding, I0);
225 const size_t j1 = clamped_upper(J1, padding, J0);
226 const size_t k1 = clamped_upper(K1, padding, K0);
227
228 if (i0 >= i1 || j0 >= j1 || k0 >= k1)
229 return;
230
231 const double inv_12dx = 1.0 / (60.0 * G.dx);
232 const double inv_12dy = 1.0 / (60.0 * G.dy);
233 const double inv_12dz = 1.0 / (60.0 * G.dz);
234 const double inv_2dx = 1.0 / (2.0 * G.dx);
235 const double inv_2dy = 1.0 / (2.0 * G.dy);
236 const double inv_2dz = 1.0 / (2.0 * G.dz);
237
238 const ptrdiff_t sx = G.Z[0].st.sx;
239 const ptrdiff_t sy = G.Z[0].st.sy;
240 const T two_thirds = T(2) / T(3);
241
242#pragma omp parallel for collapse(2)
243 for (size_t i = i0; i < i1; ++i) {
244 for (size_t j = j0; j < j1; ++j) {
245 size_t idx_start = G.Z[0].idx(i, j, k0);
246
247 const T *p_Z[3] = {G.Z[0].ptr() + idx_start, G.Z[1].ptr() + idx_start,
248 G.Z[2].ptr() + idx_start};
249 const T *p_beta[3] = {G.beta[0].ptr() + idx_start, G.beta[1].ptr() + idx_start,
250 G.beta[2].ptr() + idx_start};
251 const T *p_theta = G.Theta.ptr() + idx_start;
252 const T *p_alpha = G.alpha.ptr() + idx_start;
253 const T *p_K = G.K.ptr() + idx_start;
254 const T *p_chi = G.chi.ptr() + idx_start;
255 const T *p_ginv[6];
256 const T *p_A[6];
257 for (int s = 0; s < 6; ++s) {
258 p_ginv[s] = G.gamma_tilde_inv[s].ptr() + idx_start;
259 p_A[s] = G.A_tilde[s].ptr() + idx_start;
260 }
261
262 T *p_rhs[3] = {rhs_Z[0].ptr() + idx_start, rhs_Z[1].ptr() + idx_start,
263 rhs_Z[2].ptr() + idx_start};
264
265 for (size_t k = k0; k < k1; ++k) {
266 const T alpha = *p_alpha;
267 const T bx = *p_beta[0];
268 const T by = *p_beta[1];
269 const T bz = *p_beta[2];
270
271 const T g_xx = *p_ginv[0];
272 const T g_xy = *p_ginv[1];
273 const T g_xz = *p_ginv[2];
274 const T g_yy = *p_ginv[3];
275 const T g_yz = *p_ginv[4];
276 const T g_zz = *p_ginv[5];
277
278 const T row0_x = g_xx;
279 const T row0_y = g_xy;
280 const T row0_z = g_xz;
281 const T row1_x = g_xy;
282 const T row1_y = g_yy;
283 const T row1_z = g_yz;
284 const T row2_x = g_xz;
285 const T row2_y = g_yz;
286 const T row2_z = g_zz;
287
288 const T A_xx = *p_A[0];
289 const T A_xy = *p_A[1];
290 const T A_xz = *p_A[2];
291 const T A_yy = *p_A[3];
292 const T A_yz = *p_A[4];
293 const T A_zz = *p_A[5];
294
295 T A_lo[3][3];
296 A_lo[0][0] = A_xx;
297 A_lo[0][1] = A_lo[1][0] = A_xy;
298 A_lo[0][2] = A_lo[2][0] = A_xz;
299 A_lo[1][1] = A_yy;
300 A_lo[1][2] = A_lo[2][1] = A_yz;
301 A_lo[2][2] = A_zz;
302
303 const T tmp0_x = A_xx * row0_x + A_xy * row0_y + A_xz * row0_z;
304 const T tmp0_y = A_xy * row0_x + A_yy * row0_y + A_yz * row0_z;
305 const T tmp0_z = A_xz * row0_x + A_yz * row0_y + A_zz * row0_z;
306
307 const T tmp1_x = A_xx * row1_x + A_xy * row1_y + A_xz * row1_z;
308 const T tmp1_y = A_xy * row1_x + A_yy * row1_y + A_yz * row1_z;
309 const T tmp1_z = A_xz * row1_x + A_yz * row1_y + A_zz * row1_z;
310
311 const T tmp2_x = A_xx * row2_x + A_xy * row2_y + A_xz * row2_z;
312 const T tmp2_y = A_xy * row2_x + A_yy * row2_y + A_yz * row2_z;
313 const T tmp2_z = A_xz * row2_x + A_yz * row2_y + A_zz * row2_z;
314
315 T A_up_matrix[3][3];
316 A_up_matrix[0][0] = row0_x * tmp0_x + row0_y * tmp0_y + row0_z * tmp0_z;
317 A_up_matrix[0][1] = A_up_matrix[1][0] =
318 row0_x * tmp1_x + row0_y * tmp1_y + row0_z * tmp1_z;
319 A_up_matrix[0][2] = A_up_matrix[2][0] =
320 row0_x * tmp2_x + row0_y * tmp2_y + row0_z * tmp2_z;
321 A_up_matrix[1][1] = row1_x * tmp1_x + row1_y * tmp1_y + row1_z * tmp1_z;
322 A_up_matrix[1][2] = A_up_matrix[2][1] =
323 row1_x * tmp2_x + row1_y * tmp2_y + row1_z * tmp2_z;
324 A_up_matrix[2][2] = row2_x * tmp2_x + row2_y * tmp2_y + row2_z * tmp2_z;
325
326 const T dK[3] = {Dx_ptr(p_K, sx, inv_12dx), Dy_ptr(p_K, sy, inv_12dy),
327 Dz_ptr(p_K, inv_12dz)};
328 const T d_chi[3] = {Dx_ptr(p_chi, sx, inv_12dx), Dy_ptr(p_chi, sy, inv_12dy),
329 Dz_ptr(p_chi, inv_12dz)};
330 const T chi_val = *p_chi;
331 const T z_vals[3] = {*p_Z[0], *p_Z[1], *p_Z[2]};
332 T d6phi[3];
333 grad_6phi_from_chi(d_chi, chi_val, d6phi);
334
335 T beta_grad[3][3];
336 for (int comp = 0; comp < 3; ++comp) {
337 beta_grad[comp][0] = Dx_ptr(p_beta[comp], sx, inv_12dx);
338 beta_grad[comp][1] = Dy_ptr(p_beta[comp], sy, inv_12dy);
339 beta_grad[comp][2] = Dz_ptr(p_beta[comp], inv_12dz);
340 }
341
342 T dA_dx[6];
343 T dA_dy[6];
344 T dA_dz[6];
345 for (int s = 0; s < 6; ++s) {
346 dA_dx[s] = Dx_ptr(p_A[s], sx, inv_12dx);
347 dA_dy[s] = Dy_ptr(p_A[s], sy, inv_12dy);
348 dA_dz[s] = Dz_ptr(p_A[s], inv_12dz);
349 }
350
351 T dA_tensor[3][3][3] = {};
352 for (int s = 0; s < 6; ++s) {
353 const int row = detail::sym_row_index[s];
354 const int col = detail::sym_col_index[s];
355 const T dx = dA_dx[s];
356 const T dy = dA_dy[s];
357 const T dz = dA_dz[s];
358 dA_tensor[0][row][col] = dx;
359 dA_tensor[0][col][row] = dx;
360 dA_tensor[1][row][col] = dy;
361 dA_tensor[1][col][row] = dy;
362 dA_tensor[2][row][col] = dz;
363 dA_tensor[2][col][row] = dz;
364 }
365
366 T Gamma_tilde_vals[3][3][3];
367 tensorium_RG::bssn::compute_tildeGamma_symbols(G, i, j, k, Gamma_tilde_vals);
368
369 T covariant[3][3][3];
370 for (int dir = 0; dir < 3; ++dir)
371 for (int a = 0; a < 3; ++a)
372 for (int b = 0; b < 3; ++b) {
373 T sum = dA_tensor[dir][a][b];
374 for (int m = 0; m < 3; ++m) {
375 sum -= Gamma_tilde_vals[m][a][dir] * A_lo[m][b];
376 sum -= Gamma_tilde_vals[m][b][dir] * A_lo[a][m];
377 }
378 covariant[dir][a][b] = sum;
379 }
380
381 T ginv_matrix[3][3];
382 ginv_matrix[0][0] = g_xx;
383 ginv_matrix[0][1] = ginv_matrix[1][0] = g_xy;
384 ginv_matrix[0][2] = ginv_matrix[2][0] = g_xz;
385 ginv_matrix[1][1] = g_yy;
386 ginv_matrix[1][2] = ginv_matrix[2][1] = g_yz;
387 ginv_matrix[2][2] = g_zz;
388
389 T div_vec[3] = {T(0), T(0), T(0)};
390 for (int ii = 0; ii < 3; ++ii)
391 for (int dir = 0; dir < 3; ++dir)
392 for (int a = 0; a < 3; ++a)
393 for (int b = 0; b < 3; ++b)
394 div_vec[ii] +=
395 ginv_matrix[ii][a] * ginv_matrix[dir][b] * covariant[dir][a][b];
396
397 const T dissZ_coef = ko_sigma / G.dx;
398
399 for (int comp = 0; comp < 3; ++comp) {
400 const T adv = bx * Dx_upwind_ptr(p_Z[comp], sx, inv_2dx, bx) +
401 by * Dy_upwind_ptr(p_Z[comp], sy, inv_2dy, by) +
402 bz * Dz_upwind_ptr(p_Z[comp], inv_2dz, bz);
403
404 const T damping = -params.kappa_z * alpha * (*p_Z[comp]);
405 const T lie_drag =
406 -(z_vals[0] * beta_grad[comp][0] + z_vals[1] * beta_grad[comp][1] +
407 z_vals[2] * beta_grad[comp][2]);
408
409 T phi_term = T(0);
410 for (int dir = 0; dir < 3; ++dir)
411 phi_term += A_up_matrix[comp][dir] * d6phi[dir];
412
413 T gradK = T(0);
414 if (comp == 0)
415 gradK = row0_x * dK[0] + row0_y * dK[1] + row0_z * dK[2];
416 else if (comp == 1)
417 gradK = row1_x * dK[0] + row1_y * dK[1] + row1_z * dK[2];
418 else
419 gradK = row2_x * dK[0] + row2_y * dK[1] + row2_z * dK[2];
420
421 const T gradK_phys = chi_val * gradK;
422 const T momentum = div_vec[comp] + phi_term - two_thirds * gradK_phys;
423 const T diss = KO6_axis_ptr(p_Z[comp], sx) + KO6_axis_ptr(p_Z[comp], sy) +
424 KO6_axis_ptr(p_Z[comp], 1);
425
426 *p_rhs[comp] = alpha * momentum + adv + lie_drag + damping + dissZ_coef * diss;
427 }
428
429 for (int c = 0; c < 3; ++c) {
430 ++p_beta[c];
431 ++p_Z[c];
432 ++p_rhs[c];
433 }
434 ++p_theta;
435 ++p_alpha;
436 ++p_K;
437 ++p_chi;
438 for (int s = 0; s < 6; ++s) {
439 ++p_ginv[s];
440 ++p_A[s];
441 }
442 }
443 }
444 }
