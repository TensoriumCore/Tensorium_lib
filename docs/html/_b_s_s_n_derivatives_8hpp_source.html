37template <typename T>
38void spectral_derivative_1D(tensorium::Vector<T> &field, tensorium::Vector<T> &dfield, T dx) {
39 using C = std::complex<T>;
40 using FFT = tensorium::SpectralFFT<T>;
41
42 tensorium::Vector<C> field_fft(field.size());
43 for (size_t i = 0; i < field.size(); ++i)
44 field_fft(i) = field(i);
45
46 FFT::forward(field_fft);
47
48 const T L = dx * field.size();
49 for (size_t k = 0; k < field.size(); ++k) {
50 int n = (k <= field.size() / 2) ? k : (int)k - (int)field.size();
51 C factor = C(0, 2.0 * M_PI * n / L);
52 field_fft(k) *= factor;
53 }
54
55 FFT::backward(field_fft);
56 for (size_t i = 0; i < field.size(); ++i)
57 dfield(i) = field_fft(i).real();
58}

60template <typename T>
61void spectral_partial_scalar_3D(const tensorium::Tensor<T, 3> &scalar_field, T dx, T dy, T dz,
62 tensorium::Tensor<T, 4> &grad_out) {
63 using namespace tensorium;
64 using C = std::complex<T>;
65 using FFT = SpectralFFT<T>;
66
67 const auto &shape = scalar_field.shape();
68 const size_t NX = shape[0], NY = shape[1], NZ = shape[2];
69
70 grad_out.resize({NX, NY, NZ, 3});
71
72 Tensor<C, 3> field_cplx({NX, NY, NZ});
73 for (size_t i = 0; i < NX; ++i)
74 for (size_t j = 0; j < NY; ++j)
75 for (size_t k = 0; k < NZ; ++k)
76 field_cplx(i, j, k) = C(scalar_field(i, j, k), 0.0);
77
78 FFT::forward_3D(field_cplx);
79
80 for (size_t i = 0; i < NX; ++i) {
81 int ni = (i <= NX / 2) ? int(i) : int(i) - int(NX);
82 T kx = 2.0 * M_PI * ni / (NX * dx);
83
84 for (size_t j = 0; j < NY; ++j) {
85 int nj = (j <= NY / 2) ? int(j) : int(j) - int(NY);
86 T ky = 2.0 * M_PI * nj / (NY * dy);
87
88 for (size_t k = 0; k < NZ; ++k) {
89 int nk = (k <= NZ / 2) ? int(k) : int(k) - int(NZ);
90 T kz = 2.0 * M_PI * nk / (NZ * dz);
91
92 C f_hat = field_cplx(i, j, k);
93
94 field_cplx(i, j, k) = f_hat;
95
96 grad_out(i, j, k, 0) = (f_hat * C(0, kx)).real();
97 grad_out(i, j, k, 1) = (f_hat * C(0, ky)).real();
98 grad_out(i, j, k, 2) = (f_hat * C(0, kz)).real();
99 }
100 }
101 }
102
103 for (int dim = 0; dim < 3; ++dim) {
104 Tensor<C, 3> dfield_cplx({NX, NY, NZ});
105
106 for (size_t i = 0; i < NX; ++i)
107 for (size_t j = 0; j < NY; ++j)
108 for (size_t k = 0; k < NZ; ++k)
109 dfield_cplx(i, j, k) = C(grad_out(i, j, k, dim), 0.0);
110
111 FFT::backward_3D(dfield_cplx);
112
113 for (size_t i = 0; i < NX; ++i)
114 for (size_t j = 0; j < NY; ++j)
115 for (size_t k = 0; k < NZ; ++k)
116 grad_out(i, j, k, dim) = dfield_cplx(i, j, k).real();
117 }
118}

147template <typename T, typename ScalarFunc>
148tensorium::Vector<T> partial_scalar(const tensorium::Vector<T> &X, T dx, T dy, T dz,
149 ScalarFunc &&func) {
150 tensorium::Vector<T> grad(3);
151 const auto idx = detail::spatial_indices(X);
152 const std::array<T, 3> spacings{dx, dy, dz};
153
154 for (size_t axis = 0; axis < 3; ++axis) {
155 const T h = spacings[axis];
156 auto sample = [&](int step) {
157 auto Xs = X;
158 detail::shift_axis(Xs, idx, axis, step * h);
159 return func(Xs);
160 };
161
162 T gm2 = sample(-2);
163 T gm1 = sample(-1);
164 T gp1 = sample(1);
165 T gp2 = sample(2);
166 grad(axis) = (-gp2 + 8 * gp1 - 8 * gm1 + gm2) / (12 * h);
167 }

393template <typename T>
394tensorium::Tensor<T, 2> compute_dt_gamma_from_beta(const tensorium::Vector<T> &beta_cov,
395 const tensorium::Tensor<T, 2> &partial_beta_cov,
396 const tensorium::Tensor<T, 3> &christoffel) {
397
398 tensorium::Tensor<T, 2> dtg({3, 3});
399 for (size_t i = 0; i < 3; ++i)
400 for (size_t j = 0; j < 3; ++j) {
401 T Di_bj = partial_beta_cov(i, j);
402 T Dj_bi = partial_beta_cov(j, i);
403 for (size_t k = 0; k < 3; ++k) {
404 Di_bj -= christoffel(k, i, j) * beta_cov(k);
405 Dj_bi -= christoffel(k, j, i) * beta_cov(k);
406 }
407 dtg(i, j) = Di_bj + Dj_bi;
408 }
409 return dtg;
410}

523template <typename T, typename ScalarFunc>
524tensorium::Tensor<T, 2> covariant_scalar_second(const tensorium::Vector<T> &X, T dx, T dy, T dz,
525 ScalarFunc &&func,
526 const tensorium::Tensor<T, 3> &christoffel) {
527 using namespace tensorium;
528
529 Vector<T> grad(3);
530 compute_partial_derivatives_scalar(X, dx, dy, dz, func, grad);
531
532 Tensor<T, 2> hess({3, 3});
533 compute_second_derivatives_scalar(X, dx, dy, dz, std::forward<ScalarFunc>(func), hess);
534
535 Tensor<T, 2> result({3, 3});
536 for (int i = 0; i < 3; ++i)
537 for (int j = 0; j < 3; ++j) {
538 result(i, j) = hess(i, j);
539 for (int k = 0; k < 3; ++k)
540 result(i, j) -=
541 christoffel(k, i, j) * grad(k); // D_i D_j φ = ∂_i ∂_j φ - Γ^k_{ij} ∂_k φ
542 }
543 return result;
544}

552 hessian_out.resize(NX, NY, NZ, 3, 3);
553
554 auto get = [&](int i, int j, int k) -> T {
555 if (i < 0)
556 i = 0;
557 if (j < 0)
558 j = 0;
559 if (k < 0)
560 k = 0;
561 if (i >= NX)
562 i = NX - 1;
563 if (j >= NY)
564 j = NY - 1;
565 if (k >= NZ)
566 k = NZ - 1;
567 return scalar_field(i, j, k);
568 };
569
570 for (size_t i = 0; i < NX; ++i) {
571 for (size_t j = 0; j < NY; ++j) {
572 for (size_t k = 0; k < NZ; ++k) {
573
574 auto d2f_xx = (-get(i + 2, j, k) + 16 * get(i + 1, j, k) - 30 * get(i, j, k) +
575 16 * get(i - 1, j, k) - get(i - 2, j, k)) /
576 (12 * dx * dx);
577 auto d2f_yy = (-get(i, j + 2, k) + 16 * get(i, j + 1, k) - 30 * get(i, j, k) +
578 16 * get(i, j - 1, k) - get(i, j - 2, k)) /
579 (12 * dy * dy);
580 auto d2f_zz = (-get(i, j, k + 2) + 16 * get(i, j, k + 1) - 30 * get(i, j, k) +
581 16 * get(i, j, k - 1) - get(i, j, k - 2)) /
582 (12 * dz * dz);
583
584 auto d2f_xy = (get(i + 1, j + 1, k) - get(i + 1, j - 1, k) - get(i - 1, j + 1, k) +
585 get(i - 1, j - 1, k)) /
586 (4 * dx * dy);
587 auto d2f_xz = (get(i + 1, j, k + 1) - get(i + 1, j, k - 1) - get(i - 1, j, k + 1) +
588 get(i - 1, j, k - 1)) /
589 (4 * dx * dz);
590 auto d2f_yz = (get(i, j + 1, k + 1) - get(i, j + 1, k - 1) - get(i, j - 1, k + 1) +
591 get(i, j - 1, k - 1)) /
592 (4 * dy * dz);
593
594 hessian_out(i, j, k, 0, 0) = d2f_xx;
595 hessian_out(i, j, k, 1, 1) = d2f_yy;
596 hessian_out(i, j, k, 2, 2) = d2f_zz;
597
598 hessian_out(i, j, k, 0, 1) = hessian_out(i, j, k, 1, 0) = d2f_xy;
599 hessian_out(i, j, k, 0, 2) = hessian_out(i, j, k, 2, 0) = d2f_xz;
600 hessian_out(i, j, k, 1, 2) = hessian_out(i, j, k, 2, 1) = d2f_yz;
601 }
602 }
603 }
604}

606template <typename T>
607tensorium::Tensor<T, 5> covariant_scalar_second_3D(const tensorium::Tensor<T, 3> &chi,
608 const tensorium::Tensor<T, 5> &christoffel, T dx,
609 T dy, T dz) {
610 const auto &shape = chi.shape();
611 const size_t NX = shape[0], NY = shape[1], NZ = shape[2];
612
613 tensorium::Tensor<T, 5> result({NX, NY, NZ, 3, 3});
614
615 tensorium::Vector<T> grad(3);
616 tensorium::Tensor<T, 2> hess({3, 3});
617
618 for (size_t i = 0; i < NX; ++i)
619 for (size_t j = 0; j < NY; ++j)
620 for (size_t k = 0; k < NZ; ++k) {
621 auto func = [&](const tensorium::Vector<T> &X) -> T { return chi(i, j, k); };
622
623 compute_partial_derivatives_scalar({T(i), T(j), T(k)}, dx, dy, dz, func, grad);
624 compute_second_derivatives_scalar({T(i), T(j), T(k)}, dx, dy, dz, func, hess);
625
626 for (int a = 0; a < 3; ++a)
627 for (int b = 0; b < 3; ++b) {
628 T val = hess(a, b);
629 for (int c = 0; c < 3; ++c)
630 val -= christoffel(i, j, k, c, a, b) * grad(c);
631 result(i, j, k, a, b) = val;
632 }
633 }
634
635 return result;
636}

639tensorium::Tensor<T, 2> covariant_vector(const tensorium::Vector<T> &X, T dx, T dy, T dz,
640 VectorFunc &&func, const tensorium::Tensor<T, 3> &Gamma) {

655template <typename T, typename TensorFunc>
656tensorium::Tensor<T, 3> covariant_tensor2(const tensorium::Vector<T> &X, T dx, T dy, T dz,
657 TensorFunc &&func, const tensorium::Tensor<T, 3> &Gamma) {
658 auto dTij = partial_tensor2(X, dx, dy, dz, std::forward<TensorFunc>(func));
659 tensorium::Tensor<T, 2> Tij = func(X);
660 tensorium::Tensor<T, 3> result({3, 3, 3});
661
662 for (int i = 0; i < 3; ++i)
663 for (int j = 0; j < 3; ++j)
664 for (int k = 0; k < 3; ++k) {
665 result(i, j, k) = dTij(i, j, k);
666 for (int l = 0; l < 3; ++l)
667 result(i, j, k) += -Gamma(l, k, i) * Tij(l, j) - Gamma(l, k, j) * Tij(i, l);
668 }
669
670 return result;
671}

673template <typename T, typename TensorFunc>
674tensorium::Tensor<T, 3> partial_tensor2(const tensorium::Vector<T> &X, T dx, T dy, T dz,
675 TensorFunc &&func) {
676 tensorium::Tensor<T, 3> result({3, 3, 3});
677 const auto idx = detail::spatial_indices(X);
678 const std::array<T, 3> spacings{dx, dy, dz};
679
680 for (int axis = 0; axis < 3; ++axis) {
681 const T h = spacings[axis];
682 auto Xm = X;
683 auto Xp = X;
684 detail::shift_axis(Xm, idx, axis, -h);
685 detail::shift_axis(Xp, idx, axis, h);
686
687 tensorium::Tensor<T, 2> fm = func(Xm);
688 tensorium::Tensor<T, 2> fp = func(Xp);
689
690 for (int i = 0; i < 3; ++i)
691 for (int j = 0; j < 3; ++j)
692 result(i, j, axis) = (fp(i, j) - fm(i, j)) / (2 * h);
693 }
694
695 return result;
696}

698template <typename T, typename TensorFunc>
699tensorium::Tensor<T, 4> covariant_tensor2_second(const tensorium::Vector<T> &X, T dx, T dy, T dz,
700 TensorFunc &&func,
701 const tensorium::Tensor<T, 3> &Gamma) {
702 auto cov1 = [&](const tensorium::Vector<T> &Y) {
703 return covariant_tensor2<T>(Y, dx, dy, dz, func, Gamma);
704 };
705
706 const auto idx = detail::spatial_indices(X);
707 tensorium::Tensor<T, 4> result({3, 3, 3, 3}); // ∇_i ∇_j T_{kl}
708
709 for (int i = 0; i < 3; ++i)
710 for (int j = 0; j < 3; ++j)
711 for (int k = 0; k < 3; ++k)
712 for (int l = 0; l < 3; ++l) {
713 auto X_minus = X;
714 auto X_plus = X;
715 const T h = (j == 0) ? dx : (j == 1 ? dy : dz);
716 detail::shift_axis(X_minus, idx, j, -h);
717 detail::shift_axis(X_plus, idx, j, h);
718
719 T cov_minus = cov1(X_minus)(k, l, i);
720 T cov_plus = cov1(X_plus)(k, l, i);
721 result(i, j, k, l) = (cov_plus - cov_minus) / (2 * h); // ∂_j (∇_i T_kl)
722 }
723
724 return result;
725}
