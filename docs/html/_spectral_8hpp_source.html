170 static void transform_impl(CVectorT &a, bool inverse) {
171 const std::size_t N = a.size();
172 if (N <= 1 || (N & (N - 1)))
173 throw std::invalid_argument("SpectralFFT: size must be power of two");
174
175 bit_reverse(a);
176
177 const auto &twiddles = get_twiddles(N, inverse);
178
179 for (std::size_t len = 2; len <= N; len <<= 1) {
180 const std::size_t half = len >> 1;
181 const std::size_t step = N / len;
182
183#pragma omp parallel for schedule(static) if (len >= 64)
184 for (std::size_t i = 0; i < N; i += len) {
185 for (std::size_t j = 0; j < half; ++j) {
186 C t = a[i + j + half] * twiddles[j * step];
187 C u = a[i + j];
188 a[i + j] = u + t;
189 a[i + j + half] = u - t;
190 }
191 }
192 }
193
194 if (inverse) {
195 const T invN = T(1) / T(N);
196#pragma omp parallel for schedule(static)
197 for (std::size_t i = 0; i < N; ++i)
198 a[i] *= invN;
199 }
200 }

205 static void bit_reverse(CVectorT &a) {
206 const std::size_t N = a.size();
207 for (std::size_t i = 1, j = 0; i < N; ++i) {
208 std::size_t bit = N >> 1;
209 for (; j & bit; bit >>= 1)
210 j ^= bit;
211 j ^= bit;
212 if (i < j)
213 std::swap(a[i], a[j]);
214 }
