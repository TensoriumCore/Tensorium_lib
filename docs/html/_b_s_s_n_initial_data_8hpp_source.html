132template <typename T> void print_ricci_samples(const BSSNGridSoA<T> &G) {
133 size_t I0, I1, J0, J1, K0, K1;
134 G.domain_bounds(I0, I1, J0, J1, K0, K1);
135
136 size_t cx = I0 + G.dims.nx / 2;
137 size_t cy = J0 + G.dims.ny / 2;
138 size_t cz = K0 + G.dims.nz / 2;
139
140 struct Sample {
141 const char *name;
142 size_t i, j, k;
143 };
144 Sample samples[] = {{"Near (r ~ 4*dx)", cx + 4, cy, cz},
145 {"Mid (r ~ N/4) ", cx + G.dims.nx / 4, cy, cz},
146 {"Far (r ~ N/2) ", I1 - 4, cy, cz}};
147
148 printf("\n=== Ricci Tensor Samples ===\n");
149 for (const auto &s : samples) {
150 size_t id = G.alpha.idx(s.i, s.j, s.k);
151
152 T x, y, z;

155
156 T Rxx = G.Ricci[XX].ptr()[id];
157 T Rxy = G.Ricci[XY].ptr()[id];
158 T Rxz = G.Ricci[XZ].ptr()[id];

176template <typename T>
177inline void invert_gamma_tilde(BSSNGridSoA<T> &G, size_t i, size_t j, size_t k) {
178 const size_t id = G.gamma_tilde[XX].idx(i, j, k);
179
180 const T a = G.gamma_tilde[XX].ptr()[id];
181 const T b = G.gamma_tilde[XY].ptr()[id];
182 const T c = G.gamma_tilde[XZ].ptr()[id];
183 const T d = G.gamma_tilde[YY].ptr()[id];
184 const T e = G.gamma_tilde[YZ].ptr()[id];
185 const T f = G.gamma_tilde[ZZ].ptr()[id];
186
187 const T det = a * (d * f - e * e) - b * (b * f - c * e) + c * (b * e - c * d);
188
189 const T inv_det = T(1) / det;
190
191 G.gamma_tilde_inv[XX].ptr()[id] = (d * f - e * e) * inv_det;
192 G.gamma_tilde_inv[XY].ptr()[id] = (c * e - b * f) * inv_det;
193 G.gamma_tilde_inv[XZ].ptr()[id] = (b * e - c * d) * inv_det;
194 G.gamma_tilde_inv[YY].ptr()[id] = (a * f - c * c) * inv_det;
195 G.gamma_tilde_inv[YZ].ptr()[id] = (b * c - a * e) * inv_det;
196 G.gamma_tilde_inv[ZZ].ptr()[id] = (a * d - b * b) * inv_det;
197}

209template <typename T>
210inline void minkowski(BSSNGridSoA<T> &G, T M, T xc = T(0), T yc = T(0), T zc = T(0),
211 T r_floor = T(1e-6)) {
212
213 size_t I0, I1, J0, J1, K0, K1;
214 G.domain_bounds(I0, I1, J0, J1, K0, K1);
215
216#pragma omp parallel for collapse(2)
217 for (size_t i = I0; i < I1; ++i)
218 for (size_t j = J0; j < J1; ++j)
219 for (size_t k = K0; k < K1; ++k) {
220 const size_t id = G.alpha.idx(i, j, k);
221
222 G.alpha.ptr()[id] = T(1);
223 G.chi.ptr()[id] = T(1);
224 G.K.ptr()[id] = T(0);
225
226 for (int c = 0; c < 3; ++c) {
227 G.beta[c].ptr()[id] = T(0);
228 G.B[c].ptr()[id] = T(0);
229 G.tildeGamma[c].ptr()[id] = T(0);
230 }
231
232 const T one = T(1);
233 const T zero = T(0);
234
235 G.gamma_tilde[XX].ptr()[id] = one;
236 G.gamma_tilde[XY].ptr()[id] = zero;
237 G.gamma_tilde[XZ].ptr()[id] = zero;
238 G.gamma_tilde[YY].ptr()[id] = one;
239 G.gamma_tilde[YZ].ptr()[id] = zero;
240 G.gamma_tilde[ZZ].ptr()[id] = one;
241
242 G.gamma_tilde_inv[XX].ptr()[id] = one;
243 G.gamma_tilde_inv[XY].ptr()[id] = zero;
244 G.gamma_tilde_inv[XZ].ptr()[id] = zero;
245 G.gamma_tilde_inv[YY].ptr()[id] = one;
246 G.gamma_tilde_inv[YZ].ptr()[id] = zero;
247 G.gamma_tilde_inv[ZZ].ptr()[id] = one;
248
249 G.A_tilde[XX].ptr()[id] = zero;
250 G.A_tilde[XY].ptr()[id] = zero;
251 G.A_tilde[XZ].ptr()[id] = zero;
252 G.A_tilde[YY].ptr()[id] = zero;
253 G.A_tilde[YZ].ptr()[id] = zero;
254 G.A_tilde[ZZ].ptr()[id] = zero;
255 }
256 bssn::apply_halos_grid<bssn::BoundaryRadiative>(G);
257 tensorium_RG::bssn::compute_tildeGamma_contracted(G);
258 tensorium_RG::bssn::compute_ricci_bssn(G, G.Ricci);
259 double maxAbsR = 0.0;
260 size_t nSamples = 0;
261 double rMin = 1e300, rMax = 0.0;
262
263 for (size_t i = I0 + 4; i < I1 - 4; ++i)
264 for (size_t j = J0 + 4; j < J1 - 4; ++j)
265 for (size_t k = K0 + 4; k < K1 - 4; ++k) {
266
267 double x, y, z;
268 G.coords(i, j, k, x, y, z);
269 x -= xc;
270 y -= yc;
271 z -= zc;
272
273 const double r = std::sqrt(x * x + y * y + z * z);
274
275 const double r_min = 2.0;
276
277 const double r_max = 0.45 * std::min({(I1 - I0 - 1) * G.dx, (J1 - J0 - 1) * G.dy,
278 (K1 - K0 - 1) * G.dz});
279 if (r <= r_min || r >= r_max)
280 continue;
281
282 const size_t id = G.alpha.idx(i, j, k);
283
284 const double chi = G.chi.ptr()[id];
285
286 const double ginv_xx = chi * G.gamma_tilde_inv[XX].ptr()[id];
287 const double ginv_xy = chi * G.gamma_tilde_inv[XY].ptr()[id];
288 const double ginv_xz = chi * G.gamma_tilde_inv[XZ].ptr()[id];
289 const double ginv_yy = chi * G.gamma_tilde_inv[YY].ptr()[id];
290 const double ginv_yz = chi * G.gamma_tilde_inv[YZ].ptr()[id];
291 const double ginv_zz = chi * G.gamma_tilde_inv[ZZ].ptr()[id];
292
293 const double Rxx = G.Ricci[XX].ptr()[id];
294 const double Rxy = G.Ricci[XY].ptr()[id];
295 const double Rxz = G.Ricci[XZ].ptr()[id];
296 const double Ryy = G.Ricci[YY].ptr()[id];
297 const double Ryz = G.Ricci[YZ].ptr()[id];
298 const double Rzz = G.Ricci[ZZ].ptr()[id];
299
300 const double R = ginv_xx * Rxx + 2.0 * ginv_xy * Rxy + 2.0 * ginv_xz * Rxz +
301 ginv_yy * Ryy + 2.0 * ginv_yz * Ryz + ginv_zz * Rzz;
302 maxAbsR = std::max(maxAbsR, std::abs(R));
303 nSamples++;
304
305 if (nSamples == 1) {
306 printf("[DBG] chi=%g Rxx=%g Ryy=%g Rzz=%g\n", chi, Rxx, Ryy, Rzz);
307 }
308 rMin = std::min(rMin, r);
309 rMax = std::max(rMax, r);
310 }
311
312 printf("[CHECK] samples=%zu, rMin=%g, rMax=%g\n", nSamples, rMin, rMax);
313 printf("[CHECK] max |R| for r > %.3f = %.3e\n", rMin, maxAbsR);
314 print_ricci_samples(G);
315 auto constraints = make_constraint_scratch(G);
316 tensorium_RG::bssn::compute_bssn_constraints(G, G.Ricci, constraints.H, constraints.M,
317 constraints.C, 2.0, rMax, xc, yc, zc);
318
319 tensorium_RG::bssn::print_constraint_norms(G, constraints.H, constraints.M, constraints.C, 2.0,
320 rMax, xc, yc, zc);
321 tensorium_RG::bssn::assert_invariants(G, "init.minkowski");
322 tensorium_RG::bssn::project_bssn_state(G);
323}

337template <typename T>
338inline void schwarzschild_isotropic(BSSNGridSoA<T> &G, T M, T xc = T(0), T yc = T(0), T zc = T(0),
339 T r_floor = T(1e-6)) {
340
341 size_t I0, I1, J0, J1, K0, K1;
342 G.domain_bounds(I0, I1, J0, J1, K0, K1);
343 const T one = T(1);
344 const T zero = T(0);
345 const T half = T(0.5);
346 zero_z4c_fields(G);
347
348#pragma omp parallel for collapse(3)
349 for (size_t i = I0; i < I1; ++i)
350 for (size_t j = J0; j < J1; ++j)
351 for (size_t k = K0; k < K1; ++k) {
352 const size_t id = G.alpha.idx(i, j, k);
353
354 T x, y, z;
355 G.coords(i, j, k, x, y, z);
356 x -= xc;
357 y -= yc;
358 z -= zc;
359
360 T r2 = x * x + y * y + z * z;
361 T r = std::sqrt(r2);
362 if (r < r_floor)
363 r = r_floor;
364
365 const T u = (M * half) / r; // M/(2r)
366 const T psi = one + u;
367
368 const T chi = one / (psi * psi * psi * psi); // psi^-4
369 const T alpha = one / (psi * psi); // psi^-2 = sqrt(chi)
370
371 G.alpha.ptr()[id] = alpha;
372 G.chi.ptr()[id] = chi;
373 G.K.ptr()[id] = zero;
374
375 for (int c = 0; c < 3; ++c) {
376 G.beta[c].ptr()[id] = zero;
377 G.B[c].ptr()[id] = zero;
378 G.tildeGamma[c].ptr()[id] = zero;

384 G.gamma_tilde[YY].ptr()[id] = one;
385 G.gamma_tilde[YZ].ptr()[id] = zero;
386 G.gamma_tilde[ZZ].ptr()[id] = one;
387
388 invert_gamma_tilde(G, i, j, k);
389
390 G.A_tilde[XX].ptr()[id] = zero;
391 G.A_tilde[XY].ptr()[id] = zero;
392 G.A_tilde[XZ].ptr()[id] = zero;
393 G.A_tilde[YY].ptr()[id] = zero;

401 tensorium_RG::bssn::compute_ricci_bssn(G, G.Ricci); // R_ij

414 y -= yc;
415 z -= zc;
416

582 const double Rzz = G.Ricci[ZZ].ptr()[id];

584 const double R = ginv_xx * Rxx + 2.0 * ginv_xy * Rxy + 2.0 * ginv_xz * Rxz +
585 ginv_yy * Ryy + 2.0 * ginv_yz * Ryz + ginv_zz * Rzz;

587 maxAbsR = std::max(maxAbsR, std::abs(R));
588 nSamples++;
589 rMin = std::min(rMin, r);
590 rMax = std::max(rMax, r);
591 }

593 printf("[CHECK] samples=%zu, rMin=%g, rMax=%g\n", nSamples, rMin, rMax);
594 printf("[CHECK] max |R| for r > %.3f = %.3e\n", rMin, maxAbsR);

596
597 auto constraints = make_constraint_scratch(G);
598 tensorium_RG::bssn::compute_bssn_constraints(G, G.Ricci, constraints.H, constraints.M,
599 constraints.C, 2.0, rMax, 0.0, 0.0, 0.0);
600 tensorium_RG::bssn::print_constraint_norms(G, constraints.H, constraints.M, constraints.C, 2.0,
601 rMax, 0.0, 0.0, 0.0);
602 tensorium_RG::bssn::assert_invariants(G, "init.binary_schwarzschild");

678template <typename T>
679static inline void ks_kerr_r_H_l(T x, T y, T z, T M, T a, T &r, T &H, T l[3]) {
680 const T rho2 = x * x + y * y + z * z;
681 const T a2 = a * a;
682
683 const T A = rho2 - a2;
684 const T B = std::sqrt(A * A + T(4) * a2 * z * z);
685 const T r2 = T(0.5) * (A + B);
686 r = std::sqrt(std::max(r2, T(0)));
687
688 const T denom = r * r + a2;
689 const T inv_denom = T(1) / denom;
690
691 l[0] = (r * x + a * y) * inv_denom;
692 l[1] = (r * y - a * x) * inv_denom;
693 l[2] = z / std::max(r, T(1e-30));
694
695 const T r4 = r2 * r2;
696 H = (M * r * r2) / (r4 + a2 * z * z);
697}

699template <typename T>
700static inline void invert_sym_3x3(const T g_xx, const T g_xy, const T g_xz, const T g_yy,
701 const T g_yz, const T g_zz, T &inv_xx, T &inv_xy, T &inv_xz,
702 T &inv_yy, T &inv_yz, T &inv_zz, T &det) {
703 const T a = g_xx, b = g_xy, c = g_xz;
704 const T d = g_yy, e = g_yz, f = g_zz;
705
706 det = a * (d * f - e * e) - b * (b * f - c * e) + c * (b * e - c * d);
707 const T inv_det = T(1) / det;
708
709 inv_xx = (d * f - e * e) * inv_det;
710 inv_xy = (c * e - b * f) * inv_det;
711 inv_xz = (b * e - c * d) * inv_det;
712 inv_yy = (a * f - c * c) * inv_det;
713 inv_yz = (b * c - a * e) * inv_det;
714 inv_zz = (a * d - b * b) * inv_det;
715}

717template <typename T>
718inline void kerr_schild_single(BSSNGridSoA<T> &G, T M, T a, T xc = T(0), T yc = T(0), T zc = T(0),
719 T r_floor = T(1e-6)) {
720
721 size_t I0, I1, J0, J1, K0, K1;
722 G.domain_bounds(I0, I1, J0, J1, K0, K1);
723 zero_z4c_fields(G);
724
725 const T one = T(1);
726 const T zero = T(0);
727
728#pragma omp parallel for collapse(3)
729 for (size_t i = I0; i < I1; ++i)
730 for (size_t j = J0; j < J1; ++j)
731 for (size_t k = K0; k < K1; ++k) {
732
733 const size_t id = G.alpha.idx(i, j, k);
734
735 T x, y, z;
736 G.coords(i, j, k, x, y, z);
737 x -= xc;
738 y -= yc;
739 z -= zc;
740
741 const T rr = std::sqrt(x * x + y * y + z * z);
742 if (rr < r_floor) {
743 const T s = r_floor / std::max(rr, T(1e-30));
744 x *= s;
745 y *= s;
746 z *= s;
747 }
748
749 T r, H;
750 T l[3];
751 ks_kerr_r_H_l(x, y, z, M, a, r, H, l);
752
753 const T alpha = one / std::sqrt(one + T(2) * H);
754
755 const T g_xx = one + T(2) * H * l[0] * l[0];
756 const T g_xy = T(2) * H * l[0] * l[1];
757 const T g_xz = T(2) * H * l[0] * l[2];
758 const T g_yy = one + T(2) * H * l[1] * l[1];
759 const T g_yz = T(2) * H * l[1] * l[2];
760 const T g_zz = one + T(2) * H * l[2] * l[2];
761
762 T ginv_xx, ginv_xy, ginv_xz, ginv_yy, ginv_yz, ginv_zz, detg;
763 invert_sym_3x3(g_xx, g_xy, g_xz, g_yy, g_yz, g_zz, ginv_xx, ginv_xy, ginv_xz,
764 ginv_yy, ginv_yz, ginv_zz, detg);
765
766 const T chi = std::pow(detg, T(-1) / T(3));
767 const T inv_chi = one / chi;
768
769 G.alpha.ptr()[id] = alpha;
770 G.chi.ptr()[id] = chi;
771
772 const T beta_cov_x = T(2) * H * l[0];
773 const T beta_cov_y = T(2) * H * l[1];
774 const T beta_cov_z = T(2) * H * l[2];
775
776 const T beta_x = ginv_xx * beta_cov_x + ginv_xy * beta_cov_y + ginv_xz * beta_cov_z;
777 const T beta_y = ginv_xy * beta_cov_x + ginv_yy * beta_cov_y + ginv_yz * beta_cov_z;
778 const T beta_z = ginv_xz * beta_cov_x + ginv_yz * beta_cov_y + ginv_zz * beta_cov_z;
779
780 G.beta[0].ptr()[id] = beta_x;
781 G.beta[1].ptr()[id] = beta_y;
782 G.beta[2].ptr()[id] = beta_z;
783
784 G.B[0].ptr()[id] = zero;
785 G.B[1].ptr()[id] = zero;
786 G.B[2].ptr()[id] = zero;
787
788 G.gamma_tilde[XX].ptr()[id] = chi * g_xx;
789 G.gamma_tilde[XY].ptr()[id] = chi * g_xy;
790 G.gamma_tilde[XZ].ptr()[id] = chi * g_xz;
791 G.gamma_tilde[YY].ptr()[id] = chi * g_yy;
792 G.gamma_tilde[YZ].ptr()[id] = chi * g_yz;
793 G.gamma_tilde[ZZ].ptr()[id] = chi * g_zz;
794
795 G.gamma_tilde_inv[XX].ptr()[id] = inv_chi * ginv_xx;
796 G.gamma_tilde_inv[XY].ptr()[id] = inv_chi * ginv_xy;
797 G.gamma_tilde_inv[XZ].ptr()[id] = inv_chi * ginv_xz;
798 G.gamma_tilde_inv[YY].ptr()[id] = inv_chi * ginv_yy;
799 G.gamma_tilde_inv[YZ].ptr()[id] = inv_chi * ginv_yz;
800 G.gamma_tilde_inv[ZZ].ptr()[id] = inv_chi * ginv_zz;
801
802 G.K.ptr()[id] = zero;
803
804 for (int c = 0; c < 6; ++c)
805 G.A_tilde[c].ptr()[id] = zero;
806 }
807
808 bssn::apply_halos_grid<bssn::BoundaryRadiative>(G);
809 tensorium_RG::bssn::compute_tildeGamma_contracted(G);
810
811#pragma omp parallel for collapse(3)
812 for (size_t i = I0 + 3; i < I1 - 3; ++i)
813 for (size_t j = J0 + 3; j < J1 - 3; ++j)
814 for (size_t k = K0 + 3; k < K1 - 3; ++k) {
815
816 const size_t id = G.alpha.idx(i, j, k);
817
818 double Dbeta[3][3];
819 tensorium_RG::fd::covariant_D_vec(G, G.beta, i, j, k, Dbeta,
820 tensorium_RG::fd::Variance::Up,
821 tensorium_RG::fd::Variance::Down);
822
823 const T alpha = G.alpha.ptr()[id];
824 const T inv2a = T(0.5) / alpha;
825
826 const T K_xx = inv2a * (Dbeta[0][0] + Dbeta[0][0]);
827 const T K_xy = inv2a * (Dbeta[0][1] + Dbeta[1][0]);
828 const T K_xz = inv2a * (Dbeta[0][2] + Dbeta[2][0]);
829 const T K_yy = inv2a * (Dbeta[1][1] + Dbeta[1][1]);
830 const T K_yz = inv2a * (Dbeta[1][2] + Dbeta[2][1]);
831 const T K_zz = inv2a * (Dbeta[2][2] + Dbeta[2][2]);
832
833 const T chi = G.chi.ptr()[id];
834 const T inv_chi = one / chi;
835
836 const T g_xx = inv_chi * G.gamma_tilde[XX].ptr()[id];
837 const T g_xy = inv_chi * G.gamma_tilde[XY].ptr()[id];
838 const T g_xz = inv_chi * G.gamma_tilde[XZ].ptr()[id];
839 const T g_yy = inv_chi * G.gamma_tilde[YY].ptr()[id];
840 const T g_yz = inv_chi * G.gamma_tilde[YZ].ptr()[id];
841 const T g_zz = inv_chi * G.gamma_tilde[ZZ].ptr()[id];
842
843 T ginv_xx, ginv_xy, ginv_xz, ginv_yy, ginv_yz, ginv_zz, detg;
844 invert_sym_3x3(g_xx, g_xy, g_xz, g_yy, g_yz, g_zz, ginv_xx, ginv_xy, ginv_xz,
845 ginv_yy, ginv_yz, ginv_zz, detg);
846
847 const T Ktr = ginv_xx * K_xx + T(2) * ginv_xy * K_xy + T(2) * ginv_xz * K_xz +
848 ginv_yy * K_yy + T(2) * ginv_yz * K_yz + ginv_zz * K_zz;
849
850 G.K.ptr()[id] = Ktr;
851
852 const T one_third = T(1) / T(3);
853
854 const T A_xx = K_xx - one_third * g_xx * Ktr;
855 const T A_xy = K_xy - one_third * g_xy * Ktr;
856 const T A_xz = K_xz - one_third * g_xz * Ktr;
857 const T A_yy = K_yy - one_third * g_yy * Ktr;
858 const T A_yz = K_yz - one_third * g_yz * Ktr;
859 const T A_zz = K_zz - one_third * g_zz * Ktr;
860
861 G.A_tilde[XX].ptr()[id] = chi * A_xx;
862 G.A_tilde[XY].ptr()[id] = chi * A_xy;
863 G.A_tilde[XZ].ptr()[id] = chi * A_xz;
864 G.A_tilde[YY].ptr()[id] = chi * A_yy;
865 G.A_tilde[YZ].ptr()[id] = chi * A_yz;
866 G.A_tilde[ZZ].ptr()[id] = chi * A_zz;
867 }
868
869 bssn::apply_halos_grid<bssn::BoundaryRadiative>(G);
870 tensorium_RG::bssn::compute_ricci_bssn(G, G.Ricci);
871
872 double maxAbsH = 0.0;
873 double maxAbsM = 0.0;
874 double maxAbsC = 0.0;
875
876 double rMin = 1e300;
877 double rMax = 0.0;
878 size_t nSamples = 0;
879
880 const double r_min_cut = 2.0;
881 const double r_max_cut =
882 0.45 * std::min({(double)(I1 - I0 - 1) * (double)G.dx, (double)(J1 - J0 - 1) * (double)G.dy,
883 (double)(K1 - K0 - 1) * (double)G.dz});
884
885 for (size_t i = I0 + 4; i < I1 - 4; ++i)
886 for (size_t j = J0 + 4; j < J1 - 4; ++j)
887 for (size_t k = K0 + 4; k < K1 - 4; ++k) {
888
889 double x, y, z;
890 G.coords(i, j, k, x, y, z);
891 x -= (double)xc;
892 y -= (double)yc;
893 z -= (double)zc;
894
895 const double r = std::sqrt(x * x + y * y + z * z);
896 if (r <= r_min_cut || r >= r_max_cut)
897 continue;
898
899 rMin = std::min(rMin, r);
900 rMax = std::max(rMax, r);
901 nSamples++;
902 }
903
904 if (nSamples == 0) {
905 rMin = r_min_cut;
906 rMax = r_max_cut;
907 }
908
909 auto constraints = make_constraint_scratch(G);
910 tensorium_RG::bssn::compute_bssn_constraints(G, G.Ricci, constraints.H, constraints.M,
911 constraints.C, rMin, rMax, (double)xc, (double)yc,
912 (double)zc);
913
914 tensorium_RG::bssn::print_constraint_norms(G, constraints.H, constraints.M, constraints.C, rMin,
915 rMax, (double)xc, (double)yc, (double)zc);
916
917 print_ricci_samples(G);
918 tensorium_RG::bssn::assert_invariants(G, "init.kerr_schild_single");
919 tensorium_RG::bssn::project_bssn_state(G);
920}
