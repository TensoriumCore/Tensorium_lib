84 inline void BSSN(const tensorium::Vector<T> &X, T &alpha, tensorium::Vector<T> &beta,
85 tensorium::Tensor<T, 2> &gamma) const {
86 assert(X.size() == 4);
87 tensorium::Vector<T> metric_X = convert_to_metric_coordinates(X);
88 tensorium::Tensor<T, 2> g({4, 4});
89 (*this)(metric_X, g);
90
91 auto g_inv = invert_metric_tensor(g);
92
93 gamma.resize(3, 3);
94 for (size_t i = 0; i < 3; ++i)
95 for (size_t j = 0; j < 3; ++j)
96 gamma(i, j) = g(i + 1, j + 1);
97
98 tensorium::Tensor<T, 2> gamma_inv = invert_spatial_metric(gamma);
99
100 tensorium::Vector<T> beta_cov(3);
101 for (size_t i = 0; i < 3; ++i)
102 beta_cov(i) = g(0, i + 1);
103
104 beta.resize(3);
105 for (size_t i = 0; i < 3; ++i) {
106 T sum = 0.0;
107 for (size_t j = 0; j < 3; ++j)
108 sum += gamma_inv(i, j) * beta_cov(j);
109 beta(i) = sum;
110 }
111
112 const T g00_inv = g_inv(0, 0);
113 if (g00_inv >= 0)
114 throw std::runtime_error("Metric::BSSN: g^{00} must be negative for a valid ADM split");
115 alpha = std::sqrt(-1.0 / g00_inv);
116
117 validate_adm_split(g, gamma, beta, alpha);
118 }

120 T compute_conformal_factor(const tensorium::Tensor<T, 2> &gamma) const {
121 assert(gamma.dimensions[0] == 3 && gamma.dimensions[1] == 3);
122 T det = gamma(0, 0) * (gamma(1, 1) * gamma(2, 2) - gamma(1, 2) * gamma(2, 1)) -
123 gamma(0, 1) * (gamma(1, 0) * gamma(2, 2) - gamma(1, 2) * gamma(2, 0)) +
124 gamma(0, 2) * (gamma(1, 0) * gamma(2, 1) - gamma(1, 1) * gamma(2, 0));
125 return std::pow(det, -1.0 / 3.0);
126 }

164 void compute_schwarzschild(const tensorium::Vector<T> &X, tensorium::Tensor<T, 2> &g) const {
165 assert(X.size() == 4);
166 const T r = X(1);
167 const T theta = X(2);
168 const T sin_theta = std::sin(theta);
169 const T f = T(1) - T(2) * M / r;
170
171 g.resize(4, 4);
172 g.fill(T(0));
173 g(0, 0) = -f;
174 g(1, 1) = T(1) / f;
175 g(2, 2) = r * r;
176 g(3, 3) = r * r * sin_theta * sin_theta;
177 }

180 void compute_kerr(const tensorium::Vector<T> &X, tensorium::Tensor<T, 2> &g) const {
181 assert(X.size() == 4);
182 const T r = X(1);
183 const T theta = X(2);
184 const T sin_theta = std::sin(theta);
185 const T cos_theta = std::cos(theta);
186 const T sin2 = sin_theta * sin_theta;
187
188 const T Sigma = r * r + a * a * cos_theta * cos_theta;
189 const T Delta = r * r - T(2) * M * r + a * a;
190
191 g.resize({4, 4});
192 g.fill(T(0));
193
194 g(0, 0) = -(T(1) - (T(2) * M * r) / Sigma);
195 g(0, 3) = g(3, 0) = -T(2) * M * r * a * sin2 / Sigma;
196 g(1, 1) = Sigma / Delta;
197 g(2, 2) = Sigma;
198 g(3, 3) = (r * r + a * a + T(2) * M * r * a * a * sin2 / Sigma) * sin2;
199 }

202 void compute_flrw(const tensorium::Vector<T> &X, tensorium::Tensor<T, 2> &g) const {
203 assert(X.size() == 4);

205 const T r = X(1);
206 const T theta = X(2);
207 const T sin_theta = std::sin(theta);

217 g(3, 3) = a_t * a_t * r * r * sin_theta * sin_theta;

221 T kerr_schild_radius(T x, T y, T z) const {
222 const T r2 = x * x + y * y + z * z;
223 const T a2 = a * a;
224 const T term = std::sqrt((r2 - a2) * (r2 - a2) + 4 * a2 * z * z);
225 const T r = std::sqrt(0.5 * (r2 - a2 + term));
226 return r;
227 }

230 void compute_kerr_schild(const tensorium::Vector<T> &X, tensorium::Tensor<T, 2> &g) const {
231 assert(X.size() == 4);
232 const T x = X(1), y = X(2), z = X(3);
233
234 const T r = kerr_schild_radius(x, y, z);
235 const T cos_theta = (r > 1e-14) ? z / r : 0.0;
236 const T denom = r * r + a * a * cos_theta * cos_theta;
237 const T H = (denom > 1e-14) ? (M * r) / denom : 0.0;
238
239 const T denom_vec = r * r + a * a;
240 T lx = (denom_vec > 1e-14) ? (r * x + a * y) / denom_vec : 0.0;
241 T ly = (denom_vec > 1e-14) ? (r * y - a * x) / denom_vec : 0.0;
242 T lz = (r > 1e-14) ? z / r : 0.0;
243
244 T norm_l = std::sqrt(lx * lx + ly * ly + lz * lz);
245 if (norm_l > 1e-14) {
246 lx /= norm_l;
247 ly /= norm_l;
248 lz /= norm_l;
249 }
250
251 g.resize({4, 4});
252 g.fill(0);
253
254 g(0, 0) = -1.0;
255 g(1, 1) = 1.0;
256 g(2, 2) = 1.0;
257 g(3, 3) = 1.0;
258
259 const std::array<T, 4> l = {1.0, lx, ly, lz};
260
261 for (size_t mu = 0; mu < 4; ++mu)
262 for (size_t nu = 0; nu < 4; ++nu)
263 g(mu, nu) += 2.0 * H * l[mu] * l[nu];
264 }

267 if (X.size() != 4)

269
270 if (type == "schwarzschild" || type == "kerr" || type == "flrw") {

272 Xs(0) = X(0);

274 const T y = X(2);
275 const T z = X(3);
276 const T r = std::sqrt(x * x + y * y + z * z);
277 const T theta = (r > T(0)) ? std::acos(std::clamp(z / r, T(-1), T(1))) : T(0);

290 tensorium::Tensor<T, 2> invert_metric_tensor(const tensorium::Tensor<T, 2> &g) const {
291 const size_t d0 = g.dimensions[0];
292 const size_t d1 = g.dimensions[1];
293 tensorium::Matrix<T> mat({d0, d1});
294
295 for (size_t i = 0; i < d0; ++i)
296 for (size_t j = 0; j < d1; ++j)
297 mat(i, j) = g(i, j);
298
299 tensorium::Matrix<T> inv = mat.inverse();
300
301 tensorium::Tensor<T, 2> out({d0, d1});
302 for (size_t i = 0; i < d0; ++i)
303 for (size_t j = 0; j < d1; ++j)
304 out(i, j) = inv(i, j);
305
306 return out;
307 }

323 void validate_adm_split(const tensorium::Tensor<T, 2> &g, const tensorium::Tensor<T, 2> &gamma,
324 const tensorium::Vector<T> &beta, T alpha) const {
325 tensorium::Tensor<T, 2> g_recon({4, 4});
326 g_recon.fill(T(0));
327
328 T g00 = -alpha * alpha;
329 for (size_t i = 0; i < 3; ++i)
330 for (size_t j = 0; j < 3; ++j)
331 g00 += gamma(i, j) * beta(i) * beta(j);
332 g_recon(0, 0) = g00;
333
334 for (size_t i = 0; i < 3; ++i) {
335 T sum = 0.0;
336 for (size_t j = 0; j < 3; ++j)
337 sum += gamma(i, j) * beta(j);
338 g_recon(0, i + 1) = sum;
339 g_recon(i + 1, 0) = sum;
340 for (size_t j = 0; j < 3; ++j)
341 g_recon(i + 1, j + 1) = gamma(i, j);
342 }
343
344 const T tol = T(1e-6);
345 for (size_t mu = 0; mu < 4; ++mu)
346 for (size_t nu = 0; nu < 4; ++nu) {
347 T diff = std::abs(g(mu, nu) - g_recon(mu, nu));
348 T scale = std::max(T(1), std::abs(g(mu, nu)));
349 if (diff > tol * scale)
350 throw std::runtime_error("Metric::BSSN: ADM split validation failed");
351 }
352 }
