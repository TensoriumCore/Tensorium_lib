94inline double minkowski_target(BoundaryField which, int component) {
95 switch (which) {
96 case BoundaryField::Alpha:
97 case BoundaryField::Chi:
98 return 1.0;
99 case BoundaryField::Beta:
100 case BoundaryField::B:
101 case BoundaryField::TildeGamma:
102 case BoundaryField::ATilde:
103 case BoundaryField::Theta:
104 case BoundaryField::Z:
105 return 0.0;
106 case BoundaryField::GammaTilde:
107 if (component == tensorium_RG::XX || component == tensorium_RG::YY ||
108 component == tensorium_RG::ZZ)
109 return 1.0;
110 return 0.0;
111 default:
112 return 0.0;
113 }
114}

138struct BoundaryRadiative {
139 inline static double characteristic_speed = 1.0;
140 inline static double characteristic_dt = 0.0;
141
142 static inline void set_characteristic(double speed, double dt) {
143 characteristic_speed = std::max(speed, 1.0e-6);
144 characteristic_dt = std::max(dt, 0.0);
145 }
146
147 template <typename T>
148 static inline void apply_physical(Field3D<T> &, const BSSNGridSoA<T> &, BoundaryField, int) {}
149
150 template <typename T>
151 static inline void apply_halo(Field3D<T> &field, const BSSNGridSoA<T> &G, BoundaryField which,
152 int component) {
153 const auto &D = G.dims;
154 const double u_inf = detail::minkowski_target(which, component);
155
156 const size_t I0 = D.ng;
157 const size_t I1 = D.ng + D.nx;
158 const size_t J0 = D.ng;
159 const size_t J1 = D.ng + D.ny;
160 const size_t K0 = D.ng;
161 const size_t K1 = D.ng + D.nz;
162
163 auto *ptr = field.ptr();
164
165 auto coord = [&](size_t i, size_t j, size_t k, double &x, double &y, double &z) {
166 x = G.x0 + (double(i) - double(D.ng)) * G.dx;
167 y = G.y0 + (double(j) - double(D.ng)) * G.dy;
168 z = G.z0 + (double(k) - double(D.ng)) * G.dz;
169 };
170
171 auto r_of = [&](size_t i, size_t j, size_t k) {
172 double x, y, z;
173 coord(i, j, k, x, y, z);
174 double r = std::sqrt(x * x + y * y + z * z);
175 return (r > 1e-14) ? r : 1e-14;
176 };
177
178 auto set_sommerfeld = [&](size_t ob, size_t ib, double r_ob, double r_ib) {
179 double u_ib = double(ptr[ib]);
180 double du = u_ib - u_inf;
181 const double dt_wave = characteristic_dt;
182 const double c_wave = characteristic_speed;
183 double denom = r_ob;
184 if (dt_wave > 0.0)
185 denom += c_wave * dt_wave;
186 denom = std::max(denom, 1.0e-12);
187 ptr[ob] = T(u_inf + du * (r_ib / denom));
188 };
189
190 for (size_t g = 1; g <= D.ng; ++g)
191 for (size_t j = J0; j < J1; ++j)
192 for (size_t k = K0; k < K1; ++k)
193 set_sommerfeld(field.idx(I0 - g, j, k), field.idx(I0, j, k), r_of(I0 - g, j, k),
194 r_of(I0, j, k));
195
196 for (size_t g = 0; g < D.ng; ++g)
197 for (size_t j = J0; j < J1; ++j)
198 for (size_t k = K0; k < K1; ++k)
199 set_sommerfeld(field.idx(I1 + g, j, k), field.idx(I1 - 1, j, k),
200 r_of(I1 + g, j, k), r_of(I1 - 1, j, k));
201
202 for (size_t g = 1; g <= D.ng; ++g)
203 for (size_t i = I0 - D.ng; i < I1 + D.ng; ++i)
204 for (size_t k = K0; k < K1; ++k)
205 set_sommerfeld(field.idx(i, J0 - g, k), field.idx(i, J0, k), r_of(i, J0 - g, k),
206 r_of(i, J0, k));
207
208 for (size_t g = 0; g < D.ng; ++g)
209 for (size_t i = I0 - D.ng; i < I1 + D.ng; ++i)
210 for (size_t k = K0; k < K1; ++k)
211 set_sommerfeld(field.idx(i, J1 + g, k), field.idx(i, J1 - 1, k),
212 r_of(i, J1 + g, k), r_of(i, J1 - 1, k));
213
214 for (size_t g = 1; g <= D.ng; ++g)
215 for (size_t i = I0 - D.ng; i < I1 + D.ng; ++i)
216 for (size_t j = J0 - D.ng; j < J1 + D.ng; ++j)
217 set_sommerfeld(field.idx(i, j, K0 - g), field.idx(i, j, K0), r_of(i, j, K0 - g),
218 r_of(i, j, K0));
219
220 for (size_t g = 0; g < D.ng; ++g)
221 for (size_t i = I0 - D.ng; i < I1 + D.ng; ++i)
222 for (size_t j = J0 - D.ng; j < J1 + D.ng; ++j)
223 set_sommerfeld(field.idx(i, j, K1 + g), field.idx(i, j, K1 - 1),
224 r_of(i, j, K1 + g), r_of(i, j, K1 - 1));
225 }
226};
