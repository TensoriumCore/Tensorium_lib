217 __attribute__((always_inline, hot, flatten)) static inline Vector<K>
218 linear_combination(const std::vector<Vector<K>> &u, const std::vector<K> &coefs) {
219 using Simd = simd::SimdTraits<K, DefaultISA>;
220 using reg = typename Simd::reg;
221 const size_t simd_width = Simd::width;
222
223 if (u.size() != coefs.size())
224 throw std::invalid_argument("Mismatched number of vectors and coefficients");
225 if (u.empty())
226 return Vector<K>(0);
227
228 const size_t n = u[0].size();
229 for (const auto &v : u)
230 if (v.size() != n)
231 throw std::invalid_argument("Vector sizes do not match");
232
233 Vector<K> result(n);
234
235 size_t i = 0;
236 constexpr size_t W = simd_width;
237 for (; i + W - 1 < n; i += W) {
238 reg acc = Simd::zero();
239 for (size_t j = 0; j < u.size(); ++j) {
240 reg v = Simd::load(&u[j].data[i]);
241 reg c = Simd::set1(coefs[j]);
242 acc = Simd::fmadd(v, c, acc);
243 }
244 Simd::store(&result.data[i], acc);
245 }
246
247 for (; i < n; ++i) {
248 K acc = K(0);
249 for (size_t j = 0; j < u.size(); ++j)
250 acc += coefs[j] * u[j].data[i];
251 result.data[i] = acc;
252 }
253
254 return result;
255 }

265 lerp(const Vector<K> &a, const Vector<K> &b, K t) {
266 if (a.size() != b.size())

269 using reg = typename Simd::reg;

271

274

276 const reg vt1 = Simd::sub(Simd::set1(K(1)), vt);
277
278 size_t i = 0;

280 for (; i + 7 < n; i += simd_width) {
281 reg va = Simd::load(&a.data[i]);
282 reg vb = Simd::load(&b.data[i]);
283
284 reg r = Simd::fmadd(vb, vt, Simd::mul(va, vt1));
285
286 Simd::store_stream(&result.data[i], r);
287 }
288
289 for (; i < n; ++i)
290 result.data[i] = (K(1) - t) * a.data[i] + t * b.data[i];

437 cross_product(const Vector<K> &u, const Vector<K> &v) {
438 if (u.size() != 3 || v.size() != 3)
439 throw std::invalid_argument("Cross product is only defined for 3D vectors.");
440
441 Vector<K> r(3);
442 if constexpr (std::is_floating_point_v<K>) {
443 r.data[0] = std::fma(u.data[1], v.data[2], -u.data[2] * v.data[1]);
444 r.data[1] = std::fma(u.data[2], v.data[0], -u.data[0] * v.data[2]);
445 r.data[2] = std::fma(u.data[0], v.data[1], -u.data[1] * v.data[0]);
446 } else {
447 r.data[0] = u.data[1] * v.data[2] - u.data[2] * v.data[1];
448 r.data[1] = u.data[2] * v.data[0] - u.data[0] * v.data[2];
449 r.data[2] = u.data[0] * v.data[1] - u.data[1] * v.data[0];
