105inline T compute_dt_cfl(const BSSNGridSoA<T> &grid, T cfl_factor, size_t padding = 4) {
106 const T min_dx = std::min({grid.dx, grid.dy, grid.dz});
107 size_t I0, I1, J0, J1, K0, K1;
108 grid.domain_bounds(I0, I1, J0, J1, K0, K1);
109 const size_t guard = std::max<size_t>(padding, size_t(2));
110 const size_t i_begin = std::min(I0 + guard, I1);
111 const size_t j_begin = std::min(J0 + guard, J1);
112 const size_t k_begin = std::min(K0 + guard, K1);
113 const size_t i_end = (I1 > guard) ? I1 - guard : I1;
114 const size_t j_end = (J1 > guard) ? J1 - guard : J1;
115 const size_t k_end = (K1 > guard) ? K1 - guard : K1;
116
117 T max_speed = T(0);
118 for (size_t i = i_begin; i < i_end; ++i)
119 for (size_t j = j_begin; j < j_end; ++j)
120 for (size_t k = k_begin; k < k_end; ++k) {
121 const size_t id = grid.alpha.idx(i, j, k);
122 const T alpha = grid.alpha.ptr()[id];
123 const T beta_x = grid.beta[0].ptr()[id];
124 const T beta_y = grid.beta[1].ptr()[id];
125 const T beta_z = grid.beta[2].ptr()[id];
126 const T beta_mag = std::sqrt(beta_x * beta_x + beta_y * beta_y + beta_z * beta_z);
127 max_speed = std::max(max_speed, beta_mag + alpha);
128 }
129 if (max_speed <= T(0))
130 max_speed = T(1);
131 return cfl_factor * (min_dx / max_speed);
132}

172 void step(BSSNGridSoA<T> &grid, T dt, size_t step_index = 0) {
173 // Stage 1
174 apply_halos_grid<Boundary>(grid);
175 ensure_geometry(grid);
176 apply_halos_grid<Boundary>(grid);
177 evaluate_rhs(grid, stages_[0]);
178
179 // Stage 2
180 build_stage_state(grid, stages_[0], dt * T(0.5));
181 apply_halos_grid<Boundary>(stage_grid_);
182 ensure_geometry(stage_grid_);
183 apply_halos_grid<Boundary>(stage_grid_);
184 evaluate_rhs(stage_grid_, stages_[1]);
185
186 // Stage 3
187 build_stage_state(grid, stages_[1], dt * T(0.5));
188 apply_halos_grid<Boundary>(stage_grid_);
189 ensure_geometry(stage_grid_);
190 apply_halos_grid<Boundary>(stage_grid_);
191 evaluate_rhs(stage_grid_, stages_[2]);
192
193 // Stage 4
194 build_stage_state(grid, stages_[2], dt);
195 apply_halos_grid<Boundary>(stage_grid_);
196 ensure_geometry(stage_grid_);
197 apply_halos_grid<Boundary>(stage_grid_);
198 evaluate_rhs(stage_grid_, stages_[3]);
199
200 // Combine updates.
201 apply_rk_update(grid, dt);
202 ensure_geometry(grid);
203 apply_halos_grid<Boundary>(grid);
204 log_gauge_diagnostics(grid, step_index);
205
206 if (monitor_callback_ || snapshot_callback_) {
207 const double min_extent =
208 std::min({(grid.dims.nx - 1) * grid.dx, (grid.dims.ny - 1) * grid.dy,
209 (grid.dims.nz - 1) * grid.dz});
210 const double r_min = 2.0 * std::min({grid.dx, grid.dy, grid.dz});
211 const double r_max = 0.45 * min_extent;
212 compute_bssn_constraints(grid, grid.Ricci, grid.Hc, grid.Mc, grid.Cc, r_min, r_max, 0.0,
213 0.0, 0.0);
214 const auto stats = compute_constraint_monitor(grid, grid.Hc, padding_);
215 if (monitor_callback_)
216 monitor_callback_(grid, stats);
217 }
218 if (snapshot_callback_)
219 snapshot_callback_(grid, step_index);
220 }

260 void log_gauge_diagnostics(const BSSNGridSoA<T> &grid, size_t step_index) {
261 if (step_index > 10)
262 return;
263
264 const size_t total = grid.alpha.st.nx_tot * grid.alpha.st.ny_tot * grid.alpha.st.nz_tot;
265 double max_beta = 0.0;
266 double max_B = 0.0;
267 double min_alpha = std::numeric_limits<double>::infinity();
268 double min_chi = std::numeric_limits<double>::infinity();
269 double max_gamma = 0.0;
270 double max_det_drift = 0.0;
271 double max_trace_A = 0.0;
272
273 using tensorium_RG::XX;
274 using tensorium_RG::XY;
275 using tensorium_RG::XZ;
276 using tensorium_RG::YY;
277 using tensorium_RG::YZ;
278 using tensorium_RG::ZZ;
279
280 for (size_t idx = 0; idx < total; ++idx) {
281 const double alpha = double(grid.alpha.ptr()[idx]);
282 const double chi = double(grid.chi.ptr()[idx]);
283 const double bx = double(grid.beta[0].ptr()[idx]);
284 const double by = double(grid.beta[1].ptr()[idx]);
285 const double bz = double(grid.beta[2].ptr()[idx]);
286 const double Bx = double(grid.B[0].ptr()[idx]);
287 const double By = double(grid.B[1].ptr()[idx]);
288 const double Bz = double(grid.B[2].ptr()[idx]);
289 const double gamma0 = double(grid.tildeGamma[0].ptr()[idx]);
290 const double gamma1 = double(grid.tildeGamma[1].ptr()[idx]);
291 const double gamma2 = double(grid.tildeGamma[2].ptr()[idx]);
292
293 max_beta = std::max(max_beta, std::sqrt(bx * bx + by * by + bz * bz));
294 max_B = std::max(max_B, std::sqrt(Bx * Bx + By * By + Bz * Bz));
295 max_gamma = std::max({max_gamma, std::abs(gamma0), std::abs(gamma1), std::abs(gamma2)});
296 min_alpha = std::min(min_alpha, alpha);
297 min_chi = std::min(min_chi, chi);
298
299 double gxx = double(grid.gamma_tilde[XX].ptr()[idx]);
300 double gxy = double(grid.gamma_tilde[XY].ptr()[idx]);
301 double gxz = double(grid.gamma_tilde[XZ].ptr()[idx]);
302 double gyy = double(grid.gamma_tilde[YY].ptr()[idx]);
303 double gyz = double(grid.gamma_tilde[YZ].ptr()[idx]);
304 double gzz = double(grid.gamma_tilde[ZZ].ptr()[idx]);
305
306 const double det = gxx * (gyy * gzz - gyz * gyz) -
307 gxy * (gxy * gzz - gxz * gyz) +
308 gxz * (gxy * gyz - gxz * gyy);
309 max_det_drift = std::max(max_det_drift, std::abs(det - 1.0));
310
311 const double gixx = double(grid.gamma_tilde_inv[XX].ptr()[idx]);
312 const double gixy = double(grid.gamma_tilde_inv[XY].ptr()[idx]);
313 const double gixz = double(grid.gamma_tilde_inv[XZ].ptr()[idx]);
314 const double giyy = double(grid.gamma_tilde_inv[YY].ptr()[idx]);
315 const double giyz = double(grid.gamma_tilde_inv[YZ].ptr()[idx]);
316 const double gizz = double(grid.gamma_tilde_inv[ZZ].ptr()[idx]);
317 const double Axx = double(grid.A_tilde[XX].ptr()[idx]);
318 const double Axy = double(grid.A_tilde[XY].ptr()[idx]);
319 const double Axz = double(grid.A_tilde[XZ].ptr()[idx]);
320 const double Ayy = double(grid.A_tilde[YY].ptr()[idx]);
321 const double Ayz = double(grid.A_tilde[YZ].ptr()[idx]);
322 const double Azz = double(grid.A_tilde[ZZ].ptr()[idx]);
323
324 const double traceA = gixx * Axx + giyy * Ayy + gizz * Azz +
325 2.0 * (gixy * Axy + gixz * Axz + giyz * Ayz);
326 max_trace_A = std::max(max_trace_A, std::abs(traceA));
327 }
328
329 printf("[GaugeDiag step=%zu] max|beta|=%.3e max|B|=%.3e min(alpha)=%.3e min(chi)=%.3e "
330 "max|Gamma|=%.3e max_det_drift=%.3e max_traceA=%.3e\n",
331 step_index, max_beta, max_B, min_alpha, min_chi, max_gamma, max_det_drift,
332 max_trace_A);
333 }

371 const T c1 = dt / T(6);

383 stages_[3].beta[i], c1, c2, c3, c4);
