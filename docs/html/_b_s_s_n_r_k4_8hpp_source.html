120template <typename T>
121inline T compute_dt_cfl(const BSSNGridSoA<T> &grid, const CFLControl<T> &control,
122 size_t padding = 4) {
123 const T min_dx = std::min({grid.dx, grid.dy, grid.dz});
124
125 size_t I0, I1, J0, J1, K0, K1;
126 grid.domain_bounds(I0, I1, J0, J1, K0, K1);
127
128 const size_t guard = std::max<size_t>(padding, size_t(2));
129 const size_t i_begin = std::min(I0 + guard, I1);
130 const size_t j_begin = std::min(J0 + guard, J1);
131 const size_t k_begin = std::min(K0 + guard, K1);
132 const size_t i_end = (I1 > guard) ? I1 - guard : I1;
133 const size_t j_end = (J1 > guard) ? J1 - guard : J1;
134 const size_t k_end = (K1 > guard) ? K1 - guard : K1;
135
136 T max_beta = T(0);
137 T min_alpha = std::numeric_limits<T>::infinity();
138 int gradient_flag = 0;
139#pragma omp parallel for collapse(3) reduction(max : max_beta) reduction(min : min_alpha) \
140 reduction(max : gradient_flag)
141 for (size_t i = i_begin; i < i_end; ++i)
142 for (size_t j = j_begin; j < j_end; ++j)
143 for (size_t k = k_begin; k < k_end; ++k) {
144 const size_t id = grid.alpha.idx(i, j, k);
145 const T alpha = std::abs(grid.alpha.ptr()[id]);
146 const T bx = grid.beta[0].ptr()[id];
147 const T by = grid.beta[1].ptr()[id];
148 const T bz = grid.beta[2].ptr()[id];
149 const T theta_val = grid.Theta.ptr()[id];
150
151 const T beta_mag = std::sqrt(bx * bx + by * by + bz * bz);
152
153 max_beta = std::max(max_beta, beta_mag);
154 min_alpha = std::min(min_alpha, alpha);
155 // Gauge-shock sensors: strong shifts, collapsing lapse, or large Theta.
156 if (alpha < T(0.1) || beta_mag > T(0.8) || std::abs(theta_val) > T(1.0))
157 gradient_flag = 1;
158 }
159
160 const T alpha_floor = T(0.1);
161 const T a = std::max(min_alpha, alpha_floor);
162
163 const T gauge_term = control.gauge_speed * std::sqrt(T(1) / a);
164
165 T denom = max_beta + gauge_term;
166 if (denom <= T(0))
167 denom = T(1);
168
169 T cfl_number = control.cfl;
170 if (gradient_flag != 0)
171 cfl_number = std::min(cfl_number, T(0.12));
172
173 return cfl_number * (min_dx / denom);
174}

213 void log_full_bssn_diagnostics(const BSSNGridSoA<T> &grid, size_t step, double r_min,
214 double r_max, double chi_cut = 0.0) {
215 size_t I0, I1, J0, J1, K0, K1;
216 grid.domain_bounds(I0, I1, J0, J1, K0, K1);
217 const size_t guard = std::max(padding_, size_t(2));
218 const size_t i0 = std::min(I0 + guard, I1);
219 const size_t j0 = std::min(J0 + guard, J1);
220 const size_t k0 = std::min(K0 + guard, K1);
221 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
222 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
223 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
224
225 double max_beta = 0.0, max_B = 0.0;
226 double min_alpha = 1e100, min_chi = 1e100;
227 double max_gamma = 0.0, max_det = 0.0, max_trA = 0.0;
228 double min_theta = std::numeric_limits<double>::infinity();
229 double max_theta = -std::numeric_limits<double>::infinity();
230 double max_K = 0.0, max_A = 0.0;
231 double max_Rc = 0.0, max_R3 = 0.0;
232 size_t samples = 0;
233
234 auto upd_max_abs = [](double &m, double v) {
235 const double a = std::abs(v);
236 if (std::isfinite(a))
237 m = std::max(m, a);
238 };
239 for (size_t i = i0; i < i1; ++i)
240 for (size_t j = j0; j < j1; ++j)
241 for (size_t k = k0; k < k1; ++k) {
242 const size_t idx = grid.alpha.idx(i, j, k);
243 const double chi = double(grid.chi.ptr()[idx]);
244 if (chi_cut > 0.0 && !(chi > chi_cut))
245 continue;
246
247 const double x = double(grid.x0) + double(i - grid.dims.ng) * double(grid.dx);
248 const double y = double(grid.y0) + double(j - grid.dims.ng) * double(grid.dy);
249 const double z = double(grid.z0) + double(k - grid.dims.ng) * double(grid.dz);
250 const double r = std::sqrt(x * x + y * y + z * z);
251 if (!(r > r_min && r < r_max))
252 continue;
253 ++samples;
254
255 const double alpha = double(grid.alpha.ptr()[idx]);
256 if (std::isfinite(alpha))
257 min_alpha = std::min(min_alpha, alpha);
258 if (std::isfinite(chi))
259 min_chi = std::min(min_chi, chi);
260 const double theta = double(grid.Theta.ptr()[idx]);
261 if (std::isfinite(theta)) {
262 min_theta = std::min(min_theta, theta);
263 max_theta = std::max(max_theta, theta);
264 }
265
266 const double bx = double(grid.beta[0].ptr()[idx]);
267 const double by = double(grid.beta[1].ptr()[idx]);
268 const double bz = double(grid.beta[2].ptr()[idx]);
269 upd_max_abs(max_beta, std::sqrt(bx * bx + by * by + bz * bz));
270
271 const double Bx = double(grid.B[0].ptr()[idx]);
272 const double By = double(grid.B[1].ptr()[idx]);
273 const double Bz = double(grid.B[2].ptr()[idx]);
274 upd_max_abs(max_B, std::sqrt(Bx * Bx + By * By + Bz * Bz));
275
276 upd_max_abs(max_gamma, double(grid.tildeGamma[0].ptr()[idx]));
277 upd_max_abs(max_gamma, double(grid.tildeGamma[1].ptr()[idx]));
278 upd_max_abs(max_gamma, double(grid.tildeGamma[2].ptr()[idx]));
279
280 const double gxx = double(grid.gamma_tilde[XX].ptr()[idx]);
281 const double gxy = double(grid.gamma_tilde[XY].ptr()[idx]);
282 const double gxz = double(grid.gamma_tilde[XZ].ptr()[idx]);
283 const double gyy = double(grid.gamma_tilde[YY].ptr()[idx]);
284 const double gyz = double(grid.gamma_tilde[YZ].ptr()[idx]);
285 const double gzz = double(grid.gamma_tilde[ZZ].ptr()[idx]);
286
287 const double det = gxx * (gyy * gzz - gyz * gyz) -
288 gxy * (gxy * gzz - gxz * gyz) +
289 gxz * (gxy * gyz - gxz * gyy);
290 upd_max_abs(max_det, det - 1.0);
291
292 const double gixx = double(grid.gamma_tilde_inv[XX].ptr()[idx]);
293 const double gixy = double(grid.gamma_tilde_inv[XY].ptr()[idx]);
294 const double gixz = double(grid.gamma_tilde_inv[XZ].ptr()[idx]);
295 const double giyy = double(grid.gamma_tilde_inv[YY].ptr()[idx]);
296 const double giyz = double(grid.gamma_tilde_inv[YZ].ptr()[idx]);
297 const double gizz = double(grid.gamma_tilde_inv[ZZ].ptr()[idx]);
298
299 const double Axx = double(grid.A_tilde[XX].ptr()[idx]);
300 const double Axy = double(grid.A_tilde[XY].ptr()[idx]);
301 const double Axz = double(grid.A_tilde[XZ].ptr()[idx]);
302 const double Ayy = double(grid.A_tilde[YY].ptr()[idx]);
303 const double Ayz = double(grid.A_tilde[YZ].ptr()[idx]);
304 const double Azz = double(grid.A_tilde[ZZ].ptr()[idx]);
305
306 const double trA = gixx * Axx + giyy * Ayy + gizz * Azz +
307 2.0 * (gixy * Axy + gixz * Axz + giyz * Ayz);
308 upd_max_abs(max_trA, trA);
309 upd_max_abs(max_K, double(grid.K.ptr()[idx]));
310
311 upd_max_abs(max_A, Axx);
312 upd_max_abs(max_A, Axy);
313 upd_max_abs(max_A, Axz);
314 upd_max_abs(max_A, Ayy);
315 upd_max_abs(max_A, Ayz);
316 upd_max_abs(max_A, Azz);
317
318 const double Rxx = double(grid.Ricci[XX].ptr()[idx]);
319 const double Rxy = double(grid.Ricci[XY].ptr()[idx]);
320 const double Rxz = double(grid.Ricci[XZ].ptr()[idx]);
321 const double Ryy = double(grid.Ricci[YY].ptr()[idx]);
322 const double Ryz = double(grid.Ricci[YZ].ptr()[idx]);
323 const double Rzz = double(grid.Ricci[ZZ].ptr()[idx]);
324 upd_max_abs(max_Rc, Rxx);
325 upd_max_abs(max_Rc, Rxy);
326 upd_max_abs(max_Rc, Rxz);
327 upd_max_abs(max_Rc, Ryy);
328 upd_max_abs(max_Rc, Ryz);
329 upd_max_abs(max_Rc, Rzz);
330
331 const double R3 = gixx * Rxx + giyy * Ryy + gizz * Rzz +
332 2.0 * (gixy * Rxy + gixz * Rxz + giyz * Ryz);
333 upd_max_abs(max_R3, R3);
334 }
335
336 if (samples == 0) {
337 printf("[BSSN %04zu] empty-domain r in (%.3e, %.3e) chi_cut=%.3e\n", step, r_min, r_max,
338 chi_cut);
339 return;
340 }
341 if (min_alpha > 1e90)
342 min_alpha = 0.0;
343 if (min_chi > 1e90)
344 min_chi = 0.0;
345
346 const double theta_min_print = std::isfinite(min_theta) ? min_theta : 0.0;
347 const double theta_max_print = std::isfinite(max_theta) ? max_theta : 0.0;
348
349 printf("[BSSN %04zu]\n"
350 "\talpha_min=% .3e\tchi_min=% .3e\tTheta_min=% .3e\tTheta_max=% .3e\n"
351 "\tbeta_max=% .3e\tB_max=% .3e\tGamma_max=% .3e\n"
352 "\tdet_err=% .3e\ttrA=% .3e\tK_max=% .3e\tA_max=% .3e\n"
353 "\tRicci_cmax=% .3e\tR3_max=% .3e\tsamples=%zu\n",
354 step, min_alpha, min_chi, theta_min_print, theta_max_print, max_beta, max_B,
355 max_gamma, max_det, max_trA, max_K, max_A, max_Rc, max_R3, samples);
356 }

358 void step(BSSNGridSoA<T> &grid, T dt, size_t step_index = 0) {
359#if defined(TENSORIUM_BSSN_VALIDATE_TILDE_GAMMA_SYMBOLS)
360 tensorium_RG::bssn::detail::begin_contracted_warning_step(step_index);
361#endif
362 boundary_dt_ = dt;
363 prepare_state_for_rhs(grid);
364 evaluate_rhs(grid, stages_[0]);
365
366 build_stage_state(grid, stages_[0], dt * T(0.5));
367 prepare_state_for_rhs(stage_grid_);
368 evaluate_rhs(stage_grid_, stages_[1]);
369
370 build_stage_state(grid, stages_[1], dt * T(0.5));
371 prepare_state_for_rhs(stage_grid_);
372 evaluate_rhs(stage_grid_, stages_[2]);
373
374 build_stage_state(grid, stages_[2], dt);
375 prepare_state_for_rhs(stage_grid_);
376 evaluate_rhs(stage_grid_, stages_[3]);
377
378 apply_rk_update(grid, dt);
379 prepare_state_for_rhs(grid);
380
381 const double min_extent =
382 std::min({(grid.dims.nx - 1) * grid.dx, (grid.dims.ny - 1) * grid.dy,
383 (grid.dims.nz - 1) * grid.dz});
384 const double r_min = 2.0 * std::min({grid.dx, grid.dy, grid.dz});
385 const double r_max = 0.45 * min_extent;
386
387 log_gauge_diagnostics(grid, step_index);
388
389 const double chi_cut = 1e-7;
390 log_full_bssn_diagnostics(grid, step_index, r_min, r_max, chi_cut);
391
392 if (monitor_callback_ || snapshot_callback_) {
393 auto H_tmp = tensorium_RG::make_field(grid.alpha.st);
394 Field3D<T> M_tmp[3];
395 Field3D<T> C_tmp[3];
396 for (int q = 0; q < 3; ++q) {
397 M_tmp[q] = tensorium_RG::make_field(grid.alpha.st);
398 C_tmp[q] = tensorium_RG::make_field(grid.alpha.st);
399 }
400 compute_bssn_constraints(grid, grid.Ricci, H_tmp, M_tmp, C_tmp, r_min, r_max, 0.0, 0.0,
401 0.0);
402 auto stats = compute_constraint_monitor(grid, H_tmp, padding_);
403 populate_constraint_norms(grid, M_tmp, stats, padding_);
404 if (monitor_callback_)
405 monitor_callback_(grid, stats);
406 }
407
408 if (snapshot_callback_)
409 snapshot_callback_(grid, step_index);
410
411#if defined(TENSORIUM_BSSN_VALIDATE_TILDE_GAMMA_SYMBOLS)
412 tensorium_RG::bssn::detail::finalize_contracted_warning_step();
413#endif
414 report_kernel_timers(step_index);
415 }

427 void add_dissipation(const Field3D<T> &u, Field3D<T> &rhs, T sigma) {
428 const size_t nx = u.st.nx_tot;
429 const size_t ny = u.st.ny_tot;
430 const size_t nz = u.st.nz_tot;
431 const T *in = u.ptr();
432 T *out = rhs.ptr();
433 const T coef = sigma / 16.0;
434
435#pragma omp parallel for collapse(3)
436 for (size_t i = 2; i < nx - 2; ++i) {
437 for (size_t j = 2; j < ny - 2; ++j) {
438 for (size_t k = 2; k < nz - 2; ++k) {
439 size_t idx = i * u.st.sx + j * u.st.sy + k * u.st.sz;
440 T diss_x = in[idx - 2 * u.st.sx] - 4.0 * in[idx - u.st.sx] + 6.0 * in[idx] -
441 4.0 * in[idx + u.st.sx] + in[idx + 2 * u.st.sx];
442 T diss_y = in[idx - 2 * u.st.sy] - 4.0 * in[idx - u.st.sy] + 6.0 * in[idx] -
443 4.0 * in[idx + u.st.sy] + in[idx + 2 * u.st.sy];
444 T diss_z = in[idx - 2 * u.st.sz] - 4.0 * in[idx - u.st.sz] + 6.0 * in[idx] -
445 4.0 * in[idx + u.st.sz] + in[idx + 2 * u.st.sz];
446 out[idx] -= coef * (diss_x + diss_y + diss_z);
447 }
448 }
449 }

522 size_t I0, I1, J0, J1, K0, K1;
523 grid.domain_bounds(I0, I1, J0, J1, K0, K1);
524 const size_t guard = std::max(padding_, size_t(2));
525 const size_t i0 = std::min(I0 + guard, I1);
526 const size_t j0 = std::min(J0 + guard, J1);
527 const size_t k0 = std::min(K0 + guard, K1);
528 const size_t i1 = (I1 > guard) ? I1 - guard : I1;
529 const size_t j1 = (J1 > guard) ? J1 - guard : J1;
530 const size_t k1 = (K1 > guard) ? K1 - guard : K1;
531
532 double max_beta = 0.0, max_B = 0.0;
533 double min_alpha = 1e100, min_chi = 1e100;
534 double max_gamma = 0.0, max_det = 0.0, max_trA = 0.0;
535 double min_theta = std::numeric_limits<double>::infinity();
536 double max_theta = -std::numeric_limits<double>::infinity();
537
538 for (size_t i = i0; i < i1; ++i)
539 for (size_t j = j0; j < j1; ++j)
540 for (size_t k = k0; k < k1; ++k) {
541 const size_t idx = grid.alpha.idx(i, j, k);
542 const double alpha = grid.alpha.ptr()[idx];
543 const double chi = grid.chi.ptr()[idx];
544 const double bx = grid.beta[0].ptr()[idx];
545 const double by = grid.beta[1].ptr()[idx];
546 const double bz = grid.beta[2].ptr()[idx];
547 const double Bx = grid.B[0].ptr()[idx];
548 const double By = grid.B[1].ptr()[idx];
549 const double Bz = grid.B[2].ptr()[idx];
550
551 max_beta = std::max(max_beta, std::sqrt(bx * bx + by * by + bz * bz));
552 max_B = std::max(max_B, std::sqrt(Bx * Bx + By * By + Bz * Bz));
553 min_alpha = std::min(min_alpha, alpha);
554 min_chi = std::min(min_chi, chi);
555 const double theta = grid.Theta.ptr()[idx];
556 if (std::isfinite(theta)) {
557 min_theta = std::min(min_theta, theta);
558 max_theta = std::max(max_theta, theta);
559 }
560 max_gamma = std::max({max_gamma, std::abs(grid.tildeGamma[0].ptr()[idx]),
561 std::abs(grid.tildeGamma[1].ptr()[idx]),
562 std::abs(grid.tildeGamma[2].ptr()[idx])});
563
564 const double gxx = grid.gamma_tilde[XX].ptr()[idx];
565 const double gxy = grid.gamma_tilde[XY].ptr()[idx];
566 const double gxz = grid.gamma_tilde[XZ].ptr()[idx];
567 const double gyy = grid.gamma_tilde[YY].ptr()[idx];
568 const double gyz = grid.gamma_tilde[YZ].ptr()[idx];
569 const double gzz = grid.gamma_tilde[ZZ].ptr()[idx];
570 const double det = gxx * (gyy * gzz - gyz * gyz) -
571 gxy * (gxy * gzz - gxz * gyz) +
572 gxz * (gxy * gyz - gxz * gyy);
573 max_det = std::max(max_det, std::abs(det - 1.0));
574
575 const double trA =
576 grid.gamma_tilde_inv[XX].ptr()[idx] * grid.A_tilde[XX].ptr()[idx] +
577 grid.gamma_tilde_inv[YY].ptr()[idx] * grid.A_tilde[YY].ptr()[idx] +
578 grid.gamma_tilde_inv[ZZ].ptr()[idx] * grid.A_tilde[ZZ].ptr()[idx] +
579 2.0 * (grid.gamma_tilde_inv[XY].ptr()[idx] * grid.A_tilde[XY].ptr()[idx] +
580 grid.gamma_tilde_inv[XZ].ptr()[idx] * grid.A_tilde[XZ].ptr()[idx] +
581 grid.gamma_tilde_inv[YZ].ptr()[idx] * grid.A_tilde[YZ].ptr()[idx]);
582 max_trA = std::max(max_trA, std::abs(trA));
583 }
584
585 const double theta_min_print = std::isfinite(min_theta) ? min_theta : 0.0;
586 const double theta_max_print = std::isfinite(max_theta) ? max_theta : 0.0;
587 printf("[Gauge %02zu]\n"
588 "\talpha_min=% .3e\tchi_min=% .3e\tTheta_min=% .3e\tTheta_max=% .3e\n"
589 "\tbeta_max=% .3e\tB_max=% .3e\tGamma_max=% .3e\n"
590 "\tdet_err=% .3e\ttrA=% .3e\n",
591 step_index, min_alpha, min_chi, theta_min_print, theta_max_print, max_beta, max_B,
592 max_gamma, max_det, max_trA);
593 }

595 void blend_field_interior(const BSSNGridSoA<T> &grid, const Field3D<T> &base,
596 const Field3D<T> &delta, Field3D<T> &dst, T scale_dt) {
597 const size_t total = base.st.nx_tot * base.st.ny_tot * base.st.nz_tot;
598 std::copy(base.ptr(), base.ptr() + total, dst.ptr());
599 for_each_interior_index(grid, padding_, [&](size_t, size_t, size_t, size_t idx) {
600 dst.ptr()[idx] += scale_dt * delta.ptr()[idx];
601 });
602 }

607 stage_grid_.y0 = base.y0;

617 stage_grid_.tildeGamma[i], scale_dt);

621 blend_field_interior(base, base.gamma_tilde[s], delta.gamma_tilde[s],
622 stage_grid_.gamma_tilde[s], scale_dt);

625 }

629 const Field3D<T> &k2, const Field3D<T> &k3, const Field3D<T> &k4, T c1,

638 });
