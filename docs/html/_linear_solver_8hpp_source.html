214 int max_iter = 2000) {
215 static_assert(std::is_floating_point<K>::value,
216 "Jacobi solver requires floating-point type.");
217 assert(A.rows == A.cols && "Matrix A must be square");
218 assert(A.rows == b.size() && "Matrix/vector size mismatch");
219 const size_t n = A.rows;
220 Vector<K> x(n, K(0));
221 Vector<K> x_new(n, K(0));
222
223 for (int iter = 0; iter < max_iter; ++iter) {
224 Vector<K> Ax = A.mul_vec(x);
225#pragma omp parallel for schedule(dynamic, 4)
226 for (size_t i = 0; i < n; ++i) {
227 K diag = A(i, i);
228 if (std::abs(diag) < K(1e-10))
229 throw std::runtime_error("Jacobi: division by near-zero on diagonal, matrix likely not diagonally dominant.");
230 K sigma = Ax[i] - diag * x[i];
231 x_new[i] = (b[i] - sigma) / diag;
232 }
233
234 K err = K(0);
235#pragma omp parallel for reduction(+ : err) schedule(dynamic)
236 for (size_t i = 0; i < n; ++i) {
237 K diff = x_new[i] - x[i];
238 err += diff * diff;
239 }
240 err = std::sqrt(err);
241
242 if (err < tol)
243 break;
244
245 x = x_new;
246 }
247
248 return x;
249 }
