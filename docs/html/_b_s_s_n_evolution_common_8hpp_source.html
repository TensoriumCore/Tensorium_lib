38inline InteriorRegion interior_bounds(const BSSNGridSoA<T> &grid, size_t padding) {
39 size_t I0, I1, J0, J1, K0, K1;
40 grid.domain_bounds(I0, I1, J0, J1, K0, K1);
41
42 InteriorRegion region;
43 region.i0 = clamped_lower(I0, padding, I1);
44 region.j0 = clamped_lower(J0, padding, J1);
45 region.k0 = clamped_lower(K0, padding, K1);
46 region.i1 = clamped_upper(I1, padding, I0);
47 region.j1 = clamped_upper(J1, padding, J0);
48 region.k1 = clamped_upper(K1, padding, K0);
49 return region;
50}

74template <typename T> inline double compute_cfl_scale(const BSSNGridSoA<T> &grid, T dt) {
75 const double dt_val = double(dt);
76 if (!(dt_val > 0.0))
77 return 0.0;
78 const double dx_min = std::min({double(grid.dx), double(grid.dy), double(grid.dz)});
79 if (!(dx_min > 0.0))
80 return 0.0;
81
82 const size_t total = grid.alpha.st.nx_tot * grid.alpha.st.ny_tot * grid.alpha.st.nz_tot;
83 const T *p_alpha = grid.alpha.ptr();
84 const T *p_bx = grid.beta[0].ptr();
85 const T *p_by = grid.beta[1].ptr();
86 const T *p_bz = grid.beta[2].ptr();
87
88 double max_speed = 0.0;
89#pragma omp parallel for reduction(max:max_speed)
90 for (ptrdiff_t idx = 0; idx < static_cast<ptrdiff_t>(total); ++idx) {
91 const double alpha = double(p_alpha[idx]);
92 const double bx = std::abs(double(p_bx[idx]));
93 const double by = std::abs(double(p_by[idx]));
94 const double bz = std::abs(double(p_bz[idx]));
95 const double local = std::max({bx, by, bz}) + alpha;
96 if (std::isfinite(local))
97 max_speed = std::max(max_speed, local);
98 }
99
100 const double cfl = max_speed * dt_val / dx_min;
101 constexpr double cfl_ref = 0.5;
102 if (!(cfl_ref > 0.0))
103 return 0.0;
104 const double ratio = cfl / cfl_ref;
105 return std::clamp(ratio, 0.0, 1.0);
106}

151template <typename T>
152inline void compute_RicciZ4(const BSSNGridSoA<T> &G, size_t i, size_t j, size_t k, T Z4corr[6]) {
153 using namespace tensorium_RG::fd;
154
155 const double inv_12dx = 1.0 / (60.0 * G.dx);
156 const double inv_12dy = 1.0 / (60.0 * G.dy);
157 const double inv_12dz = 1.0 / (60.0 * G.dz);
158 const ptrdiff_t sx = G.alpha.st.sx;
159 const ptrdiff_t sy = G.alpha.st.sy;
160
161 const size_t idx = G.alpha.idx(i, j, k);
162
163 const T chi = G.chi.ptr()[idx];
164 const T inv_chi = T(1) / chi;
165
166 T gamma_phys[3][3];
167 T gamma_phys_inv[3][3];
168
169 const T *p_gamma[6];
170 const T *p_gamma_inv[6];
171 for (int s = 0; s < 6; ++s) {
172 p_gamma[s] = G.gamma_tilde[s].ptr() + idx;
173 p_gamma_inv[s] = G.gamma_tilde_inv[s].ptr() + idx;
174 }
175
176 for (int s = 0; s < 6; ++s) {
177 int row = 0;
178 int col = 0;
179 sym_index_to_pair(s, row, col);
180 const T gt = *p_gamma[s];
181 const T gt_inv = *p_gamma_inv[s];
182 const T g_phys = gt * inv_chi;
183 const T g_phys_inv = gt_inv * chi;
184 gamma_phys[row][col] = g_phys;
185 gamma_phys[col][row] = g_phys;
186 gamma_phys_inv[row][col] = g_phys_inv;
187 gamma_phys_inv[col][row] = g_phys_inv;
188 }
189
190 const T *p_chi = G.chi.ptr() + idx;
191 const T d_chi[3] = {Dx_ptr(p_chi, sx, inv_12dx), Dy_ptr(p_chi, sy, inv_12dy),
192 Dz_ptr(p_chi, inv_12dz)};
193
194 T d_g_phys[3][3][3];
195 for (int s = 0; s < 6; ++s) {
196 int row = 0;
197 int col = 0;
198 sym_index_to_pair(s, row, col);
199 const T *p_g = p_gamma[s];
200 const T d_gt_x = Dx_ptr(p_g, sx, inv_12dx);
201 const T d_gt_y = Dy_ptr(p_g, sy, inv_12dy);
202 const T d_gt_z = Dz_ptr(p_g, inv_12dz);
203 const T g_val = gamma_phys[row][col];
204 const T val_x = (d_gt_x - g_val * d_chi[0]) * inv_chi;
205 const T val_y = (d_gt_y - g_val * d_chi[1]) * inv_chi;
206 const T val_z = (d_gt_z - g_val * d_chi[2]) * inv_chi;
207 d_g_phys[0][row][col] = val_x;
208 d_g_phys[0][col][row] = val_x;
209 d_g_phys[1][row][col] = val_y;
210 d_g_phys[1][col][row] = val_y;
211 d_g_phys[2][row][col] = val_z;
212 d_g_phys[2][col][row] = val_z;
213 }
214
215 T gamma_conn[3][3][3];
216 for (int up = 0; up < 3; ++up)
217 for (int lo1 = 0; lo1 < 3; ++lo1)
218 for (int lo2 = 0; lo2 < 3; ++lo2) {
219 T sum = T(0);
220 for (int m = 0; m < 3; ++m)
221 sum += gamma_phys_inv[up][m] *
222 (d_g_phys[lo1][lo2][m] + d_g_phys[lo2][lo1][m] -
223 d_g_phys[m][lo1][lo2]);
224 gamma_conn[up][lo1][lo2] = T(0.5) * sum;
225 }
226
227 const T *p_Z[3] = {G.Z[0].ptr() + idx, G.Z[1].ptr() + idx, G.Z[2].ptr() + idx};
228 const T z_contra[3] = {*p_Z[0], *p_Z[1], *p_Z[2]};
229
230 T Z_cov[3] = {T(0), T(0), T(0)};
231 for (int row = 0; row < 3; ++row)
232 for (int m = 0; m < 3; ++m)
233 Z_cov[row] += gamma_phys[row][m] * z_contra[m];
234
235 T dZ_contra[3][3];
236 for (int comp = 0; comp < 3; ++comp) {
237 const T *p_vec = p_Z[comp];
238 dZ_contra[0][comp] = Dx_ptr(p_vec, sx, inv_12dx);
239 dZ_contra[1][comp] = Dy_ptr(p_vec, sy, inv_12dy);
240 dZ_contra[2][comp] = Dz_ptr(p_vec, inv_12dz);
241 }
242
243 T partial_Zcov[3][3];
244 for (int dir = 0; dir < 3; ++dir)
245 for (int b = 0; b < 3; ++b) {
246 T sum = T(0);
247 for (int m = 0; m < 3; ++m) {
248 sum += d_g_phys[dir][b][m] * z_contra[m];
249 sum += gamma_phys[b][m] * dZ_contra[dir][m];
250 }
251 partial_Zcov[dir][b] = sum;
252 }
253
254 T covZ[3][3];
255 for (int dir = 0; dir < 3; ++dir)
256 for (int b = 0; b < 3; ++b) {
257 T val = partial_Zcov[dir][b];
258 for (int m = 0; m < 3; ++m)
259 val -= gamma_conn[m][dir][b] * Z_cov[m];
260 covZ[dir][b] = val;
261 }
262
263 T divZ = T(0);
264 for (int dir = 0; dir < 3; ++dir)
265 for (int b = 0; b < 3; ++b)
266 divZ += gamma_phys_inv[dir][b] * covZ[dir][b];
267
268 const T two_thirds = T(2) / T(3);
269 for (int s = 0; s < 6; ++s)
270 Z4corr[s] = T(0);
271
272 for (int a = 0; a < 3; ++a)
273 for (int b = a; b < 3; ++b) {
274 const int s = tensorium_RG::fd::sym6(a, b);
275 const T sym = covZ[a][b] + covZ[b][a];
276 const T val = sym - two_thirds * gamma_phys[a][b] * divZ;
277 Z4corr[s] = val;
278 }
279}
