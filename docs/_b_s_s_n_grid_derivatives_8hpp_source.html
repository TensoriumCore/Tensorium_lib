126template <typename Field>
127inline double Dxy4(const Field &f, size_t i, size_t j, size_t k, double dx, double dy) {
128 double sum = 0.0;
129
130 const int idx[4] = {-2, -1, 1, 2};
131 const double w[4] = {-1.0, 8.0, -8.0, 1.0};
132
133 for (int a = 0; a < 4; ++a) {
134 for (int b = 0; b < 4; ++b) {
135 sum += w[a] * w[b] * get(f, i + idx[a], j + idx[b], k);
136 }
137 }
138
139 return sum / (144.0 * dx * dy);
140}

184template <typename Field>
185inline double KO6_axis(const Field &f, size_t i, size_t j, size_t k, Axis axis) {
186 auto sample = [&](int offset) {
187 if (axis == X)
188 return get(f, i + offset, j, k);
189 if (axis == Y)
190 return get(f, i, j + offset, k);
191 return get(f, i, j, k + offset);
192 };
193 const double s = (sample(-3) - 6.0 * sample(-2) + 15.0 * sample(-1) - 20.0 * sample(0) +
194 15.0 * sample(1) - 6.0 * sample(2) + sample(3)) /
195 64.0;
196 return s;
197}
