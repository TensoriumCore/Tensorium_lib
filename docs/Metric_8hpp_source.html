208 void compute_kerr_schild(const tensorium::Vector<T>& X, tensorium::Tensor<T, 2>& g) const {
209 assert(X.size() == 4);
210 const T x = X(1), y = X(2), z = X(3);
211
212 const T r = kerr_schild_radius(x, y, z);
213 const T cos_theta = (r > 1e-14) ? z / r : 0.0;
214 const T denom = r * r + a * a * cos_theta * cos_theta;
215 const T H = (denom > 1e-14) ? (M * r) / denom : 0.0;
216
217 const T denom_vec = r * r + a * a;
218 T lx = (denom_vec > 1e-14) ? (r * x + a * y) / denom_vec : 0.0;
219 T ly = (denom_vec > 1e-14) ? (r * y - a * x) / denom_vec : 0.0;
220 T lz = (r > 1e-14) ? z / r : 0.0;
221
