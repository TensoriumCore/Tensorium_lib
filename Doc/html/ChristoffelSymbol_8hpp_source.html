115 morpheus::Tensor<T, 2> gh({dim, dim}), gl({dim, dim});

118 Xh = X;
119 Xl = X;

121 Xl(mu) -= h;

126 for (size_t nu = 0; nu < dim; ++nu)
127 for (size_t lam = 0; lam < dim; ++lam)
128 d_metric(lam, nu, mu, 0) = (gh(lam, nu) - gl(lam, nu)) / (T(2) * h);
129 }

132 tmp.fill(T(0));
133 for (size_t lam = 0; lam < dim; ++lam)

136 tmp(lam, nu, mu, 0) = T(0.5) * (
137 d_metric(nu, lam, mu, 0) +
