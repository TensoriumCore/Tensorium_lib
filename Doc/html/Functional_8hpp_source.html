176/*
177 * @brief Matrix multiplication
178 * @param A First matrix
179 * @param B Second matrix
180 * @return Resulting matrix
181 * @note Assumes A.cols() == B.rows()
182 * @note Uses SIMD for optimization
183 * @note OpenMP parallelization for large matrices, for small matrices, it is not worth the overhead
184 * @note (For matrix 4x4, 8x8 and 16x16, there's a SIMD fallback to ensure L1 and L2 cache storage)
185 */
186
187 template <typename T>
188 Matrix<T> mul_mat(const Matrix<T>& A, const Matrix<T>& B) {
189 if (A.rows == 3 && A.cols == 3 && B.rows == 3 && B.cols == 3) {
190 const MatrixKernel<T> kernelA(A);
191 return kernelA.mul_mat3x3(B);
192 }
193 if (A.rows == 4 && A.cols == 4 && B.rows == 4 && B.cols == 4) {
194 const MatrixKernel<T> kernelA(A);
195 return kernelA.mul_mat4x4(B);
196 }
197 return A.mul_mat(B);
198 }
