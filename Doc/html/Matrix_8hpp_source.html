392 inline Matrix<K> inverse() const {
393 if (rows != cols)
394 throw std::invalid_argument("Matrix must be square");
395
396 const auto n = rows;
397 Matrix<K> M(n, n);
398 Matrix<K> Inv(n, n);
399
400 for (auto i = decltype(n)(0); i < n; ++i) {
401 for (auto j = decltype(n)(0); j < n; ++j) {
402 M(i, j) = operator()(i, j);
403 Inv(i, j) = (i == j) ? K(1) : K(0);
404 }
405 }
406
407 using SimdT = simd::SimdTraits<K, DefaultISA>;
408 for (auto i = decltype(n)(0); i < n; ++i) {
409 auto piv = i;
410 auto maxv = MathsUtils::_abs(M(i, i));
411 for (auto r = i + 1; r < n; ++r) {
412 auto v = MathsUtils::_abs(M(r, i));
413 if (v > maxv) { maxv = v; piv = r; }
414 }
415 if (maxv < static_cast<K>(1e-6))
416 throw std::runtime_error("Matrix is singular or nearly singular.");
417
418 if (piv != i) {
419 M.swap_rows(i, piv);
420 Inv.swap_rows(i, piv);
421 }
422
423 auto diag = M(i, i);
424 auto diag_inv = K(1) / diag;
425 for (auto j = 0u; j < n; ++j) {
426 M(i, j) *= diag_inv;
427 Inv(i, j) *= diag_inv;
428 }
429
430#pragma omp parallel for schedule(dynamic, UNROLL)
431 for (auto j = 0u; j < n; ++j) {
432 if (j != i) {
433 auto f = M(j, i);
434 for (auto k = 0u; k < n; ++k) {
435 M(j, k) -= f * M(i, k);
436 Inv(j, k) -= f * Inv(i, k);
437 }
438 }
439 }
440 }

442 return Inv;

449

452 using SimdT = simd::SimdTraits<K, DefaultISA>;
453 const size_t simd_width = SimdT::width;

456 for (size_t j = 0; j < n; ++j)
457 M(i, j) = operator()(i, j);
458
459 K det_sign = K(1);
460
461 for (size_t i = 0; i < n; ++i) {
462 size_t piv = i;
463 auto maxv = MathsUtils::_abs(M(i, i));
464 for (size_t r = i + 1; r < n; ++r) {
465 auto v = MathsUtils::_abs(M(r, i));
466 if (v > maxv) { maxv = v; piv = r; }
467 }
468 if (maxv < static_cast<K>(1e-12))
469 return K(0);
470
471 if (piv != i) {
472 M.swap_rows(i, piv);
473 det_sign = -det_sign;
474 }
475
476 for (size_t j = i + 1; j < n; ++j) {
477 auto f = M(j, i) / M(i, i);
478 M(j, i) = K(0);
479
480 auto f_vec = SimdT::set1(-f);
481 size_t k = i + 1;
482
483 for (; k + simd_width - 1 < n; k += simd_width) {
484 auto mjk = SimdT::load(&M(j, k));
485 auto mik = SimdT::load(&M(i, k));
486 mjk = SimdT::fmadd(f_vec, mik, mjk);
487 SimdT::store(&M(j, k), mjk);
488 }

490 M(j, k) -= f * M(i, k);
491 }
492 }
493 }

496 for (size_t i = 0; i < n; ++i)
497 det *= M(i, i);

504 Matrix<K> M(*this);
505 const size_t m = rows;
506 const size_t n = cols;
507 size_t r = 0;
508
509 for (size_t col = 0; col < n; ++col) {
510 size_t pivot_row = r;
511 for (size_t i = r; i < m; ++i) {
512 if (MathsUtils::_abs(M(i, col)) > std::abs(M(pivot_row, col)))
513 pivot_row = i;
514 }
515

517 continue;
518
519 if (pivot_row != r)
520 M.swap_rows(pivot_row, r);
521
522 for (size_t i = r + 1; i < m; ++i) {
523 auto f = M(i, col) / M(r, col);
524 M(i, col) = 0;
525 for (size_t j = col + 1; j < n; ++j)
526 M(i, j) -= f * M(r, j);
527 }

530 }
